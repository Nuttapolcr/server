//=============================================================
// Refill Auto (rAthena)
//  - script ลับ (ไม่มี NPC ให้คุย) วน loop อ่าน donate.status = 1
//  - attachrid(account_id) แจกเมื่อผู้เล่นออนไลน์ ? status = 2
//
// setarray .get_cash_amount[จำนวนบาท], Cash, BonusPoint, qtyไอเทม1, qtyไอเทม2, ...
//   [0] = #CASHPOINTS
//   [1] = #BONUSPOINT (ถ้า .@bonus_item = 0) หรือจำนวนไอเทมโบนัส (ถ้า .@bonus_item > 0)
//   [2+] = จำนวนไอเทมใน .@extra_item[] ตามลำดับ
//
// แพ็กเกจตรง config/refill.php
//=============================================================
-	script	Refill_Auto	-1,{
end;
OnInit:
	bindatcmd "testpromo22", strnpcinfo(3)+"::OnTestPromo22", 99, 99;
	// ซิงก์ยอดเติมย้อนหลังทั้งหมดที่แจกสำเร็จแล้ว แยกตาม account_id
	// rank = 0 เพื่อให้ระบบ VIP Rank คำนวณใหม่จาก YAML ตอนบัญชีล็อกอิน
	query_sql(
		"INSERT INTO `vip_rank_account` (`account_id`,`donation_total`,`rank`) "+
		"SELECT `account_id`,SUM(`amount`),0 FROM `donate` "+
		"WHERE `status` = 2 AND `amount` > 0 AND `account_id` > 0 GROUP BY `account_id` "+
		"ON DUPLICATE KEY UPDATE `donation_total` = VALUES(`donation_total`), `rank` = 0"
	);

	set .@bonus_item, 0;
	// 0 = ค่า index+1 เป็น Bonus Point (#BONUSPOINT)
	// >0 = แจกเป็นไอเทม (เช่น 501) ตามจำนวน index+1 แบบสคริปต์เก่า

	setarray .@extra_item[0], 0;
	// ไอเทมพิเศษเพิ่มต่อแพ็ก — ใส่ item id เช่น 502,503 หรือ 0 = ไม่แจก

	// [บาท], Cash, Bonus/BonusItemQty, extra0, extra1, ...
	setarray .get_cash_amount[50],    500,    0, 0, 0, 0;
	setarray .get_cash_amount[100],   1000,   0, 0, 0, 0;
	setarray .get_cash_amount[300],   3000,   300, 0, 0, 0;
	setarray .get_cash_amount[500],   5000,   550, 0, 0, 0;
	setarray .get_cash_amount[1000],  10000,  1200, 0, 0, 0;
	setarray .get_cash_amount[1500],  15000,  2000, 0, 0, 0;
	setarray .get_cash_amount[3000],  30000,  4500, 0, 0, 0;
	setarray .get_cash_amount[5000],  50000,  7100, 0, 0, 0;
	setarray .get_cash_amount[10000], 100000, 15000, 0, 0, 0;

	// โปรโมชั่นเติมเงิน 3 ส.ค. 2569 09:00 - 9 ส.ค. 2569 23:59 (เวลาเซิร์ฟเวอร์)
	// รางวัลแจกต่อ 1 รายการเติมเงิน และทำงานก่อนเปลี่ยน donate.status เป็น 2
	setarray .promo_amount[0], 300, 500, 1000, 3000, 5000, 10000;
	setarray .promo_bsb[0], 5, 9, 18, 54, 90, 180;
	setarray .promo_pet_egg[0], 3, 5, 12, 40, 75, 180;
	setarray .promo_bloody_branch[0], 0, 2, 4, 12, 20, 40;
	setarray .promo_refine_5[0], 0, 0, 1, 3, 5, 10;
	setarray .promo_refine_10[0], 0, 0, 0, 1, 3, 5;
	setarray .promo_refine_15[0], 0, 0, 0, 0, 1, 3;

	while (1) {
		query_sql "SELECT `id`,`amount`,`account_id` FROM `donate` WHERE `status` = 1 ORDER BY `id` ASC LIMIT 20",
			.donate_id, .donate_amount, .donate_account_id;
		sleep 500;

		if (getarraysize(.donate_id) > 0) {
			for (set .i, 0; .i < getarraysize(.donate_id); set .i, .i + 1) {
				if (.donate_amount[.i] <= 0)
					continue;
				if (attachrid(.donate_account_id[.i]) != 1)
					continue;

				set .@amt, .donate_amount[.i];
				set .@rank_ok, vip_rank_update(.@amt, 1, "refill:" + .donate_id[.i], "npc_refill_auto");
				if (.@rank_ok != 1) {
					dispbottom "[Refill] ไม่สามารถบันทึกยอดสะสม Rank ได้ กรุณาติดต่อทีมงาน", 0xFF0000;
					detachrid;
					continue;
				}

				set .@cash, .get_cash_amount[.@amt] * 2;
				if (.@cash > 0) {
					set #CASHPOINTS, #CASHPOINTS + .@cash;
					dispbottom "[Refill] ได้รับ " + callfunc("F_InsertComma", .@cash) + " Cash รวม " + callfunc("F_InsertComma", #CASHPOINTS) + " Cash", 0xAEFF00;
				}

				set .@bonus_val, .get_cash_amount[(.@amt + 1)];
				if (.@bonus_val > 0) {
					if (.@bonus_item > 0) {
						getitem .@bonus_item, .@bonus_val;
						dispbottom "[Refill] ได้รับไอเทมโบนัส " + getitemname(.@bonus_item) + " x" + callfunc("F_InsertComma", .@bonus_val), 0xFFFB00;
					} else {
						set #CASHPOINTS, #CASHPOINTS + .@bonus_val;
						dispbottom "[Refill] ได้รับโบนัส " + callfunc("F_InsertComma", .@bonus_val) + " Bonus Point", 0xFFFB00;
					}
				}

				// ใช้เวลาของ map-server เพื่อไม่ให้รับรางวัลนอกช่วงโปรโมชั่น
				.@promo_now = gettime(DT_YEAR) * 100000000 + gettime(DT_MONTH) * 1000000 + gettime(DT_DAYOFMONTH) * 10000 + gettime(DT_HOUR) * 100 + gettime(DT_MINUTE);
				if (.@promo_now <= 202608092359) {
					.@promo_index = inarray(.promo_amount[0], .@amt);
					if (.@promo_index >= 0) {
						getitem 6635, .promo_bsb[.@promo_index];
						getitem 22571, .promo_pet_egg[.@promo_index];
						getitem 12103, .promo_bloody_branch[.@promo_index];
						if (.promo_refine_5[.@promo_index] > 0)
							getitem 1030015, .promo_refine_5[.@promo_index];
						if (.promo_refine_10[.@promo_index] > 0)
							getitem 1030016, .promo_refine_10[.@promo_index];
						if (.promo_refine_15[.@promo_index] > 0)
							getitem 1030017, .promo_refine_15[.@promo_index];
						dispbottom "[โปรโมชั่นเติมเงิน] ได้รับรางวัลพิเศษตามยอดเติม " + callfunc("F_InsertComma", .@amt) + " บาท", 0x66CCFF;
					}
				}

				if (getarraysize(.@extra_item) > 0 && .@extra_item[0] > 0) {
					for (set .b, 0; .b < getarraysize(.@extra_item); set .b, .b + 1) {
						set .@qty, .get_cash_amount[(.@amt + 2 + .b)];
						if (.@qty > 0) {
							getitem .@extra_item[.b], .@qty;
							dispbottom "[Refill] ได้รับ " + getitemname(.@extra_item[.b]) + " x" + callfunc("F_InsertComma", .@qty), 0xFFFFFF;
						}
					}
				}

				query_sql "UPDATE `donate` SET `status` = 2, `updated_at` = NOW() WHERE `id` = " + .donate_id[.i] + " AND `status` = 1 LIMIT 1";
				detachrid;
			}
			deletearray .donate_id[0], getarraysize(.donate_id);
			deletearray .donate_amount[0], getarraysize(.donate_amount);
			deletearray .donate_account_id[0], getarraysize(.donate_account_id);
		}
		sleep 5000;
	}
	end;

OnTestPromo22:
	if (.@atcmd_numparameters < 1) {
		dispbottom "วิธีใช้: @testpromo22 <300|500|1000|3000|5000|10000>";
		end;
	}

	.@amt = atoi(.@atcmd_parameters$[0]);
	.@promo_index = inarray(.promo_amount[0], .@amt);
	if (.@promo_index < 0) {
		dispbottom "ยอดทดสอบไม่ถูกต้อง: ใช้ได้เฉพาะ 300, 500, 1000, 3000, 5000 หรือ 10000";
		end;
	}

	getitem 6635, .promo_bsb[.@promo_index];
	getitem 22571, .promo_pet_egg[.@promo_index];
	getitem 12103, .promo_bloody_branch[.@promo_index];
	if (.promo_refine_5[.@promo_index] > 0)
		getitem 1030015, .promo_refine_5[.@promo_index];
	if (.promo_refine_10[.@promo_index] > 0)
		getitem 1030016, .promo_refine_10[.@promo_index];
	if (.promo_refine_15[.@promo_index] > 0)
		getitem 1030017, .promo_refine_15[.@promo_index];
	dispbottom "[ทดสอบโปรโมชั่นเติมเงิน] ได้รับรางวัลของยอด " + callfunc("F_InsertComma", .@amt) + " บาทแล้ว", 0x66CCFF;
	end;
}
