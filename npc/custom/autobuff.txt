	set @eac, eaclass();

	mes .@menu$;
	
	if((@eac&EAJ_BASEMASK) != EAJ_ACOLYTE
		&& (@eac&EAJ_UPPERMASK) != EAJ_ALCHEMIST
		&& (@eac&EAJ_UPPERMASK) != EAJ_CRUSADER
		&& (@eac&EAJ_UPPERMASK) != EAJ_BARDDANCER
		) {
		mes "ระบบนี้ใช้ได้เฉพาะอาชีพ Acolyte, Crusader, Bard, Dancer และ Alchemist เท่านั้น";
		close;
	}

		mes "คุณต้องอยู่ในปาร์ตี้ก่อนจึงจะใช้ระบบ AutoBuff ได้";
		close;

	mes "^777777---------- การฟื้นฟู ----------^000000";

	mes "รักษาอัตโนมัติ: " + (ab_getautobuffint(0,0) > 0 ? "^008000เปิด^000000" : "^FF0000ปิด^000000");

	mes "ยา HP/SP    : " + (ab_getautobuffint(1,0) > 0 ? "^008000เปิด^000000" : "^FF0000ปิด^000000");

	// Potion Pitcher (เฉพาะ Alchemist)
	if((@eac&EAJ_UPPERMASK) == EAJ_ALCHEMIST){
		mes "Potion Pitchr: " + (ab_getautobuffint(14,0) > 0 ? "^008000เปิด^000000" : "^FF0000ปิด^000000");
	}

	mes " ";
	mes "^777777---------- การบัฟ ----------^000000";

	mes "สกิลบัฟ     : " + (ab_getautobuffint(3,0) > 0 ? "^008000เปิด^000000" : "^FF0000ปิด^000000");

	// ไอเทมบัฟ
	mes "ไอเทมบัฟ    : " + (ab_getautobuffint(5,0) > 0 ? "^008000เปิด^000000" : "^FF0000ปิด^000000");

	// ให้ความสำคัญบัฟก่อน
	mes "บัฟก่อนเสมอ : " + (ab_getautobuffint(9,0) ? "^008000เปิด^000000" : "^FF0000ปิด^000000");

	// คืนชีพ (เฉพาะ Priest)
	if(BaseJob == Job_Priest || Class == Job_Arch_Bishop || Class == Job_Arch_Bishop_T || Class == Job_Baby_Arch_Bishop || Class == Job_Cardinal){
		mes "คืนชีพ      : " + (ab_getautobuffint(2,0) ? "^008000เปิด^000000" : "^FF0000ปิด^000000");
	}

	mes " ";
	mes "^777777---------- ปาร์ตี้ & ติดตาม ----------^000000";

	// ติดตาม
	mes "ติดตาม      : " + (ab_getautobuffint(4,0) > 0 ? "^008000เปิด^000000" : "^FF0000ปิด^000000");

	// รับเข้าปาร์ตี้
	mes "รับปาร์ตี้   : " + (ab_getautobuffint(8,0) ? "^008000เปิด^000000" : "^FF0000ปิด^000000");

	// คำสั่ง PM
	switch(ab_getautobuffint(5,0)){
		case 0: mes "คำสั่ง PM   : ^008000ทั้งปาร์ตี้^000000"; break;
		case 1: mes "คำสั่ง PM   : ^FFAA00หัวหน้าเท่านั้น^000000"; break;
		case 2: mes "คำสั่ง PM   : ^FF0000ปิด^000000"; break;
		case 3: mes "คำสั่ง PM   : ^FFAA00เฉพาะสมาชิก^000000"; break;
	}

	mes " ";
	mes "^777777---------- อื่นๆ ----------^000000";

	mes "ซื้ออัตโนมัติ : " + (ab_getautobuffint(13,1) > 0 ? "^008000เปิด^000000" : "^FF0000ปิด^000000");

	// ขายอัตโนมัติ
	mes "ขายอัตโนมัติ : " + (ab_getautobuffint(11,0) > 0 ? "^008000เปิด (" + ab_getautobuffint(11,0) + " รายการ)^000000" : "^FF0000ปิด^000000");

			else

			break;
		case 2:
			goto(config_menu);
			break;
	}
	end;

autobuy_menu2:
	clear;
	.@menu$ = "[ซื้อไอเทมอัตโนมัติ]";
	mes .@menu$;
	mes "--- การตั้งค่าปัจจุบัน ---";
	mes "เลือกไอเทมที่ต้องการซื้ออัตโนมัติ";
	mes ab_getautobuffstring(13,0);
	.@autobuy_menu = select("เพิ่ม/แก้ไขไอเทม","ลบไอเทม","ล้างรายการ","[ยกเลิก]");
	switch(.@autobuy_menu){
		case 1:
			clear;
			mes .@menu$;
			mes "--- รายการไอเทมที่สามารถซื้อได้ ---";
			mes ab_getautobuffstring(13,-1);
			
			if(getarraysize(.@buy_add_menu_ids) == 0){
				clear;
				mes .@menu$;
				mes "ขออภัย ไม่มีไอเทมที่สามารถซื้อได้";
				next;
				goto(autobuy_menu2);
			}
			
			.@buy_add_menu_list$ = .@buy_add_menu_list$ + ":[ยกเลิก]";
			.@i = select(.@buy_add_menu_list$);
			
			if(.@i > getarraysize(.@buy_add_menu_ids))
				goto(autobuy_menu2);
			
			.@i--;
			.@selected_item = .@buy_add_menu_ids[.@i];
			
			do {
				clear;
				mes .@menu$;
				mes "ระบุจำนวนที่ต้องการซื้อ (1-1000)";
				input .@buy_quantity;
			} while(.@buy_quantity < 1 || .@buy_quantity > 1000);
			
			do {
				clear;
				mes .@menu$;
				mes "ซื้อเมื่อไอเทมเหลือน้อยกว่า (0-999)";
				input .@buy_trigger;
			} while(.@buy_trigger < 0 || .@buy_trigger > 999);
			
			ab_setautobuff("19;1;" + .@selected_item + ";" + .@buy_quantity + ";" + .@buy_trigger);
			break;
			
		case 2:
			clear;
			mes .@menu$;
			mes "--- เลือกไอเทมที่ต้องการลบ ---";
			mes ab_getautobuffstring(13,0);
			
			if (getarraysize(.@buy_menu_ids) == 0) {
				clear;
				mes .@menu$;
				mes "ไม่มีไอเทมในรายการซื้ออัตโนมัติ";
				next;
				goto(autobuy_menu2);
			}
			
			// แสดง menu
			.@buy_menu_list$ = .@buy_menu_list$ + ":[ยกเลิก]";
			.@i = select(.@buy_menu_list$);
			
			if(.@i > getarraysize(.@buy_menu_ids))
				goto(autobuy_menu2);
			
			.@i--;
			ab_setautobuff("19;0;" + .@buy_menu_ids[.@i] + ";0;0");
			break;
			
		case 3:
			ab_setautobuff("19;-1;0;0;0");
			break;
			
		case 4:
			goto(config_menu);
			break;
	}
	goto(autobuy_menu2);
	end;

				case 2:
					clear;

					mes "ใส่ ID ของไอเทมที่ต้องการขายอัตโนมัติ (0 เพื่อยกเลิก)";
					input .@item_id;
					
					if(.@item_id == 0)
						goto(autosell_menu2);
					
					ab_setautobuff("20;1;" + .@item_id);
					break;
					
				case 3:
					goto(autosell_menu2);
					break;
			}
			break;
			
		case 2:
			clear;
			mes .@menu$;
			mes "--- เลือกไอเทมที่ต้องการลบ ---";
			mes ab_getautobuffstring(14,0);

				clear;
				mes .@menu$;
				mes "ไม่มีไอเทมในรายการขายอัตโนมัติ";
				next;
				goto(autosell_menu2);
			}
			
			// แสดง menu
			.@sell_menu_list$ = .@sell_menu_list$ + ":[ยกเลิก]";
			.@i = select(.@sell_menu_list$);
			
			if(.@i > getarraysize(.@sell_menu_ids))
				goto(autosell_menu2);
			
			.@i--;
			ab_setautobuff("20;0;" + .@sell_menu_ids[.@i]);
			break;
			
		case 3:
			ab_setautobuff("20;-1;0");
			break;
			
		case 4:
			goto(config_menu);
			break;
	}
	goto(autosell_menu2);
	end;

	.@j = 0;

	for (.@i = 0; .@i < getskilllv(.@lheal_skill[0]); .@i++)
		.@lheal_skill2[.@i] = .@lheal_skill[0];

	if(getarraysize(.@lheal_skill2) == 0){
		clear;
		mes .@menu$;
		mes "ขออภัย คุณไม่มีสกิลรักษาที่ได้รับอนุญาต";
		next;
		goto(regeneration_menu);

	.@healskill_menu1$ = "[ยกเลิก]:" + .@healskill_menu1$;

	if(.@i == 1)
		goto(regeneration_menu);
	else

	.@heal_skilllv = .@i + 1;

		.@heal_menu = select("แก้ไข","ปิด","[ยกเลิก]");
		switch(.@heal_menu){
			case 1 :	
				do {
					clear;
					mes .@menu$;
					if(.@heal_skilllv == 5)
						mes "เลือกค่า SP % ที่จะใช้สกิล (เช่น: 10 = ใช้สกิลเมื่อ SP ต่ำกว่า 10%)";

						mes "เลือกค่า HP % ที่จะใช้สกิล (เช่น: 10 = ใช้สกิลเมื่อ HP ต่ำกว่า 10%)";

						do {
							clear;
							mes .@menu$;
							mes ab_getautobuffstring(12,0,.@lheal_skill2[.@i],.@heal_skilllv);

							if(getarraysize(.@player_heal_ids) == 0){

								mes "ขออภัย คุณเป็นคนเดียวในปาร์ตี้";
								next;
								goto(potion_pitcher_menu);
							}
				
							.@k = select( .@player_heal_list$ ) - 1;

							if(.@player_heal_state[.@player_heal_ids[.@k]])
