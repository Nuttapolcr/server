//=== ฟังก์ชั่นหลัก ========================================================
//=   arg(0): ใช้สำหรับการแสดงข้อความต้อนรับ.
//=   arg(1): ใช้สำหรับการแสดงเมนูที่จะปรากฏ.
//=   arg(2): ใช้การข้อมูลของเมนูที่จะปรากฏใน F_KafInfo.
//=   arg(3): ค่าใช้จ่ายสำหรับเปิดคลังเก็บของ
//=   arg(4): ค่าใช้จ่ายสำหรับการเช่ารถเข็น
//==========================================================================
function	script	F_Kafra	{
	callfunc "F_ClearGarbage"; // ล้างตัวแปรเก่าที่ไม่ได้ใช้
	.@welcome = getarg(0);
	.@menu_num = getarg(1);

	// ข้อความต้อนรับของพนักงานคาฟรา
	switch(.@welcome){
		default:
		case 0:
			// ข้อความปกติ (เก่า)
			mes "[Kafra Employee]";
			mes "ยืนดีต้อนรับสู่";
			mes "คาฟรา คอร์โปเรชั่น.";
			mes "คาฟรา จะอยู่เคียงข้างคุณ";
			mes "ในทุกๆที่ที่คุณไป.";
			mes "มีอะไรให้ฉันช่วยเหลือบ้างหรือเปล่า ?";
			break;
		case 1:
			// ข้อความพิเศษสำหรับเมือง Niflheim
			mes "[Kafra Employee]";
			mes "^666666ย-ยินนนนน-ด-ดีต้อนรับบบบบ";
			mes "สู้ ค-คาาาาฟราาา";
			mes "ค-คอร์-ป-โปปปปปป... เรชั่นนนนน^000000";
			break;
		case 2:
			// ข้อความคาฟราในปราสาทกิลด์
			set @GID,getcharid(2);
			mes "[Kafra Employee]";
	 		mes "ยืนดีต้อนรับสมาชิกกิลด์. ^ff0000" + GetGuildName(@GID) + "^000000.";
			mes "คาฟรา คอร์โปเรชั่น จะอยู่กับคุณในทุกๆที่.";
			break;
		case 3:
			// ข้อความพิเศษสำหรับเมือง Amatsu (เก่า)
			mes "[Kafra Employee]";
			mes "ดังนั้น, คุณได้เดินทางมาจากสถานที่ที่ไกลมากๆเพื่อศึกษาวัฒนธรรมของเรา, หรือ คุณต้องการมาเพียงแค่ชมสถานที่ ?";
			mes "ในสองกรณีนี้, ทำไมคุณไม่ลองหยุดพักสักครู่ ?";
			mes "อากาศที่น่าสดชื่น บวกกับ";
			mes "ดอกไม้ที่น่าริ่นรมย์เช่นนี้.";
			break;		
		case 4:
			// ข้อความพิเศษสำหรับเมือง Louyang และ Ayothaya (เก่า)
			mes "[Kafra Employee]";
			mes "จากที่ที่ห่างไกล";
			mes "บริการของคาฟราจะอยู่ในทุกๆที่, ที่ทำให้คุณรู้สึก";
			mes "ไม่ได้อยู่ห่างไกลจากบ้าน.";
			break;		
		case 5:
			break;
	}
	next;
	while (1) {
		deletearray @K_Menu0$[0],getarraysize(@K_Menu0$);
		if (.@welcome == 2) {
			// เมนูสำหรับคาฟราในปราสาท (ฟรี เทเลพอร์ต, ฟรี คลังเก็บของ, ฟรี รถเข็น)
			setarray @K_Menu0$[0],"~ ใช้งานคลังเก็บของ","~ ใช้งานคลังเก็บของ (กิลด์)","~ เช่ารถเข็น","~ ใช้บริการเทเลพอร์ต","~ ยกเลิก";
		} else {
			switch(.@menu_num){
				// บันทึกจุดเกิด และ เก็บของ เท่านั้น
				case 1:	setarray @K_Menu0$[0],"~ บันทึกจุดเกิด","~ ใช้งานคลังเก็บของ","~ ยกเลิก"; break;
				// เก็บของ เท่านั้น
				case 2:	setarray @K_Menu0$[0],"~ ใช้งานคลังเก็บของ","~ ยกเลิก"; break;
				// ไม่มีฟังก์ชั่นการใช้งานในส่วนของเทเลพอร์ต (ทั่วไป)
				case 3: setarray @K_Menu0$[0],"~ บันทึกจุดเกิด","~ ใช้งานคลังเก็บของ","~ เช่ารถเข็น","~ ยกเลิก"; break;
				// Case 4 สำหรับ Einbroch ไม่มีข้อความสำหรับการเทเลพอร์ต.
				// ไม่มี บันทึกจุดเกิด, หรือ เทเลพอร์ต. (ทั่วไป)
				case 5:	setarray @K_Menu0$[0],"~ ใช้งานคลังเก็บของ","~ เช่ารถเข็น","~ ยกเลิก"; break;
				// เก็บของ และ ตรวจสอบรายละเอียดข้อมูล เท่านั้น.
				case 6:	setarray @K_Menu0$[0],"~ ใช้งานคลังเก็บของ","~ ยกเลิก"; break;
				// บันทึกจุดเกิด, Storage, and Pushcart only (Kafra Warehouse)
				case 7:	setarray @K_Menu0$[0],"~ บันทึกจุดเกิด","~ ใช้งานคลังเก็บของ","~ เช่ารถเข็น","~ ยกเลิก"; break;
				// บันทึกจุดเกิด, Storage, Other Check information. (Turbo track)
				case 8: setarray @K_Menu0$[0],"~ บันทึกจุดเกิด","~ ใช้งานคลังเก็บของ","~ ยกเลิก"; break;
				// No บันทึกจุดเกิด (Rune Knight)
				case 9: setarray @K_Menu0$[0],"~ ใช้งานคลังเก็บของ","~ เช่ารถเข็น","~ ใช้บริการเทเลพอร์ต","~ ยกเลิก"; break;
				// Storage, บันทึกจุดเกิด, and Pushcart (Dewata, reorder of case 7)
				case 10: setarray @K_Menu0$[0],"~ ใช้งานคลังเก็บของ","~ บันทึกจุดเกิด","~ เช่ารถเข็น","~ ยกเลิก"; break;
				// Default message (obsolete)
				default: setarray @K_Menu0$[0],"~ บันทึกจุดเกิด","~ ใช้งานคลังเก็บของ","~ ใช้บริการเทเลพอร์ต","~ เช่ารถเข็น","~ ยกเลิก"; break;
			}
		}
		set .@menu$,"";
		for (set .@i, 0; .@i < getarraysize(@K_Menu0$); set .@i, .@i + 1)
			set .@menu$, .@menu$ + @K_Menu0$[.@i] + ":";	
		set .@j, select (.@menu$) - 1;
		if (@K_Menu0$[.@j] == "~ บันทึกจุดเกิด") {
			return;
		} else if (@K_Menu0$[.@j] == "~ ใช้งานคลังเก็บของ") {
			// ไม่เรียกเก็บค่าธรรมเนียมสำหรับคลังเก็บของกิลด์
			if(.@welcome == 2) 
				callfunc "F_KafStor",2,0,0;
			else 
				callfunc "F_KafStor",0,getarg(3),.@welcome;
			next;
		} else if (@K_Menu0$[.@j] == "~ ใช้บริการเทเลพอร์ต") {
			// แสดงประกาศของ Einbroch "ไม่มีการใช้งานฟังก์ชั่นเทเลพอร์ต".
			if(.@menu_num != 4) callfunc "F_KafTele",.@welcome;
			else {
				mes "[Kafra Employee]";
				mes "เพราะ ^FF0000ข้อจำกัด";
				mes "ของการตกลงเกี่ยวกับการขนส่ง^000000,";
				mes "คาฟรา คอร์โปเรชั่น ไม่สามารถที่จะ";
				mes "ให้บริการเกี่ยวกับการเทเลพอร์ตในส่วนนี้ได้";
				mes "ในสาธารณรัฐ Schwaltzvalt.";
				next;
				mes "[Kafra Employee]";
				mes "เราขอให้คุณช่วยใช้บริการจาก";
				mes "Airship Service";
				mes "ขอบคุณสำหรับความร่วมมือ";
				mes "และ ความเข้าใจ.";
			}
			next;
		} else if (@K_Menu0$[.@j] == "~ เช่ารถเข็น") {
			if(callfunc("F_KafCart",.@welcome,getarg(4)) == 1) 
				next;
		//} else if (@K_Menu0$[.@j] == "~ ตรวจสอบรายละเอียดข้อมูล") {
		//	callfunc "F_KafInfo",getarg(2);
		//	next;
		} else if (@K_Menu0$[.@j] == "~ ยกเลิก"){
			callfunc "F_KafEnd",.@welcome,0;
			end;
		} else if (@K_Menu0$[.@j] == "~ ใช้งานคลังเก็บของ (กิลด์)") {
			callfunc "F_KafStor",1,0;
			next;
		}
	}
}

// ฟังก์ชั่นคลังเก็บของ
//===================================================
function	script	F_KafStor	{
	.@type = getarg(0);
	.@fee = getarg(1);

	// ไม่สามารถใช้งานในส่วนของคลังเก็บของกิลด์
	if(.@type == 1){
		if(guildopenstorage()){
			mes "[Kafra Employee]";
			mes "ฉันขออภัยด้วย, แต่สมาชิกกิลด์อื่นกำลังใช้งานคลังเก็บของของกิลด์อยู่";
			mes "ในตอนนี้.  โปรดรอจนกว่าสมาชิกคนอื่นใช้เสร็จก่อน.";
			close2;
			cutin "", 255;
			end;
		}
		cutin "", 255;
		close;
	}
	// ไม่สามารถใช้งานในส่วนของคลังเก็บของปกติ (ทักษะสกิลพื้นฐานไม่เพียงพอ)
	if(!callfunc("F_CanOpenStorage")){
		mes "[Kafra Employee]";
		// ข้อความพิเศษสำหรับเมือง Niflheim
		if (getarg(2) == 1) {
			mes "^666666ส-เส-เสียใจด้วยยย,";
			mes "ค-คุณ";
			mes "ต้องงง... ร-เรียนรู้";
			mes "ท-ทักษะ. . . ส-สกิล. . . อ-อย่างน้อยที่เลเวล 6...^000000";
			return;
		}
		mes "ฉันขออภัยด้วย, แต่คุณ";
		mes "ต้องเรียนรู้ทักษะของ Novice's";
		mes "อย่างน้อยเลเวล 6 ขึ้นไป";
		mes "เพื่อใช้งานบริการคลังเก็บของ.";
		return;
	}
	// การเข้าถึงคลังเก็บของปกติ
	if(.@type != 2){
		// ใช้งาน "Free Ticket สำหรับการใช้งานคลังเก็บของ" ถ้ามี.
		if(countitem(7059)) delitem 7059,1;
		else {
			if(Zeny<.@fee){
				mes "[Kafra Employee]";
				// ข้อความพิเศษสำหรับเมือง Niflheim
				if (getarg(2) == 1) {
					percentheal -50,-50;
					mes "^666666เงินนนนนน...";
					mes "ต้องมี ง-เงินนน...!";
					mes "จ-จำนวนนนน 150... zeny...";
					mes "อ๊ากกก! ล-เลือดดดด~!^000000";
					return;
				}
				// ข้อความปกติ
				mes "ฉันขออภัยด้วย, แต่ดูเหมือนว่า";
				mes "จำนวนเงินของคุณไม่เพียงพอ";
				mes "สำหรับการใช้งานบริการคลังเก็บของ.";
				mes "ค่าธรรมเนียมคือ "+.@fee+" zeny.";
				return;
			}
			set Zeny, Zeny-.@fee;
			set RESRVPTS, RESRVPTS + (.@fee/5);
		}
	}

	if (.@type != 2) {
		// ข้อความพิเศษสำหรับเมือง Niflheim
		if (getarg(2) == 1) {
			percentheal 0,-10;
			mes "[Kafra Employee]";
			for (set .@i, 0; .@i < 5; set .@i, .@i + 1)
				mes "^666666ขอบ คุณ.. สำหรับ... การใช้งาน...^000000";
		}
		// ข้อความปกติ
		else {
			mes "[Kafra Employee]";
			mes "ตอนนี้, ฉันได้เปิด";
			mes "คลังเก็บของสำหรับคุณแล้ว.";
			mes "ขอบคุณสำหรับการใช้งาน";
			mes "จากบริการคาฟรา.";
		}
	}
	callfunc("F_CheckKafCode");	//ฟังก์ชั่นตรวจสอบรหัสผ่านถ้าถูกตั้งไว้
	close2;
	openstorage;
	cutin "", 255;
	end;
}

// ฟังก์ชั่น เทเลพอร์ต
//===================================================
function	script	F_KafTele	{
	mes "[Kafra Employee]";
	mes "โปรดเลือก";
	mes "จุดหมายปลายทางที่คุณจะไป.";
	next;
	set .@menu$, "";
	for (set .@i, 0; .@i <= 6; set .@i, .@i + 1)
		set .@menu$, .@menu$ + @wrpC$[.@i] + ":";
	set .@j, select (.@menu$) - 1;
	if (@wrpC$[.@j] == "~ ยกเลิก") return;
	// ใช้ "Free Ticket สำหรับคาฟราเทเลพอร์ต" ถ้ามี.
	// จะไม่ถูกใช้ถ้าหากอยู่ในปราสาทกิลด์
	if(countitem(7060) > 0 && getarg(0) != 2) delitem 7060,1;
	else {
		if (Zeny<@wrpP[.@j]){
			mes "[Kafra Employee]";
			mes "ฉันขออภัยด้วย,แต่ดูเหมือนว่า";
			mes "คุณมีจำนวนเงินไม่เพียงพอ";
			mes "สำหรับการใช้บริการเทเลพอร์ต";
			mes "ค่าธรรมเนียมจุดหมายปลายทาง";
			mes ""+@wrpD$[.@j]+" คือ "+callfunc("F_InsertComma",@wrpP[.@j])+" zeny.";
			close2;
			cutin "", 255;
			end;
		}
		set Zeny, Zeny - @wrpP[.@j];
		set RESRVPTS, RESRVPTS + (@wrpP[.@j]/16);
	}
	cutin "", 255;
	if (@wrpD$[.@j] == "Al De Baran") warp "aldebaran",168,112;
	else if (@wrpD$[.@j] == "Alberta") warp "alberta", 117, 56;
	else if (@wrpD$[.@j] == "Comodo") warp "comodo", 209, 143;
	else if (@wrpD$[.@j] == "Comodo Pharos Beacon") warp "cmd_fild07", 127, 134;

function	script	F_KafCart	{
	// ตรวจสอบว่าผู้เล่นได้เป็นอาชีพ Merchant หรือไม่
	if(BaseClass != Job_Merchant){
		mes "[Kafra Employee]";
		mes "ฉันขออภัยด้วย, แต่";
		mes "รถเข็น มีไว้บริการสำหรับ";
		if (checkre(0))
			mes "อาชีพ Merchant เท่านั้น.";
		else {
			mes "อาชีพ Merchants กับ";
			mes "Blacksmiths และ Alchemists เท่านั้น";
		}
		return 1;
	}
	// ตรวจสอบว่าผู้เล่นมีรถเข็นอยู่หรือไม่
	else if(checkcart() == 1){
		mes "[Kafra Employee]";
		mes "คุณมีรถเข็นอยู่แล้ว";
		mes "น่าเสียดายที่เราไม่สามารถ";
		mes "ให้เช่าได้มากกว่าหนึ่ง";
		mes "ในแต่ละครั้ง.";
		return 1;
	}
	//ตรวจสอบว่าผู้เล่นเรียนรู้ทักษะสกิล Pushcart หรือไม่
	else if(getskilllv("MC_PUSHCART") == 0) {
		mes "[Kafra Employee]";
		mes "คุณจะสามารถเช่ารถเข็นได้หลังจากเรียนรู้ทักษะสกิล \"Push Cart\".";
		return 1;
	}
	// ใช้ "Free Ticket สำหรับคาฟราเช่ารถเข็น" ถ้ามี.
	// จะไม่ถูกใช้ถ้าหากอยู่ในปราสาทกิลด์.
	if(countitem(7061) > 0 && getarg(0) != 2) delitem 7061,1;
	else {
		mes "[Kafra Employee]";
		mes "สำหรับการเช่ารถเข็น";
		mes "ค่าธรรมเนียม "+getarg(1)+" zeny.";
		mes "คุณต้องการเช่ารถเข็นหรือไม่ ?";
		next;
		if (select("~ เช่ารถเข็น.:~ ยกเลิก") == 2) return 0;
			if(Zeny<getarg(1)){
				mes "[Kafra Employee]";
				mes "ฉันขออภัยด้วย, แต่ดูเหมือนว่า";
				mes "คุณมีจำนวนเงินไม่เพียงพอ";
				mes "สำหรับเช่ารถเข็น";
				mes "ค่าธรรมเนียมคือ "+getarg(1)+" zeny.";
				return 1;
			}
			set Zeny,Zeny-getarg(1);
			set RESRVPTS, RESRVPTS + 48;
	}
	setcart;
	return 1;
}

// ฟังก์ชั่นตรวจสอบรหัสผ่าน
//===================================================
function	script	F_CheckKafCode	{
	if(!#kafra_code) return;
	mes "โปรดใส่รหัสผ่านคลังเก็บของ";
	set @code_,0;
	input @code_;
	if(@code_ != #kafra_code) {
		dispbottom "รหัสผ่านผิดพลาด.";
		close2;
		cutin "",255;
		end;
	}
	set @kafcode_try,0;
	set @code_,0;
	return;
}

// === ฟังก์ชั่น ตั้งค่า / เปลี่ยน / ลบรหัสผ่าน ===
// getarg(0): ชื่อ NPC
// getarg(1): ชื่อบริษัท
//=======================================================
function	script	F_SetKafCode	{
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	if(#kafra_code) {
		mes "ขณะนี้คลังเก็บของของคุณกำลังถูกป้องกันด้วยรหัสผ่าน. คุณต้องการที่จะทำอะไร ?";
		next;
		switch (select("~ เปลี่ยนรหัสผ่าน -> 5,000z:~ ลบรหัสผ่าน -> 1,000z:~ ยกเลิก")) {
			case 1:
				mes .@npc_name$;
				mes "ในขั้นตอนแรก, โปรดใส่ ^0000FFรหัสผ่านเก่า^000000.";
				set @code,callfunc("F_EntKafCode");
				if(!@code || @code != #kafra_code) {
					mes "รหัสผ่านผิดพลาด. คุณไม่สามารถตั้งค่ารหัสผ่านใหม่ได้.";
					emotion ET_SCRATCH;
					break;
				}
				next;
				callsub S_SET,.@npc_name$,.@comp_name$;
				break;
			case 2:
				mes .@npc_name$;
				mes "โปรด,กรอกรหัสผ่านของคุณก่อนที่จะทำการลบ.";
				set @code,callfunc("F_EntKafCode");
				if(!@code) {
					mes "รหัสผ่านยังไม่ถูกลบ.";
					emotion ET_SCRATCH;
					break;
				}
				next;
				mes .@npc_name$;
				if(Zeny < 1000) {
					mes "จำนวนเงินของคุณไม่เพียงพอ.";
					emotion ET_MONEY;
					break;
				}
				set Zeny,Zeny-1000;
				if(@code == #kafra_code) {
					set #kafra_code,0;
					mes "การลบรหัสผ่านเสร็จสมบูรณ์.";
					mes "ขอบคุณที่ใช้บริการจาก คาฟรา "+.@comp_name$+".";
					emotion ET_THANKS;
				} else {
					mes "รหัสผ่านผิดพลาด. เราจะไม่คืนจำนวนเงิน 1000z.";
					mes "โปรด, ใส่รหัสผ่านให้ถูกต้องในครั้งถัดไป.";
					emotion ET_SORRY;
				}
				break;
			case 3:
				break;
		}
	} else {
		mes .@comp_name$+" ภูมิใจนำเสนอบริการใหม่ :";
		mes "การป้องกันคลังเก็บของของคุณด้วยรหัสผ่าน.";
		next;
		switch (select("~ ตั้งค่ารหัสผ่าน -> 5,000z:~ ยกเลิก")) {
			case 1: callsub S_SET,.@npc_name$,.@comp_name$; break;
			case 2: break;		
		}
	}
	close2;
	cutin "",255;
	end;

S_SET:
	mes .@npc_name$;
	mes "โปรด ^FF0000กรอกรหัสผ่านใหม่^000000 เพื่อป้องกันมิจฉาชีพเข้าถึงคลังสินค้าของคุณ.";
	set @code,callfunc("F_EntKafCode");
	if(!@code) {
		mes "รหัสผ่านยังไม่ถูกเปลี่ยน.";
		emotion ET_SCRATCH;
		return;
	}
	next;
	mes .@npc_name$;
	if(Zeny < 5000) {
		mes "คุณมีจำนวนเงินไม่เพียงพอ.";
		emotion ET_MONEY;
		return;
	}
	set Zeny,Zeny-5000;
	set #kafra_code,@code;
	mes "คุณได้รับการป้องกันคลังสินค้าด้วยรหัสผ่านเรียบร้อยแล้ว.";
	mes "ขอบคุณสำหรับการใช้งาน คาฟรา "+.@comp_name$+".";
	emotion ET_THANKS;
	return;
}

// ฟังก์ชั่นการตรวจสอบรหัสผ่านขั้นพื้นฐาน

function	script	F_EntKafCode	{
	mes "โปรดกรอกหมายเลข 1000~10000000";
	set @code_,0;
	set @kafcode_try,@kafcode_try+1;
	if(@kafcode_try>10) {
		set @kafcode_try,0;
		logmes "กำลังพยายามเปลี่ยนรหัสผ่านสำหรับการเก็บของ.";
	}
	if(input(@code_) == 1) {
		mes "คุณไม่สามารถใช้รหัสผ่านดังกล่าวได้.";
		return 0;
	}
	if(@code_<1000) {
		mes "คุณกรอกรหัสผ่านสั้นเกินไป.";
		return 0;
	}
	return @code_;
}
