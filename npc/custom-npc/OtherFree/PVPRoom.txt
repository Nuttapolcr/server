OnPCKillEvent:

	if(@last_kill == killedrid) {
		if(++@last_killamt >= 50) {
			atcommand "@jailfor 1h "+strcharinfo(0);
			announce strcharinfo(0)+" ทำการปั้มคะแนนสนามประลอง "+rid2name(@last_kill)+".",bc_all;

		@last_killamt = 0;
	}
	@last_kill = killedrid;
	end;

	@last_killamt = 0;
	end;

}
