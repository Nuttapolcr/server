function	script	upgrade5all	{
for (set .@part, 0; .@part <= 9; set .@part, .@part + 1) {
while (getequiprefinerycnt(.@part) < 5 && getequipisenableref(.@part)) {
successrefitem .@part;
}
}
return;
}
