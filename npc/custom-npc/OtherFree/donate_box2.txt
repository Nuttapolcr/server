	end;
	
OnInit:

		"slot_v2_06", "slot_v2_07", "slot_v2_08", "slot_v2_09", "slot_v2_10",
		"slot_v2_11", "slot_v2_12", "slot_v2_13", "slot_v2_14", "slot_v2_15",
		"slot_v2_16", "slot_v2_17", "slot_v2_18", "slot_v2_19", "slot_v2_20";
		
	setarray $@DONATE_V2_item[0],
		12210, 39017, 12210, 39001, 12210,
		39003, 12210, 39011, 616, 31165,
		12210, 31180, 12210, 31074, 12210,
		39004, 12210, 38038, 12210, 31094;
		
	//กำหนดโอกาส 100=ง่าย 50=กลาง 10=ยาก
	
	setarray $@DONATE_V2_item_rate[0],
		15, 30, 15, 30, 15,
		30, 15, 30, 15, 30,
		15, 30, 15, 30, 15,

	setarray $@DONATE_V2_count[0],

	// system
	setarray $@DONATE_V2_size_db[0],
		getarraysize($@DONATE_V2_filename$);
	end;
}
function	script	DONATE_V2_NextSlot	{
	set .@img,getarg(0) + 1;
	
	if (.@img >= $@DONATE_V2_size_db[0])
		set .@img,0;
		
	cutin $@DONATE_V2_filename$[.@img], 4;
	sleep2 getarg(1);
	
	return .@img;
}

-	script	DONATE_V2#NPC	FAKE_NPC,{

OnPCLogoutEvent:
	set CostumeBox_Playing,0;
	end;

OnPlay:
	if(CostumeBox_Playing != 0){
		mes "^ff0000[ แจ้งเตือน ]^000000";
		mes "......โปรดรอสักครู่.....";
		mes "ไม่สามารถหมุนได้ ในขณะนี้";
		mes "เอากล่องของท่านกลับคืนไปน่ะ";
		getitem 27999,1;
		close;
	}
	pcblockmove getcharid(3),1;
	set CostumeBox_Playing,1;
	set .@img,0;
	set .@round,rand(15,30);
	set .@speed,10;
	for (set .@i,0; .@i < .@round; set .@i,.@i+1)
	{
		set .@mini_round,rand($@DONATE_V2_size_db[0]/2, $@DONATE_V2_size_db[0]);
		for (set .@j,0; .@j < .@mini_round; set .@j,.@j+1)
		set .@img,callfunc("DONATE_V2_NextSlot", .@img, .@speed);
		set .@speed,.@speed+10;
	}
	
	set .@round,rand($@DONATE_V2_size_db[0]/2, $@DONATE_V2_size_db[0]);
	
	for (set .@i,0; .@i < .@round; set .@i,.@i+1)
	{
		set .@img,callfunc("DONATE_V2_NextSlot", .@img, .@speed);
		set .@speed,.@speed+50;
	}
	set .@min,($@DONATE_V2_size_db[0]/2)/2;
	set .@round,rand(.@min, $@DONATE_V2_size_db[0]/2);
	set .@round_per_50,.@round/2;
	set .@round_per_25,.@round_per_50/2;

	for (set .@i,0; .@i < .@round; set .@i,.@i+1)
	{
		set .@img,callfunc("DONATE_V2_NextSlot", .@img, .@speed);

		if (.@i < .@round_per_25)
			set .@speed,.@speed+25;
		else if (.@i < .@round_per_50)
			set .@speed,.@speed+100;
		else
			set .@speed,.@speed+250;
			
		if(.@i == .@round-1 && $@DONATE_V2_item_rate[.@img] < rand(100)){
			set .@i,.@i-1;
		}
	}
	
	pcblockmove getcharid(3),0;
	set .@prize,$@DONATE_V2_item[.@img];
	set CostumeBox_Playing,0;
	
	
	mes "^ff0000[ Donate Costume V.2 ]^000000";
	mes "ว้าวๆๆๆๆๆๆๆๆๆ";
	mes "คุณได้รับไอเท็มคือ :";
	mes "^ff0000- "+getitemname(.@prize)+"^000000";
	mes "ทำไมวันนี้คุณโชคดีเหลือเกิน ?";
	announce "[ Donate Costume Vol.2 : คุณ "+strcharinfo(0)+" ได้รับไอเท็ม "+getitemname(.@prize)+" จำนวน "+$@DONATE_V2_count[.@img]+" ea. ]",bc_all,0xfcff00;
	getitem .@prize, $@DONATE_V2_count[.@img];
	close2;
	cutin "", 255;
	end;

}
