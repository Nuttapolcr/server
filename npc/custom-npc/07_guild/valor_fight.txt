// ValorFight WoE castle and safe lobby.

function	script	F_ValorFightActive	{
	.@idx = getvariableofnpc(.ValorFightIndex, "WOE_CONTROL");
	.@count = getvariableofnpc(.CastleCount, "WOE_CONTROL");
	.@active = getvariableofnpc(.Active[0], "WOE_CONTROL");
	if (.@idx < 0 || .@idx >= .@count)
		return 0;
	return agitcheck() && (.@active & (1 << .@idx));
}

function	script	F_ValorFight2Active	{
	.@idx = getvariableofnpc(.ValorFight2Index, "WOE_CONTROL");
	.@count = getvariableofnpc(.CastleCount, "WOE_CONTROL");
	.@active = getvariableofnpc(.Active[0], "WOE_CONTROL");
	.@event_active = getvariableofnpc(.active, "Agit#hero_ent2");
	if (.@idx < 0 || .@idx >= .@count)
		return 0;
	return agitcheck() && (.@active & (1 << .@idx)) && .@event_active;
}

function	script	F_ValorFightWarp	{
	if (!callfunc("F_ValorFightActive")) {
		message strcharinfo(0), "[WoE Warp] ValorFight ยังไม่เปิดในรอบนี้.";
		return;
	}
	warp "hero_ent1",getarg(0),getarg(1);
	return;
}

function	script	F_ValorFight2Warp	{
	if (!callfunc("F_ValorFight2Active")) {
		message strcharinfo(0), "[WoE Warp] ValorFight Map 2 ยังไม่เปิดในรอบนี้.";
		return;
	}
	if (getcharid(2) <= 0) {
		message strcharinfo(0), "[WoE Warp] ต้องสังกัดกิลด์จึงจะเข้า ValorFight Map 2 ได้.";
		return;
	}
	warp "hero_ent2",getarg(0),getarg(1);
	return;
}
