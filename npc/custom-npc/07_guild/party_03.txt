Start:
	
	mes .mes$;

	}
	if($party_03_team_b){

			$party_03_point_b = $party_03_point_b -1;

				goto OnEmpBreak_a;

			}

		sleep 1000;
		if(.@sec <= 0){
			announce "Party 03 : การแข่งขันเริ่มแล้ว ลุย ลุย ลุย !!!",bc_all|bc_woe;
			goto OnStart;
			end;
		}
	}
	end;

	donpcevent ""+strnpcinfo(0)+"::OnOpen";

	donpcevent ""+strnpcinfo(0)+"::OnCheckTime";

				donpcevent ""+strnpcinfo(0)+"::OnEnding";

			"- เรียกผู้เล่นเข้า ห้องรับรอง",

				"- Team A",

				"- ล้างค่า Team A/B"

			}
			if(.@menu_team == 2){
				input .@gid;
				clear;
				mes .mes$;
				mes "TEAM B : "+getpartyname(.@gid)+"";
				mes "ใช่หรือไม่ ?";
				.@menu_team_b = select(
					"- ตกลง"
				);
				$party_03_team_b = .@gid;
			}
			if(.@menu_team == 3){
				$party_03_team_a = 0;
				$party_03_team_b = 0;
				$party_03_score_a = 0;
				$party_03_score_b = 0;
			}
		}
		if(.@menu == 2){
			donpcevent ""+strnpcinfo(0)+"::OnRecallRoom";
		}
		if(.@menu == 3){
			donpcevent ""+strnpcinfo(0)+"::OnRecall";
		}
		if(.@menu == 4){ donpcevent ""+strnpcinfo(0)+"::OnTimeStart"; }
		if(.@menu == 5){ $status_party_03 = 1; }
		if(.@menu == 6){
			$status_party_03 = 1;

		if(.@menu == 7){
			.@menu_wall = select(
				"- เปิด",
				"- ปิด"
			);
			if(.@menu_wall == 1){
				donpcevent ""+strnpcinfo(0)+"::OnOpen";

			if(.@menu_wall == 2){
				donpcevent ""+strnpcinfo(0)+"::OnClose";
			}
		}
		if(.@menu == 8){
			$party_03_point_a = .point_win;
			$party_03_point_b = .point_win;
			MapRespawnGuildID "2009rwc_04",.@GID,7;
		}
		if(.@menu == 9){
			donpcevent ""+strnpcinfo(0)+"::OnExit";
		}
		if(.@menu == 10){
				clear;
				mes "[Party Info]";
				mes "กำลังโหลดรายชื่อปาร์ตี้ทั้งหมด...";

					mes "ไม่พบปาร์ตี้ในฐานข้อมูล.";
					close;
				}
				
				for (.@i = 0; .@i < .@count; .@i++) {
					dispbottom "ID: " + .@pid[.@i] + " | Party: " + .@pname$[.@i];

	end;

	.mes$ = "^FF3300[ Party 03 ]^000000";
	
	waitingroom "[#] Party 03",0;
