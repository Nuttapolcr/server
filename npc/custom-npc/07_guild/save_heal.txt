// =====================================================
// Global CONFIG: วัน/เวลา PACK_WAR (แก้ได้ที่นี่)
// =====================================================
-	script	WarPackCfg	-1,{

OnInit:
	// เวลาเริ่ม-จบ (24 ชั่วโมง)
	set $WAR_START_H, 20;   // เริ่ม 20:30
	set $WAR_START_M, 30;
	set $WAR_END_H,   23;   // จบ 23:30
	set $WAR_END_M,   30;

	// เลือก "วัน" ที่อนุญาต (0=อา,1=จ,2=อ,3=พ,4=พฤ,5=ศ,6=ส)
	deletearray $@WAR_DAYS[0], getarraysize($@WAR_DAYS);
	setarray $@WAR_DAYS[0], 0, 1, 2, 3, 4, 5, 6;

	end;
}

// =====================================================
// Global FUNCTIONS (อยู่นอกบล็อก NPC)
// =====================================================

function	script	F_war_is_active	{
	set .@H,   atoi(gettimestr("%H",20));
	set .@M,   atoi(gettimestr("%M",20));
	set .@nowm, .@H*60 + .@M;
	set .@dow, atoi(gettimestr("%w",20));
	set .@start, $WAR_START_H*60 + $WAR_START_M;
	set .@end,   $WAR_END_H*60   + $WAR_END_M;
	set .@check_day, (.@start > .@end && .@nowm < .@end) ? ((.@dow+6)%7) : .@dow;
	set .@in_day, 0;

	if (getarraysize($@WAR_DAYS) > 0) {
		for (set .@i,0; .@i<getarraysize($@WAR_DAYS); set .@i,.@i+1)
			if ($@WAR_DAYS[.@i] == .@check_day) { set .@in_day,1; break; }
	} else {
		set .@in_day, 1;
	}
	if (!.@in_day) return 0;

	if (.@start <= .@end)
		return (.@nowm >= .@start && .@nowm < .@end);
	else
		return (.@nowm >= .@start || .@nowm < .@end);
}

function	script	F_war_days_text	{
	setarray .@d$[0],"อา","จ","อ","พ","พฤ","ศ","ส";
	set .@s$, "";
	if (getarraysize($@WAR_DAYS) > 0) {
		for (set .@i,0; .@i<getarraysize($@WAR_DAYS); set .@i,.@i+1)
			set .@s$, .@s$ + ((.@i)?"/":"") + .@d$[$@WAR_DAYS[.@i]];
	} else {
		set .@s$, "อา/จ/อ/พ/พฤ/ศ/ส";
	}
	return .@s$;
}

// =====================================================
// NPC: Buffer Service
// =====================================================
function	script	F_PremiumServiceStatsBuff	{
	if (sc_check(SC_PREMIUMSERVICE)) {
		sc_start SC_INCREASEAGI,getarg(0),10;
		sc_start SC_BLESSING,getarg(0),10;
		sc_start SC_INCALLSTATUS,getarg(0),3;
	}
	return;
}

	set .@Price, 0;       // ค่าบริการ Zeny

		dispbottom "[Buffer Service] รออีก "+(.@delay-(gettimetick(2)-@delay_heal))+" วินาที ถึงจะทำการ Buff รอบต่อไปได้";

		if (callfunc("F_war_is_active")) {
			set @delay_heal, gettimetick(2);
			
			sc_end SC_INCREASEAGI;
			sc_end SC_BLESSING;

			sc_start SC_INCREASEAGI,900000,10;
			sc_start SC_BLESSING,900000,10;
			sc_start4 SC_POEMBRAGI,300000,5,70,90,10;
			callfunc "F_PremiumServiceStatsBuff",900000;

			atcommand "@repairall";
			identifyall;
			percentheal 100,0;
			specialeffect2 EF_BLESSING;
			specialeffect2 EF_INCAGILITY;

			// ล้างดีบัฟ
			sc_end SC_POSION;
			sc_end SC_FREEZE;
			sc_end SC_SLEEP;
			sc_end SC_Stan;
			sc_end SC_STUN;
			sc_end SC_STONE;
			sc_end SC_SILENCE;
			sc_end SC_BLIND;
			sc_end SC_Bleeding;
			sc_end SC_CONFUSION;
			sc_end SC_CURSE;
			sc_end SC_Hallucination;
			sc_end SC_ORCISH;

			dispbottom "[VIP WAR] คุณได้รับบัฟจาก VIP WAR";
			end;
		} else {
			// นอกช่วง WAR -> ไม่แจก WAR, ไม่กินดีเลย์ (ให้ไปเช็ก VIP / ฟรี / เสียเงิน ต่อ)

			           + " เวลา " + $WAR_START_H + ":" + (($WAR_START_M<10)?"0":"") + $WAR_START_M
			           + "–" + $WAR_END_H + ":" + (($WAR_END_M<10)?"0":"") + $WAR_END_M + " น.";
		}
	}

	if ( vip_status(1) ) {
		set @delay_heal, gettimetick(2);
		set .@repair_cost, 1000;
		if (getbrokenid(1) > 0) {
			if (Zeny >= .@repair_cost) {
				Zeny -= .@repair_cost;
				atcommand "@repairall";
				specialeffect2 101;
				dispbottom "[VIP Card] ซ่อมอุปกรณ์เสียหายเรียบร้อย หัก " + .@repair_cost + " Zeny.";
			} else {
				dispbottom "[VIP Card] คุณมี Zeny ไม่พอ (" + .@repair_cost + "z) ไม่สามารถซ่อมอุปกรณ์ได้";
				//end;
			}
		}
		identifyall;
		sc_end SC_INCREASEAGI;
		sc_end SC_BLESSING;
		sc_start SC_INCREASEAGI,300000,10;
		sc_start SC_BLESSING,300000,10;
		sc_start4 SC_POEMBRAGI,300000,5,70,90,10;
		callfunc "F_PremiumServiceStatsBuff",300000;
		specialeffect2 EF_BLESSING;
		specialeffect2 EF_INCAGILITY;
		sc_end SC_POSION;
		sc_end SC_FREEZE;
		sc_end SC_SLEEP;
		sc_end SC_Stan;
		sc_end SC_STUN;
		sc_end SC_STONE;
		sc_end SC_SILENCE;
		sc_end SC_BLIND;
		sc_end SC_Bleeding;
		sc_end SC_CONFUSION;
		sc_end SC_CURSE;
		sc_end SC_Hallucination;
		sc_end SC_ORCISH;

		dispbottom "[VIP Card] คุณได้รับบัฟจาก VIP Card";
		end;
	}

	// ----- 3) ฟรีสำหรับเลเวล <= 70 -----
	if (BaseLevel <= 70) {
		set @delay_heal, gettimetick(2);
		percentheal 100,0;

		sc_end SC_INCREASEAGI;
		sc_end SC_BLESSING;

		sc_start SC_INCREASEAGI,300000,5;
		sc_start SC_BLESSING,300000,5;
		sc_start4 SC_POEMBRAGI,300000,5,70,90,10;

		identifyall;
		specialeffect2 509;
		specialeffect2 EF_BLESSING;
		specialeffect2 EF_INCAGILITY;

		sc_end SC_POSION;
		sc_end SC_FREEZE;
		sc_end SC_SLEEP;
		sc_end SC_Stan;
		sc_end SC_STUN;
		sc_end SC_STONE;
		sc_end SC_SILENCE;
		sc_end SC_BLIND;
		sc_end SC_Bleeding;
		sc_end SC_CONFUSION;
		sc_end SC_CURSE;
		sc_end SC_Hallucination;
		sc_end SC_ORCISH;

		dispbottom "[Buff] คุณได้รับบัฟฟรี (เลเวลไม่ถึง 70)";
		end;
	}

	// ----- 4) เสียเงิน -----
	if (.@Price) {
		message strcharinfo(0), "ค่าบริการ " + callfunc("F_InsertComma",.@Price) + " Zeny.";
		if (Zeny < .@Price) end;
		if (select("^0055FFค่าบริการ 1000z^000000:^777777ขอบคุณจ้า^000000") == 2) end;
		Zeny -= .@Price;
	}
	
	set .@repair_cost2, 2000;
	if (getbrokenid(1) > 0) {
		if (Zeny >= .@repair_cost2) {
			Zeny -= .@repair_cost2;
			atcommand "@repairall";
			specialeffect2 101; // เอฟเฟกต์ซ่อมของ
			dispbottom "[Buff] ซ่อมอุปกรณ์เสียหายเรียบร้อย หัก " + .@repair_cost2 + " Zeny.";
		} else {
			dispbottom "[Buff] คุณมี Zeny ไม่พอ (" + .@repair_cost2 + "z) ไม่สามารถซ่อมอุปกรณ์ได้";
		}
	}
	identifyall;

	set @delay_heal, gettimetick(2);
	
	percentheal 100, 100;
	
	sc_end SC_INCREASEAGI;
	sc_end SC_BLESSING;

	sc_start SC_INCREASEAGI,900000,5;
	sc_start SC_BLESSING,900000,5;
	sc_start4 SC_POEMBRAGI,300000,5,70,90,10;
	callfunc "F_PremiumServiceStatsBuff",900000;

	specialeffect2 509;
	specialeffect2 EF_BLESSING;
	specialeffect2 EF_INCAGILITY;

	sc_end SC_POSION;
	sc_end SC_FREEZE;
	sc_end SC_SLEEP;
	sc_end SC_Stan;
	sc_end SC_STUN;
	sc_end SC_STONE;
	sc_end SC_SILENCE;
	sc_end SC_BLIND;
	sc_end SC_Bleeding;
	sc_end SC_CONFUSION;
	sc_end SC_CURSE;
	sc_end SC_Hallucination;
	sc_end SC_ORCISH;

	dispbottom "[Buff] คุณได้รับบัฟ (เลเวลตั้งแต่ 70) ชำระค่าบริการ 1000";
