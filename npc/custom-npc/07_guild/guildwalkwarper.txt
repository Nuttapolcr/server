//===== Warp Visibility Checker NPC ========================
-	script	WarpChecker	-1,{
OnInit:
	// Start the timer to check every 5 seconds
	initnpctimer;
	end;

OnTimer1000:
	// Check and update warp visibility every 5 seconds
	callsub CheckWarps;
	initnpctimer; // Restart timer
	end;

CheckWarps:
	// Get current WOE settings
	set .@castle_warp, getvariableofnpc(.CastleWarp,"WOE_CONTROL");
	set .@active, getvariableofnpc(.Active,"WOE_CONTROL");
	
	// Castle index to NPC name mapping
	setarray .@npc_names$[0], 
		"prtg1","prtg2","prtg3","prtg4","prtg5",
		"payg1","payg2","payg3","payg4","payg5",
		"gefg1","gefg2","gefg3","gefg4","gefg5",
		"aldeg1","aldeg2","aldeg3","aldeg4","aldeg5";
	
	// Update regular castle warps (0-19)
	for(set .@i,0; .@i<20; set .@i,.@i+1) {
		if (.@castle_warp || (.@active&(1<<.@i))) {
			hideoffnpc .@npc_names$[.@i];
		} else {
			hideonnpc .@npc_names$[.@i];
		}
	}
	
	// Handle main castle warps (index 20)
	if (.@castle_warp || (.@active&(1<<20))) {
		hideoffnpc "maincastlewarp#1";
		hideoffnpc "maincastlewarp#2";
	} else {
		hideonnpc "maincastlewarp#1";
		hideonnpc "maincastlewarp#2";
	}
	return;
}

	end;
}
