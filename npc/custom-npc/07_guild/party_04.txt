maintown,204,123,4	script	party_04	708,{
Start:
	
	mes .mes$;
	mes "Party ID : "+getcharid(1)+"";
	mes "Party Name : "+getpartyname(getcharid(1))+"";
	if($party_04_team_a){
		mes "TEAM A : "+getpartyname($party_04_team_a)+" ["+$party_04_score_a+" GAME]";
	}
	if($party_04_team_b){
		mes "TEAM B : "+getpartyname($party_04_team_b)+" ["+$party_04_score_b+" GAME]";
	}
	if(getgmlevel()>=99){
		clear;
		goto OnMenu;
		end;
	}
	end;
	
OnPCKillEvent:
	getmapxy(.@map$,.@x,.@y,BL_PC);
	if(.@map$ == "2012rwc_07"){
		if(getcharid(1) == $party_04_team_a){
			$party_04_point_b = $party_04_point_b -1;
			announce "Party 04 : ผู้เล่น "+strcharinfo(0)+" ปาร์ตี้ "+getpartyname($party_04_team_a)+" ได้สังหาร "+rid2name(killedrid)+" ปาร์ตี้ "+getpartyname($party_04_team_b)+"",bc_all|bc_woe;
			donpcevent ""+strnpcinfo(0)+"::OnAnn";
			if($party_04_point_b <= 0){
				goto OnEmpBreak_b;
				end;
			}
			end;
		}
		if(getcharid(1) == $party_04_team_b){
			$party_04_point_a = $party_04_point_a -1;
			announce "Party 04 : ผู้เล่น "+strcharinfo(0)+" ปาร์ตี้ "+getpartyname($party_04_team_b)+" ได้สังหาร "+rid2name(killedrid)+" ปาร์ตี้ "+getpartyname($party_04_team_a)+"",bc_all|bc_woe;
			donpcevent ""+strnpcinfo(0)+"::OnAnn";
			if($party_04_point_a <= 0){
				goto OnEmpBreak_a;
				end;
			}
			end;
		}
	}
	end;

OnAnn:
	announce "Party 04 : ปาร์ตี้ "+getpartyname($party_04_team_a)+" (เหลือ "+$party_04_point_a+" คน) VS ปาร์ตี้ "+getpartyname($party_04_team_b)+" (เหลือ "+$party_04_point_b+" คน)",bc_all|bc_woe;
	end;
	
OnTimeStart:
	$status_party_04 = 0;
	.end_game = 0;
	.@sec = .time_ann;

	donpcevent ""+strnpcinfo(0)+"::OnRecall";

	}
	while(1){
		announce "Party 04 : การแข่งขันกำลังจะเริ่ม เหลือเวลาอีก "+.@sec+" วินาที",bc_all|bc_woe;
		.@sec --;
		sleep 1000;
		if(.@sec <= 0){
			announce "Party 04 : การแข่งขันเริ่มแล้ว ลุย ลุย ลุย !!!",bc_all|bc_woe;
			goto OnStart;
			end;
		}
	}
	end;
	
OnStart:
	donpcevent ""+strnpcinfo(0)+"::OnOpen";
	.@GID = $party_04_team_a;
	donpcevent ""+strnpcinfo(0)+"::OnCheckTime";
	end;

OnEmpBreak_a:
	.@GID = 0;
	announce "Party 04 : ผู้ชนะคือ ปาร์ตี้ "+getpartyname($party_04_team_b)+"",bc_all|bc_woe;
	$party_04_score_b ++;
	donpcevent ""+strnpcinfo(0)+"::OnEnding";
	end;

OnEmpBreak_b:
	.@GID = 0;
	announce "Party 04 : ผู้ชนะคือ ปาร์ตี้ "+getpartyname($party_04_team_a)+"",bc_all|bc_woe;
	$party_04_score_a ++;
	donpcevent ""+strnpcinfo(0)+"::OnEnding";

	if(($party_04_score_a >= .game_win) && ($party_04_score_b >= .game_win)){

		sleep 1500;

		$party_04_score_b = 0;

		end;
	}

	if($party_04_score_a >= .game_win){

		sleep 1500;
		announce "Party 04 : ผลการแข่ง ปาร์ตี้ "+getpartyname($party_04_team_a)+" ชนะไปด้วย "+$party_04_score_a+"-"+$party_04_score_b+" GAME",bc_all|bc_woe;

		$party_04_score_b = 0;

		$party_04_point_b = .point_win;

		$party_04_team_b = 0;

		.@sec_time	= .@time2-gettimetick(2);

		.@hour_time	= .@min_time/60;
		.@text_time	= .@min_time-(60*.@hour_time);
		.@text2_time	= .@sec_time-(60*.@min_time);

			.@hour_time$ = ""+.@hour_time+"";
		}
		if(.@text_time >= 0 && .@text_time <= 9){
			.@text_time$ = "0"+.@text_time+"";
		}
		else if(.@text_time > 9){
			.@text_time$ = ""+.@text_time+"";
		}
		if(.@text2_time >= 0 && .@text2_time <= 9){
			.@text2_time$ = "0"+.@text2_time+"";

			.@text2_time$ = ""+.@text2_time+"";

				donpcevent ""+strnpcinfo(0)+"::OnEnding";
			}
			if($party_04_point_a < $party_04_point_b){
				announce "Party 04 : "+getpartyname($party_04_team_b)+" เป็นฝ่ายชนะ "+$party_04_point_b+" คน vs "+$party_04_point_a+" คน",bc_all|bc_woe;
				$party_04_score_b ++;
				donpcevent ""+strnpcinfo(0)+"::OnEnding";
			}
			end;
		}
		sleep 1000;
	}
	end;
	

OnMenu:
	
	while(1){
		mes .mes$;
		if($party_04_team_a){
			mes "TEAM A : "+getpartyname($party_04_team_a)+" ["+$party_04_score_a+" GAME]";
		}
		if($party_04_team_b){
			mes "TEAM B : "+getpartyname($party_04_team_b)+" ["+$party_04_score_b+" GAME]";
		}
		.@menu = select(
			"- ตั้งค่า Team",
			"- เรียกผู้เล่นเข้า ห้องรับรอง",
			"- เรียกผู้เล่นเข้า สนาม",

			"- หยุด+ตัดจบเกมส์นับผลเลย",
			"- เปิด-ปิด กำแพง",
			"- เตะผู้เล่นออกจากสนาม",
			"- เตะผู้เล่นออกจากห้องรับรอง",
			"- ดูข้อมูล Party-ID"
		);
		if(.@menu == 1){
			.@menu_team = select(
				"- Team A",
				"- Team B",
				"- ล้างค่า Team A/B"
			);
			if(.@menu_team == 1){
				input .@gid;
				clear;
				mes .mes$;
				mes "TEAM A : "+getpartyname(.@gid)+"";
				mes "ใช่หรือไม่ ?";
				.@menu_team_a = select(
					"- ตกลง"
				);
				$party_04_team_a = .@gid;
			}
			if(.@menu_team == 2){
				input .@gid;
				clear;
				mes .mes$;
				mes "TEAM B : "+getpartyname(.@gid)+"";
				mes "ใช่หรือไม่ ?";
				.@menu_team_b = select(
					"- ตกลง"
				);
				$party_04_team_b = .@gid;
			}
			if(.@menu_team == 3){
				$party_04_team_a = 0;
				$party_04_team_b = 0;
				$party_04_score_a = 0;
				$party_04_score_b = 0;
			}
		}
		if(.@menu == 2){
			donpcevent ""+strnpcinfo(0)+"::OnRecallRoom";
		}
		if(.@menu == 3){
			donpcevent ""+strnpcinfo(0)+"::OnRecall";
		}
		if(.@menu == 4){ donpcevent ""+strnpcinfo(0)+"::OnTimeStart"; }
		if(.@menu == 5){ $status_party_04 = 1; }
		if(.@menu == 6){
			$status_party_04 = 1;
		}
		if(.@menu == 7){
			.@menu_wall = select(
				"- เปิด",
				"- ปิด"
			);
			if(.@menu_wall == 1){
				donpcevent ""+strnpcinfo(0)+"::OnOpen";
			}
			if(.@menu_wall == 2){
				donpcevent ""+strnpcinfo(0)+"::OnClose";
			}
		}
		if(.@menu == 8){
			$party_04_point_a = .point_win;
			$party_04_point_b = .point_win;
			MapRespawnGuildID "2012rwc_07",.@GID,7;
		}
		if(.@menu == 9){
			donpcevent ""+strnpcinfo(0)+"::OnExit";
		}
		if(.@menu == 10){
				clear;
				mes "[Party Info]";
				mes "กำลังโหลดรายชื่อปาร์ตี้ทั้งหมด...";
				mes "ดูข้อมูลที่ช่องแชท";
				.@count = query_sql("SELECT party_id, name FROM party ORDER BY party_id ASC", .@pid, .@pname$);
				
				if (.@count <= 0) {
					mes "ไม่พบปาร์ตี้ในฐานข้อมูล.";
					close;
				}
				
				for (.@i = 0; .@i < .@count; .@i++) {
					dispbottom "ID: " + .@pid[.@i] + " | Party: " + .@pname$[.@i];
				}
				close;
			}
		clear;
	}
	end;

OnOpen:
	
	donpcevent "Board party_04#1::OnOpen";
	donpcevent "Board party_04#2::OnOpen";
	donpcevent "Board party_04#3::OnOpen";
	donpcevent "Board party_04#4::OnOpen";
	donpcevent "Board party_04#5::OnOpen";
	donpcevent "Board party_04#6::OnOpen";

	warpparty "a_room_04",64,133,$party_04_team_a;

		warpparty "2012rwc_07",46,80,$party_04_team_a;

	end;

	addrid(5),1,"a_room_04";

	getmapxy(.@map$,.@x,.@y,BL_PC);

		end;
	}
	end;
	
OnInit:
	
	.mes$ = "^FF3300[ Party 04 ]^000000";
	
	waitingroom "[#] Party 04",0;
	
	$status_party_04 = 0;
	
	.max_member			= 7;				//	กำหนดจำนวนของสมาชิกปาร์ตี้

	.point_win			= .max_member;		//	ตั้งค่า แต้มห่างกี่แต้ม ชนะเลย
	.time_ann			= 10;				//	กำหนดเวลาประกาศ ก่อนเริ่มกิจกรรม	(วินาที)

	$party_04_point_a	= .point_win;

	bindatcmd "party_04",""+strnpcinfo(0)+"::OnMenu",99,0;
	
	setmapflag "2012rwc_07",mf_loadevent;
