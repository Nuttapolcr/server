function	script	F_CS2_Open	{
	.@castle_count = getvariableofnpc(.CastleCount, "WOE_CONTROL");
	.@active = getvariableofnpc(.Active[0], "WOE_CONTROL");
	.@valor_idx = getvariableofnpc(.ValorFightIndex, "WOE_CONTROL");

	if (.@castle_count <= .@valor_idx || .@valor_idx < 0) {
		dispbottom "[WoE Warp] ระบบปราสาทไม่พร้อมใช้งาน.";
		return;
	}

	if (!agitcheck() || !(.@active & (1 << .@valor_idx))) {
		dispbottom "[WoE Warp] ValorFight ยังไม่เปิดในรอบนี้.";
		return;
	}

	sc_end SC_POEMBRAGI; sc_end SC_SERVICE4U; sc_end SC_ENCHANTARMS; sc_end SC_ASSUMPTIO;
	sc_end SC_AURABLADE; sc_end SC_MAGICPOWER; sc_end SC_CARTBOOST; sc_end SC_TRUESIGHT;
	sc_end SC_FLEEFOOD;  sc_end SC_HITFOOD;

	set lastwarp$, "guild_vs2-1";
	set lastwarpx, 49;
	set lastwarpy, 49;

	warp "guild_vs2-1",49,49;
	return;
}

-	script	CS2#UI	-1,{
OnStart:
	callfunc "F_CS2_Open";
	end;
}
