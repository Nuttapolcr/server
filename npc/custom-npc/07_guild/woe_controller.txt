function Disp_Owner {
	set .@o, getcastledata(getarg(0),1);
	if (.@o) announce "The ["+getcastlename(getarg(0))+"] castle "+((getarg(1))?" ได้ถูกยึดแล้ว ":"is currently held")+" โดย the ["+getguildname(.@o)+"] guild.",bc_all|bc_woe;
	else announce "The ["+getcastlename(getarg(0))+"] castle is currently unoccupied.",bc_all|bc_woe;
	return;
}
function TimeLabel; function SlotNow;
function Add_Zero {
	return ((getarg(0)<10)?"0":"")+getarg(0)+(getarg(1,0)?".":":")+"00";
}

function TimeLabel {
	set .@s, getarg(0);
	if (.@s < 0) set .@s, 0;
	if (.@s > 48) set .@s, 48;
	set .@h, .@s/2;
	set .@m, (.@s%2)?30:0;
	return ((.@h<10)?"0":"")+.@h+((getarg(1,0))?".":":")+((.@m==0)?"00":"30");
}
function SlotNow {
	return (gettime(3)*2) + ((gettime(2) >= 30) ? 1 : 0);

			for (set .@i,0; .@i<.Size; set .@i, .@i+4) {
				set .@day,  $WOE_CONTROL[.@i];
				set .@start,$WOE_CONTROL[.@i+1];
				set .@end,  $WOE_CONTROL[.@i+2];
				set .@menu$, .@menu$+" ~ "+.Days$[.@day]+" ("+TimeLabel(.@start,1)+"-"+TimeLabel(.@end,1)+"):";
			}
			set .@menu$, .@menu$+" ~ ^777777Cancel^000000";
			set .@i, select(.@menu$)-1;
			if (.@i == (.Size/4)) break;
			mes "[ลบ วันเวลากิลวอร์]";
			mes "คุณต้องการที่จะลบรายการ ^0055FF"+.Days$[$WOE_CONTROL[.@i*4]]+"^000000 ออกจากระบบ?";
			mes "ถ้าหากทำรายการแล้วจะไม่สามารถยกเลิกได้.";
			next;
			set .@j, select(" ~ ^FF0000ลบ รายการที่เลือก...^000000: ~ ^777777ยกเลิก^000000");
			mes "[Remove Session]";
			if (.@j == 2)
				mes "ยกเลิกการทำรายการแล้วค่ะ.";
			else {
				deletearray $WOE_CONTROL[.@i*4],4;
				set .Size, getarraysize($WOE_CONTROL);
				mes "รายการได้ถูกลบออกจากระบบแล้วค่ะ.";
			}
			next;
			break;
		case 3:
			mes "[Reload Settings]";
			mes "การทำรายการนี้จะทำให้เริ่มระบบใหม่";
			mes "ทั้งหมด.";
			if (agitcheck() || agitcheck2()) {
				mes " ";
				mes "โปรดทราบว่าถ้าหากทำรายการนี้แล้วจะส่งผลกระทบต่อการใช้งานปัจจุบัน.";
			}
			next;
			set .@i, select(" ~ ^0055FFReload settings...^000000: ~ ^777777ยกเลิก^000000");
			mes "[Reload Settings]";
			if (.@i == 2) mes "ยกเลิกรายการเรียบร้อยแล้วค่ะ.";
			else {
				set .Init,1;
				donpcevent "WOE_CONTROL::OnMinute00";
				mes "ทำการ Reload เรียบร้อยแล้วค่ะ.";
			}
			next;
			break;
		case 4:
			mes "[หยุดระบบกิลวอร์]";
			if (!agitcheck() && !agitcheck2())  {
				mes "กิลวอร์ ได้สิ้นสุดลงแล้ว.";
				next;
				break;
			}
			mes "คำสั่งนี้จะดำเนินการได้อย่างปลอดภัยหากกิลวอร์ได้เริ่มอยู่.";
			mes " ";
			mes "คุณต้องการที่จะสิ้นสุดกิลวอร์หรือไม่?";
			next;
			set .@i, select(" ~ ^FF0000สิ้นสุดรายการ...^000000:"+((.Options&1)?" ~ ^FF0000สิ้นสุดรายการและไม่ได้รางวัล...^000000":"")+": ~ ^777777ยกเลิก^000000");
			mes "[หยุดระบบกิลวอร์]";
			if (.@i == 3)
				mes "ยกเลิกรายการเรียบร้อยแล้วค่ะ.";
			else {
				set .ForceEnd, .@i;
				donpcevent "WOE_CONTROL::OnWOEEnd";
				mes "กิลวอร์ ได้ถูกปิดลงแล้ว.";
			}
			next;
			break;
		case 5:
			set .@d,1; break;
		}
		if (.@d) { set .@d,0; break; }
		} break;
	case 5:
		close;
	} }
}
