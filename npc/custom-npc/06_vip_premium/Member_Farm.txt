function	script	FRM_CalcLv	{
	.@pt = getarg(0);

	setarray .@th[1],
		$Farm_Point_Lv1,$Farm_Point_Lv2,$Farm_Point_Lv3,$Farm_Point_Lv4,$Farm_Point_Lv5,$Farm_Point_Lv6,$Farm_Point_Lv7;

	.@lv = 0;
	for (.@i = getarraysize(.@th) - 1; .@i >= 1; --.@i) {
		if (.@pt >= .@th[.@i]) { .@lv = .@i; break; }
	}
	return .@lv;
}

function	script	FRM_RecalcAndApply	{
	.@acc = getarg(0);

	.@id = .@lv = .@pt = 0;
	query_sql("SELECT `id`,`farm_level`,`farm_point` FROM `farm_system` WHERE account_id = "+.@acc, .@id, .@lv, .@pt);

	if (!.@id) {
		query_sql("INSERT INTO `farm_system` (`account_id`,`userid`,`farm_level`,`farm_point`) VALUES ("+.@acc+",'',0,0)");
		.@lv = 0;
		.@pt = 0;
	}

	.@new = callfunc("FRM_CalcLv", .@pt);

	if (.@new != .@lv)
		query_sql("UPDATE `farm_system` SET `farm_level` = "+.@new+" WHERE account_id = "+.@acc);

	setarray .@SC[1],
		SC_FARMING_LV1,SC_FARMING_LV2,SC_FARMING_LV3,SC_FARMING_LV4,SC_FARMING_LV5,SC_FARMING_LV6,SC_FARMING_LV7;

	for (.@i = 1; .@i < getarraysize(.@SC); ++.@i)
		sc_end .@SC[.@i];

	if (.@new > 0)
		sc_start4 .@SC[.@new], INFINITE_TICK, 0,0,0,0;

	return .@new;
}


function	script	FRM_ParseAmt	{
	.@n = getarg(0);
	if (.@n >= 1000000) {
		.@w = .@n / 1000000;
		.@t = (.@n % 1000000) / 100000;
		if (.@t > 0) return ""+.@w+"."+.@t+"M";
		return ""+.@w+"M";
	} else if (.@n >= 1000) {
		.@w = .@n / 1000;
		.@t = (.@n % 1000) / 100;
		if (.@t > 0) return ""+.@w+"."+.@t+"K";
		return ""+.@w+"K";
	}
	return ""+.@n;
}
