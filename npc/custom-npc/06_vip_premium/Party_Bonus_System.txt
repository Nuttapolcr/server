-	script	Party_Bonus_System#Main	-1,{
	end;

OnPCLoginEvent:
OnPCLogoutEvent:
OnPCDieEvent:
OnPCReviveEvent:
OnSec:
	if (getcharid(1) == 0 || readparam(Hp) <= 0) {
		for (.@i = 0; .@i <= 9; .@i++) sc_end getvariableofnpc(.SC[.@i], strnpcinfo(3));
		@last_partyexp_idx = -1;
		addtimer 3000, strnpcinfo(3)+"::OnSec";
		end;
	}

	.@pid = getcharid(1);
	getpartymember .@pid, 1;
	getpartymember .@pid, 2;

	.@self_aid = getcharid(3);
	.@self_cid = getcharid(0);
	.@self_map$ = strcharinfo(3);

	set .@cnt, 0;
	for (.@p = 0; .@p < $@partymembercount; .@p++) {
		if (!isloggedin($@partymemberaid[.@p], $@partymembercid[.@p])) continue;
		if ($@partymembercid[.@p] == .@self_cid) continue;
		if ($@partymemberaid[.@p] == .@self_aid) continue;
		if (!attachrid($@partymemberaid[.@p])) continue;

		if (readparam(Hp) > 0 && strcharinfo(3) == .@self_map$)
			.@cnt++;

		attachrid(.@self_aid);
	}

	if (.@cnt <= 0) {
		if (@last_partyexp_idx != -1) {
			for (.@i = 0; .@i <= 9; .@i++) sc_end getvariableofnpc(.SC[.@i], strnpcinfo(3));
			@last_partyexp_idx = -1;
		}
		addtimer 3000, strnpcinfo(3)+"::OnSec";
		end;
	}

	if (getpartyshare(.@pid) != 1) {
		for (.@i = 0; .@i <= 9; .@i++) sc_end getvariableofnpc(.SC[.@i], strnpcinfo(3));
		@last_partyexp_idx = -1;
		addtimer 3000, strnpcinfo(3)+"::OnSec";
		end;
	}

	.@idx = (.@cnt >= 9) ? 9 : .@cnt;

	if (@last_partyexp_idx != .@idx) {
		for (.@i = 0; .@i <= 9; .@i++) sc_end getvariableofnpc(.SC[.@i], strnpcinfo(3));
		sc_start getvariableofnpc(.SC[.@idx], strnpcinfo(3)), -1, 1;
		@last_partyexp_idx = .@idx;
	}

	addtimer 3000, strnpcinfo(3)+"::OnSec";
	end;

OnInit:
	setarray .SC[0],
		SC_PARTYEXP,
		SC_PARTYEXP_LV1,
		SC_PARTYEXP_LV2,
		SC_PARTYEXP_LV3,
		SC_PARTYEXP_LV4,
		SC_PARTYEXP_LV5,
		SC_PARTYEXP_LV6,
		SC_PARTYEXP_LV7,
		SC_PARTYEXP_LV8,
		SC_PARTYEXP_LV9;
	end;
}
