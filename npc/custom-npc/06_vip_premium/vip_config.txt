// callfunc("F_GetBuffetTime", <account_id>);
function	script	F_GetBuffetTime	{
	.@rst = query_sql("SELECT UNIX_TIMESTAMP(expire) FROM `buffet_time` WHERE account_id = " + getarg(0) + " LIMIT 1", .@expire);
	if(.@rst == -1)
		return false;
	return .@expire;
}

// callfunc("F_AddBuffetTime", <account_id>, <second>);
function	script	F_AddBuffetTime	{
	.@aid = getarg(0);
	.@sec = getarg(1);
	.@time = callfunc("F_GetBuffetTime", .@aid);
	if(.@time >= gettimetick(2))
		.@rst = query_sql("REPLACE INTO `buffet_time` VALUES (" + .@aid + ", FROM_UNIXTIME(" + (.@time + .@sec) + "), 0)");
	else
		.@rst = query_sql("REPLACE INTO `buffet_time` VALUES (" + .@aid + ", FROM_UNIXTIME(" + (gettimetick(2) + .@sec) + "), 0)");
	return (.@rst != -1);
}

// ใช้ตั๋ว VIP 1 วัน
function	script	1d_vip_ticket	{
	// ถ้ายังไม่มี VIP อยู่ ให้เพิ่ม VIP 1 วัน
	if (!vip_status(VIP_STATUS_EXPIRE)) {
		atcommand "@vip +1d "+strcharinfo(0)+"";
		#CASH_VIP_TIME = gettimetick(2) + 86400;
		sc_start SC_VIPSTATUS, 86400 * 1000, 1;
		delitem "VIP_Ticket_1_Day",1;
		end;
	}
	// มี VIP อยู่แล้ว: ไม่อนุญาตซ้อน ให้แจ้งหมดอายุ
	if (#CASH_VIP_TIME > gettimetick(2)) {
		message strcharinfo(0),"VIP ของคุณจะหมดอายุใน: "+ gettimestr("%d-%m-%Y %H:%M:%S",21,vip_status(VIP_STATUS_EXPIRE)) +"";
		dispbottom "จนกว่าจะหมดเวลาจะไม่สามารถใช้อันใหม่ได้!";
		end;
	}
}
