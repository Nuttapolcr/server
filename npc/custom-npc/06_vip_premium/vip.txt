	end;

OnPCLoginEvent:
OnVipCheck:
	if(vip_status(VIP_STATUS_ACTIVE)){
		dispbottom "[VIP System] Vip ของท่านจะหมดลงเมื่อถึง " + gettimestr("%H:%M:%S %d-%m-%Y",21,vip_status(VIP_STATUS_EXPIRE)),0xff4000;
