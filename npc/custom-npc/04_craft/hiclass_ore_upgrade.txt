//= Random ore upgrades from the HICLASS sheet.

morocc,139,150,4	script	NPC อัพเกรดแร่	810,{
	mes "[NPC อัพเกรดแร่]";
	mes "เลือกประเภทแร่ที่ต้องการอัพเกรด";
	next;

	.@menu = select(
		"[Rare] Jewel (NT) -> [Epic] Jewel",
		"[Epic] Jewel (NT) -> [Legend] Jewel",
		"[Rare] Essence (NT) -> [Epic] Essence",
		"[Epic] Essence (NT) -> [Legend] Essence",
		"[Rare] Fragment (NT) -> [Epic] Fragment",
		"[Epic] Fragment (NT) -> [Legend] Fragment",
		"ยกเลิก"
	);
	if (.@menu == 7) {
		close;
	}

	.@index = .@menu - 1;

	mes "[NPC อัพเกรดแร่]";
	mes "วัตถุดิบที่ใช้:";
	mes "- ^i[" + .InputItem[.@index] + "] x " + .InputAmount[.@index];
	mes "- ^i[1000373] x " + .FragmentAmount[.@index];
	mes "- Zeny " + .ZenyCost[.@index];
	mes " ";
	mes "โอกาสได้รับ:";
	for (.@i = 0; .@i < .RewardCount[.@index]; .@i++) {
		.@reward_index = .RewardStart[.@index] + .@i;
		mes "- ^i[" + .RewardItem[.@reward_index] + "] x " + .RewardAmount[.@reward_index] + " (" + .RewardRate[.@reward_index] + "%)";
	}
	next;

	if (select("ยืนยัน:ยกเลิก") == 2) {
		close;
	}

	if (
		countitem(.InputItem[.@index]) < .InputAmount[.@index] ||
		countitem(1000373) < .FragmentAmount[.@index] ||
		Zeny < .ZenyCost[.@index]
	) {
		mes "[NPC อัพเกรดแร่]";
		mes "วัตถุดิบไม่เพียงพอ";
		close;
	}
	for (.@i = 0; .@i < .RewardCount[.@index]; .@i++) {
		.@reward_index = .RewardStart[.@index] + .@i;
		if (!checkweight(.RewardItem[.@reward_index], .RewardAmount[.@reward_index])) {
			mes "[NPC อัพเกรดแร่]";
			mes "กรุณาลดน้ำหนักหรือเคลียร์ช่องเก็บของก่อน";
			close;
		}
	}

	delitem .InputItem[.@index],.InputAmount[.@index];
	delitem 1000373,.FragmentAmount[.@index];
	Zeny -= .ZenyCost[.@index];

	.@roll = rand(100) + 1;
	.@rate = 0;
	for (.@i = 0; .@i < .RewardCount[.@index]; .@i++) {
		.@reward_index = .RewardStart[.@index] + .@i;
		.@rate += .RewardRate[.@reward_index];
		if (.@roll <= .@rate) {
			.@reward = .RewardItem[.@reward_index];
			.@amount = .RewardAmount[.@reward_index];
			break;
		}
	}

	getitem .@reward,.@amount;
	mes "[NPC อัพเกรดแร่]";
	mes "อัพเกรดสำเร็จ";
	mes "ได้รับ ^i[" + .@reward + "] x " + .@amount;
	close;

OnInit:
	setarray .InputItem[0],
		1001685,1001686,1001688,1001689,1001691,1001692;
	setarray .InputAmount[0],
		5,10,5,10,5,10;
	setarray .FragmentAmount[0],
		5,10,5,10,5,10;
	setarray .ZenyCost[0],
		100000,200000,100000,200000,100000,200000;
	setarray .RewardStart[0],
		0,6,12,16,20,24;
	setarray .RewardCount[0],
		6,6,4,4,4,4;
	setarray .RewardItem[0],
		1000398,1001686,1000398,1001686,1000398,1001686,
		1000403,1001687,1000403,1001687,1000403,1001687,
		1001139,1001689,1001139,1001689,
		1001141,1001690,1001141,1001690,
		1000442,1001692,1000442,1001692,
		1000447,1001693,1000447,1001693;
	setarray .RewardAmount[0],
		1,1,2,2,3,3,
		1,1,2,2,3,3,
		1,1,2,2,
		1,1,2,2,
		1,1,2,2,
		1,1,2,2;
	setarray .RewardRate[0],
		20,30,15,25,4,6,
		20,30,15,25,4,6,
		30,40,13,17,
		30,40,13,17,
		30,40,13,17,
		30,40,13,17;

	waitingroom "NPC อัพเกรดแร่",0;
	end;
}
