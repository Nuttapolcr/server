//===== HICLASS Ore Upgrade ==================================
//= Random ore upgrades from the HICLASS sheet.
//============================================================

morocc,139,150,4	script	NPC อัพเกรดแร่	810,{
	mes "[NPC อัพเกรดแร่]";
	mes "เลือกประเภทแร่ที่ต้องการอัพเกรด";
	next;

	.@menu = select(
		"[Rare] Jewel (NT) -> [Epic] Jewel",
		"[Epic] Jewel (NT) -> [Legend] Jewel",
		"[Rare] Essence (NT) -> [Epic] Essence",
		"[Epic] Essence (NT) -> [Legend] Essence",
		"[Rare] Fragment (NT) -> [Epic] Fragment",
		"[Epic] Fragment (NT) -> [Legend] Fragment",
		"ยกเลิก"
	);
	if (.@menu == 7) {
		close;
	}

	.@index = .@menu - 1;

	mes "[NPC อัพเกรดแร่]";
	mes "วัตถุดิบที่ใช้:";
	mes "- ^i[" + .InputItem[.@index] + "] x " + .InputAmount[.@index];
	mes "- ^i[1000373] x " + .FragmentAmount[.@index];
	mes "- Zeny " + .ZenyCost[.@index];
	mes " ";
	mes "โอกาสได้รับ:";
	mes "- ^i[" + .TradeReward[.@index] + "] x 1 (40%)";
	mes "- ^i[" + .NTReward[.@index] + "] x 1 (60%)";
	next;

	if (select("ยืนยัน:ยกเลิก") == 2) {
		close;
	}

	if (
		countitem(.InputItem[.@index]) < .InputAmount[.@index] ||
		countitem(1000373) < .FragmentAmount[.@index] ||
		Zeny < .ZenyCost[.@index]
	) {
		mes "[NPC อัพเกรดแร่]";
		mes "วัตถุดิบไม่เพียงพอ";
		close;
	}

	delitem .InputItem[.@index],.InputAmount[.@index];
	delitem 1000373,.FragmentAmount[.@index];
	Zeny -= .ZenyCost[.@index];

	if (rand(100) < 40) {
		.@reward = .TradeReward[.@index];
	} else {
		.@reward = .NTReward[.@index];
	}

	getitem .@reward,1;
	mes "[NPC อัพเกรดแร่]";
	mes "อัพเกรดสำเร็จ";
	mes "ได้รับ ^i[" + .@reward + "] x 1";
	close;

OnInit:
	setarray .InputItem[0],
		1001685,1001686,1001688,1001689,1001691,1001692;
	setarray .InputAmount[0],
		5,10,5,10,5,10;
	setarray .FragmentAmount[0],
		20,30,20,30,20,30;
	setarray .ZenyCost[0],
		100000,200000,100000,200000,100000,200000;
	setarray .TradeReward[0],
		1000398,1000403,1001139,1001141,1000442,1000447;
	setarray .NTReward[0],
		1001686,1001687,1001689,1001690,1001692,1001693;

	waitingroom "NPC อัพเกรดแร่",0;
	end;
}
