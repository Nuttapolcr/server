		getitemname(13115)+":"+getitemname(21000)+":"+getitemname(16019)+":"+
		getitemname(18112)+":"+getitemname(13316)+":"+getitemname(1030046)+":"+
		getitemname(1030047)+":"+getitemname(1030048)+":"+getitemname(15015)+":"+
		getitemname(2562)+":"+getitemname(2460)+":"+getitemname(2149)+":ยกเลิก")) {
		case 1: .@item_id = 1292; break;
		case 2: .@item_id = 1394; break;
		case 3: .@item_id = 1491; break;
		case 4: .@item_id = 1585; break;
		case 5: .@item_id = 2015; break;
		case 6: .@item_id = 13071; break;
		case 7: .@item_id = 13115; break;
		case 8: .@item_id = 21000; break;

		case 10: .@item_id = 18112; break;

		case 15: .@item_id = 15015; break;
		case 16: .@item_id = 2562; break;
		case 17: .@item_id = 2460; break;

			close;

		mes "[Rookie Box]";
		mes "น้ำหนักในตัวคุณไม่เพียงพอ";

	getitem .@item_id,1;
	mes "[Rookie Box]";
	mes "คุณได้รับ ^00AA00"+getitemname(.@item_id)+"^000000";
	close;
}
