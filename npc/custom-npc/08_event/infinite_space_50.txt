function	script	F_ValorInfiniteDay	{
	// Shift local server time by five hours so the date changes at 05:00.
	return atoi(gettimestr("%Y%m%d",21,gettimetick(2) - 18000));
}

	case 1013: case 1025: case 1030: case 1242:
		.@level = 103; .@hp = 70000; break;
	case 1055: case 1019: case 1105: case 1160:
		.@level = 104; .@hp = 85000; break;
	case 1095: case 1040: case 1001: case 1011:
		.@level = 105; .@hp = 100000; break;
	case 1058: case 1165: case 1254: case 1321:
		.@level = 106; .@hp = 115000; break;
	case 1687: case 1323: case 1391: case 1317:
		.@level = 107; .@hp = 130000; break;
	case 1271: case 1064: case 1279:
		.@level = 108; .@hp = 145000; break;
	case 1274: case 1493:
		.@level = 109; .@hp = 160000; break;
	case 1089: case 1092: case 1088: case 1093: case 1283:
		.@level = 105; .@hp = 900000; break;

function	script	F_ValorInfiniteWarpToFloor	{
	.@map$ = getarg(0);
	.@floor = getarg(1);
	if (.@floor < 1 || .@floor > 50)
		.@floor = 1;

	setarray .@entry_x[0],
		30,30,30,30,30,30,30,30,30,30,
		112,112,112,112,112,112,112,112,112,112,
		198,198,198,198,198,198,198,198,198,198,
		280,280,280,280,280,280,280,280,280,280,
		362,362,362,362,362,362,362,362,362,362;
	setarray .@entry_y[0],
		10,41,73,105,137,220,253,285,317,349,
		10,41,73,105,137,220,253,285,317,349,
		10,41,73,105,137,220,245,277,309,341,
		10,41,73,105,137,220,245,277,309,341,
		10,41,73,105,137,220,245,277,309,341;
	warp .@map$,.@entry_x[.@floor - 1],.@entry_y[.@floor - 1];
	return;
}

			close;

		case IE_OK:
			.@floor = getinstancevar('floor,.@instance_id);
			.@map$ = instance_mapname("1@infi",.@instance_id);
			callfunc "F_ValorInfiniteWarpToFloor",.@map$,.@floor;
			end;
		case IE_OTHER:
			mes "ปาร์ตี้มี Instance อื่นทำงานอยู่";
			close;
		default:
			mes "ไม่สามารถกลับเข้าสู่ Infinity Space ได้";
			close;
		}
	}

	.@day = callfunc("F_ValorInfiniteDay");
	if (#VALOR_INFINITE_DAY == .@day) {
		mes "ไอดีเกมนี้ลง Infinity Space ไปแล้ว";
		mes "สามารถลงใหม่ได้หลังเวลา 05:00 น. ของวันถัดไป";
		close;
	}

	.@mode = select("Easy - รับความเสียหาย 100%:Hard - รับความเสียหาย 20%:ยกเลิก");
	if (.@mode == 3)

	mes "เลือกโหมด ^0000FF" + ((.@mode == 1) ? "Easy" : "Hard") + "^000000";
	mes "เมื่อเข้าสู่ดันเจี้ยนแล้ว ไอดีเกมนี้จะไม่สามารถลงอีกโหมดได้จนกว่าจะรีเซต";

		close;

		mes "สร้าง Infinity Space ไม่สำเร็จ (Error " + .@instance_id + ")";
		close;

	setinstancevar('vis_owner_aid,getcharid(3),.@instance_id);

	case IE_OK:
		#VALOR_INFINITE_DAY = .@day;
		end;
	default:
		instance_destroy .@instance_id;
		mes "เข้าสู่ Infinity Space ไม่สำเร็จ";
		close;
	}

	.instance_name$ = "Valor Infinite Space";
	waitingroom "Infinity Space",0;
	end;
}
