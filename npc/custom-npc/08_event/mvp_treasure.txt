//===== Custom NPC: MVP Treasure ============================================
//= Source: Google Sheet gid=838586176 (บอสเวอร์ชั่นอัพเกรด รวมtitanด้วย)
//= When a listed MVP boss dies, clickable treasure chests spawn at the death
//= spot. The player clicks a chest, a per-color "open" bar runs, and on
//= completion the chest yields ONE weighted-random reward and is removed.
//=
//= Box counts per kill (independent per-color rolls, total capped at 10):
//=   Blue   : 100%       -> rand(2,6)   open 5s
//=   Red    : 10%        -> rand(1,4)   open 10s
//=   Purple : 1%         -> rand(1,3)   open 15s
//=
//= Loyal pet open-time bonuses from the Pet sheet:
//=   Pharaoh : 20% faster
//=   Medjay / CP Heka : 60% faster
//=
//= Rewards come from item_group_db (db/pre-re/item_group_db.yml):
//=   IG_MVP_TREASURE_BLUE / _RED / _PURPLE
//=   SubGroup 1 = standard bosses; SubGroup 2 = rare bosses (adds stones).
//=   One item is pulled per open via getrandgroupitem() (Rate = weight; Amount is honored).
//=
//= Implementation notes:
//=  - Each map uses a fixed pool of 10 treasure-chest NPCs. Slots are moved,
//=    enabled, and disabled for reuse, so opening a chest never accumulates NPCs.
//=  - Moving during the open cancels progressbar -> the script ends BEFORE the
//=    reward block, so the chest is NOT consumed (player can retry).
//=  - A per-slot temp flag guards against double-loot if two players open the
//=    same chest at once (script engine is single-threaded).
//=  - Unopened chests auto-despawn after .despawn seconds via their own timer.
//===========================================================================

-	script	MvpTreasure	FAKE_NPC,{
	end;

OnInit:
	// --- Standard bosses: standard loot tables ---
	// NOTE: sheet's "Giant Eggring" (A3) has no pre-re mob entry; add its id here if created.
	setarray .tierA[0],
		1087,1147,1190,1115,1086,1038,1159,1389,1046,		// OrcHero,Maya,OrcLord,Eddga,GTB,Osiris,Phreeoni,Dracula,Doppel
		1059,1150,1039,1157,1112,1251,2068,1373,1272,		// Mistress,Moonlight,Baphomet,Pharaoh,Drake,StormyKnight,Boitata,LoD,DarkLord
		1885,1630,1252,1708,1623,1492,1583,1312,			// Gopinich,WhiteLady,Hatii,Thanatos,RSX,SamuraiSpecter,TaoGunka,TurtleGen
		1785,1734,1685,1719;							// Atroce,KielD-01,Vesper,Detardeurus
	// --- Rare bosses: high-tier loot (adds Rare/Epic stones) ---
	setarray .tierS[0],
		1511,1418,1871,1768,1779,1688,1751,1832,1873,1874,1917;	// AmonRa,EvilSnakeLord,FallenBishop,Gloom,Ktullanux,LadyTanee,ValkyrieRandgris,Ifrit,Beelzebub forms,WoundedMorocc

	// Only these field/dungeon spawns use the public respawn announcement.
	// Boss rooms, instances, and event maps are deliberately excluded.
	setarray .natural_spawn$[0],
		"abyss_03:1719","ama_dun03:1492","anthell02:1147","beach_dun:1583",
		"bra_dun02:2068","ein_dun02:1623","gef_dun01:1389","gef_dun02:1046",
		"gl_chyard:1272","gld_dun01:1115","gld_dun02:1046","gld_dun03:1147",
		"gld_dun04:1272","in_sphinx5:1157","jupe_core:1685","kh_dun02:1734",
		"lou_dun03:1630","moc_pryd04:1038","mosk_dun03:1885","pay_dun04:1150",
		"prt_maze03:1039","prt_sewb4:1086","treasure02:1112","tur_dun04:1312",
		"xmas_dun02:1251","gef_fild02:1087","gef_fild10:1190","gef_fild14:1087",
		"xmas_fild01:1252","mjolnir_04:1059","moc_fild17:1159","niflheim:1373",
		"pay_fild11:1115","ra_fild02:1785","ra_fild03:1785","ra_fild04:1785",
		"ve_fild01:1785","ve_fild02:1785",
		"abbey02:1871","abbey03:1873","abbey03:1874","ayo_dun02:1688","gon_dun03:1418",
		"ice_dun03:1779","moc_pryd06:1511","odin_tem03:1751","ra_san05:1768","thor_v03:1832",
		"moc_fild22:1917";

	.chest_sprite = 1324;	// Treasure Chest sprite
	.spawn_radius = 3;		// random scatter (cells) around the death spot
	.despawn      = 600;	// seconds before an unopened chest disappears
	.pool_size    = 10;	// maximum concurrent chests per map
	end;

OnNPCKillEvent:
	// Valor MVP Battlefield has its own normal-drop rules and Thanatos chest.
	if (strcharinfo(3) == "valor_pk") end;

	// which tier is the killed mob?
	.@tier = 0;
	if (inarray(.tierA, killedrid) >= 0) .@tier = 1;
	else if (inarray(.tierS, killedrid) >= 0) .@tier = 2;
	if (!.@tier) end;

	callsub OnCreateChests, .@tier;

	// Natural bosses announce when their configured respawn delay completes.
	if (getmapxy(.@m$, .@x, .@y, BL_PC) != 0) end;
	if (inarray(.natural_spawn$, .@m$ + ":" + killedrid) >= 0) {
		.@mob_id = killedrid;
		.@respawn_delay = (.@tier == 2) ? 3600000 : 1800000;
		sleep .@respawn_delay;
		announce "บอส " + getmonsterinfo(.@mob_id, MOB_NAME) + " เกิดแล้ว", bc_all;
	}
	end;

// Bio Lab 3 MVP clones are spawned with their own death event, so they do not
// reach OnNPCKillEvent. lhz_dun.txt forwards that event here with the killer RID.
OnLhzCloneMvpDead:
	if (killedrid < 1646 || killedrid > 1651) end;
	callsub OnCreateChests, 1;	// Tier A
	end;

// callsub OnCreateChests, <tier>
OnCreateChests:
	.@tier = getarg(0);
	// death spot ~ killer's location
	if (getmapxy(.@m$, .@x, .@y, BL_PC) != 0) return;

	// independent per-color box counts, capped at 10 total (trim Blue first)
	.@blue   = rand(2,6);							// 100%
	.@red    = (rand(100) < 10) ? rand(1,4) : 0;	// 10%
	.@purple = (rand(100) < 1)  ? rand(1,3) : 0;	// 1%
	.@cap = 10 - .@purple - .@red;
	if (.@cap < 0) .@cap = 0;
	if (.@blue > .@cap) .@blue = .@cap;

	callsub OnSpawn, .@m$, .@x, .@y, "P", .@tier, .@purple;
	callsub OnSpawn, .@m$, .@x, .@y, "R", .@tier, .@red;
	callsub OnSpawn, .@m$, .@x, .@y, "B", .@tier, .@blue;
	return;

// callsub OnSpawn, <map$>, <x>, <y>, <color$>, <tier>, <count>
OnSpawn:
	.@m$   = getarg(0);
	.@x    = getarg(1);
	.@y    = getarg(2);
	.@col$ = getarg(3);
	.@tier = getarg(4);
	.@cnt  = getarg(5);
	if (.@cnt <= 0) return;

	if (.@col$ == "R")      .@disp$ = "กล่องสมบัติแดง";
	else if (.@col$ == "P") .@disp$ = "กล่องสมบัติม่วง";
	else                     .@disp$ = "กล่องสมบัติฟ้า";

	callsub OnEnsurePool, .@m$;

	for (.@k = 0; .@k < .@cnt; .@k++) {
		.@slot = -1;
		for (.@i = 0; .@i < .pool_size; .@i++) {
			if (!getd("$@mtb_active_"+.@m$+"_"+.@i)) {
				.@slot = .@i;
				break;
			}
		}
		if (.@slot < 0) return;	// all 10 slots are in use; never create more NPCs

		.@sx = .@x + rand(-.spawn_radius, .spawn_radius);
		.@sy = .@y + rand(-.spawn_radius, .spawn_radius);
		if (.@sx < 0) .@sx = 0;	// avoid out-of-bounds near map edge
		if (.@sy < 0) .@sy = 0;
		.@npc$ = getd("$@mtb_pool_"+.@m$+"_"+.@slot+"$");
		if (.@npc$ == "") return;
		setd "$@mtb_active_"+.@m$+"_"+.@slot, 1;
		setd "$@mtb_claimed_"+.@m$+"_"+.@slot, 0;
		setd "$@mtb_color_"+.@m$+"_"+.@slot+"$", .@col$;
		setd "$@mtb_tier_"+.@m$+"_"+.@slot, .@tier;
		setnpcdisplay .@npc$, .@disp$+"#"+.@m$+"|"+.@slot;
		movenpc .@npc$, .@sx, .@sy;
		enablenpc .@npc$;
		donpcevent .@npc$+"::OnArm";
	}
	return;

// callsub OnEnsurePool, <map$>
OnEnsurePool:
	.@m$ = getarg(0);
	if (getd("$@mtb_pool_ready_"+.@m$)) return;
	for (.@slot = 0; .@slot < .pool_size; .@slot++) {
		.@name$ = "MVP Treasure#"+.@m$+"|"+.@slot;
		.@npc$ = duplicate("MvpTreasureChest", .@m$, 0, 0, .@name$, .chest_sprite);
		if (.@npc$ == "") continue;
		setd "$@mtb_pool_"+.@m$+"_"+.@slot+"$", .@npc$;
		disablenpc .@npc$;
	}
	setd "$@mtb_pool_ready_"+.@m$, 1;
	return;
}

//=== Treasure chest pool template ===========================================
-	script	MvpTreasureChest	FAKE_NPC,{
	if (#pk_protect_until > gettimetick(2)) {
		dispbottom "[MVP Treasure] PK Protection users cannot open MVP Treasure.";
		end;
	}

	// hidden name = "<map>|<pool slot>"
	explode(.@p$, strnpcinfo(2), "|");
	.@map$ = .@p$[0];
	.@slot = atoi(.@p$[1]);
	if (!getd("$@mtb_active_"+.@map$+"_"+.@slot)) end;
	.@col$ = getd("$@mtb_color_"+.@map$+"_"+.@slot+"$");
	.@tier = getd("$@mtb_tier_"+.@map$+"_"+.@slot);

	// already opened?
	if (getd("$@mtb_claimed_"+.@map$+"_"+.@slot)) { disablenpc strnpcinfo(3); end; }

	// Clicking an active boss chest immediately ends post-spawn invincibility.
	endspawnprotection;

	// open time + bar color by box color
	.@sec = 5; .@bar$ = "0x33B5FF";						// Blue
	if (.@col$ == "R")      { .@sec = 10; .@bar$ = "0xFF3333"; }	// Red
	else if (.@col$ == "P") { .@sec = 15; .@bar$ = "0xCC33FF"; }	// Purple

	// Pet bonuses only apply at the same Loyal intimacy used by pet_db.yml.
	if (getpetinfo(PETINFO_INTIMATE) >= PET_INTIMATE_LOYAL) {
		.@pet_class = getpetinfo(PETINFO_CLASS);
		if (.@pet_class == 1157) { // Pharaoh
			.@sec = .@sec * 80 / 100;
		}
		else {
			setarray .@fast_open_pet[0], 22397,22402;
			if (inarray(.@fast_open_pet, .@pet_class) >= 0)
				.@sec = .@sec * 40 / 100;
		}
	}

	.@sec = vip_rank_box_time(.@sec, 1);

	progressbar .@bar$, .@sec;	// moving -> script ENDs here, chest is kept

	// --- reached only on a completed open ---
	if (getd("$@mtb_claimed_"+.@map$+"_"+.@slot)) end;	// someone else finished first
	setd "$@mtb_claimed_"+.@map$+"_"+.@slot, 1;	// claim
	setd "$@mtb_active_"+.@map$+"_"+.@slot, 0;
	stopnpctimer;

	// Grant one weighted reward from item_group_db; quantity 0 uses its configured Amount.
	.@grp = IG_MVP_TREASURE_BLUE;
	if (.@col$ == "R")      .@grp = IG_MVP_TREASURE_RED;
	else if (.@col$ == "P") .@grp = IG_MVP_TREASURE_PURPLE;
	.@sub = .@tier;
	getrandgroupitem .@grp,0,.@sub;

	disablenpc strnpcinfo(3);
	end;

OnArm:
	initnpctimer;
	end;

OnTimer600000:
	stopnpctimer;
	explode(.@p$, strnpcinfo(2), "|");
	setd "$@mtb_active_"+.@p$[0]+"_"+atoi(.@p$[1]), 0;
	disablenpc strnpcinfo(3);
	end;
}
