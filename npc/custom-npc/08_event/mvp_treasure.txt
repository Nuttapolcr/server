//===== Custom NPC: MVP Treasure ============================================
//= Source: Google Sheet gid=838586176 (บอสเวอร์ชั่นอัพเกรด รวมtitanด้วย)
//= When a listed MVP boss dies, clickable treasure chests spawn at the death
//= spot. The player clicks a chest, a per-color "open" bar runs, and on
//= completion the chest yields ONE weighted-random reward and is removed.
//=
//= Box counts per kill (independent per-color rolls, total capped at 10):
//=   Blue   : 100%       -> rand(2,6)   open 10s
//=   Red    : 10%        -> rand(1,4)   open 20s
//=   Purple : 1%         -> rand(1,3)   open 30s
//=
//= Loyal pet open-time bonuses from the Pet sheet:
//=   Pharaoh : 20% faster
//=   Medjay / CP Heka : 60% faster
//=
//= Rewards come from item_group_db (db/pre-re/item_group_db.yml):
//=   IG_MVP_TREASURE_BLUE / _RED / _PURPLE
//=   SubGroup 1 = standard bosses; SubGroup 2 = rare bosses (adds stones).
//=   One item is pulled per open via getrandgroupitem() (Rate = weight; Amount is honored).
//=
//= Implementation notes:
//=  - Chests are runtime duplicates (duplicate()) with a treasure-chest sprite.
//=    Color/tier are encoded in the hidden part of the NPC name "<disp>#C_T_ID".
//=  - Moving during the open cancels progressbar -> the script ends BEFORE the
//=    reward block, so the chest is NOT consumed (player can retry).
//=  - A per-chest temp flag $@mtbo_<id> guards against double-loot if two
//=    players open the same chest at once (script engine is single-threaded).
//=  - Unopened chests auto-despawn after .despawn seconds via their own timer.
//=  - LIMITATION: opened/expired chests are disablenpc'd (hidden), not fully
//=    unloaded (rAthena has no delnpc script cmd); cleared on @reloadscript /
//=    server restart. The UID ring (1..100000) bounds the temp-var/name space.
//===========================================================================

-	script	MvpTreasure	FAKE_NPC,{
	end;

OnInit:
	// --- Standard bosses: standard loot tables ---
	// NOTE: sheet's "Giant Eggring" (A3) has no pre-re mob entry; add its id here if created.
	setarray .tierA[0],
		1087,1147,1190,1115,1086,1038,1159,1389,1046,		// OrcHero,Maya,OrcLord,Eddga,GTB,Osiris,Phreeoni,Dracula,Doppel
		1059,1150,1039,1157,1112,1251,2068,1373,1272,		// Mistress,Moonlight,Baphomet,Pharaoh,Drake,StormyKnight,Boitata,LoD,DarkLord
		1885,1630,1252,1708,1623,1492,1583,1312,			// Gopinich,WhiteLady,Hatii,Thanatos,RSX,SamuraiSpecter,TaoGunka,TurtleGen
		1785,1734,1685,1719;							// Atroce,KielD-01,Vesper,Detardeurus
	// --- Rare bosses: high-tier loot (adds Rare/Epic stones) ---
	setarray .tierS[0],
		1511,1418,1871,1768,1779,1688,1751,1832,1873,1917;	// AmonRa,EvilSnakeLord,FallenBishop,Gloom,Ktullanux,LadyTanee,ValkyrieRandgris,Ifrit,Beelzebub,WoundedMorocc

	.chest_sprite = 1324;	// Treasure Chest sprite
	.spawn_radius = 3;		// random scatter (cells) around the death spot
	.despawn      = 600;	// seconds before an unopened chest disappears
	end;

OnNPCKillEvent:
	// which tier is the killed mob?
	.@tier = 0;
	if (inarray(.tierA, killedrid) >= 0) .@tier = 1;
	else if (inarray(.tierS, killedrid) >= 0) .@tier = 2;
	if (!.@tier) end;

	// death spot ~ killer's location
	if (getmapxy(.@m$, .@x, .@y, BL_PC) != 0) end;

	// independent per-color box counts, capped at 10 total (trim Blue first)
	.@blue   = rand(2,6);							// 100%
	.@red    = (rand(100) < 10) ? rand(1,4) : 0;	// 10%
	.@purple = (rand(100) < 1)  ? rand(1,3) : 0;	// 1%
	.@cap = 10 - .@purple - .@red;
	if (.@cap < 0) .@cap = 0;
	if (.@blue > .@cap) .@blue = .@cap;

	callsub OnSpawn, .@m$, .@x, .@y, "P", .@tier, .@purple;
	callsub OnSpawn, .@m$, .@x, .@y, "R", .@tier, .@red;
	callsub OnSpawn, .@m$, .@x, .@y, "B", .@tier, .@blue;
	end;

// callsub OnSpawn, <map$>, <x>, <y>, <color$>, <tier>, <count>
OnSpawn:
	.@m$   = getarg(0);
	.@x    = getarg(1);
	.@y    = getarg(2);
	.@col$ = getarg(3);
	.@tier = getarg(4);
	.@cnt  = getarg(5);
	if (.@cnt <= 0) return;

	if (.@col$ == "R")      .@disp$ = "กล่องสมบัติแดง";
	else if (.@col$ == "P") .@disp$ = "กล่องสมบัติม่วง";
	else                    .@disp$ = "กล่องสมบัติฟ้า";

	for (.@k = 0; .@k < .@cnt; .@k++) {
		// rolling unique id (ring 1..100000)
		$MTB_UID++;
		if ($MTB_UID > 100000 || $MTB_UID < 1) $MTB_UID = 1;
		.@id = $MTB_UID;
		setd "$@mtbo_"+.@id, 0;	// clear claim flag for the reused id

		.@sx = .@x + rand(-.spawn_radius, .spawn_radius);
		.@sy = .@y + rand(-.spawn_radius, .spawn_radius);
		if (.@sx < 0) .@sx = 0;	// avoid out-of-bounds near map edge
		if (.@sy < 0) .@sy = 0;
		.@name$ = .@disp$ + "#" + .@col$ + "_" + .@tier + "_" + .@id;
		.@ex$ = duplicate("MvpTreasureChest", .@m$, .@sx, .@sy, .@name$, .chest_sprite);
		if (.@ex$ != "") donpcevent .@ex$ + "::OnArm";
	}
	return;
}

//=== Treasure chest template (clicked duplicates run this body) =============
-	script	MvpTreasureChest	FAKE_NPC,{
	if (#pk_protect_until > gettimetick(2)) {
		dispbottom "[MVP Treasure] PK Protection users cannot open MVP Treasure.";
		end;
	}

	// hidden name = "<color>_<tier>_<id>"
	explode(.@p$, strnpcinfo(2), "_");
	.@col$ = .@p$[0];
	.@tier = atoi(.@p$[1]);
	.@id   = .@p$[2];

	// already opened?
	if (getd("$@mtbo_"+.@id)) { disablenpc strnpcinfo(3); end; }

	// open time + bar color by box color
	.@sec = 10; .@bar$ = "0x33B5FF";						// Blue
	if (.@col$ == "R")      { .@sec = 20; .@bar$ = "0xFF3333"; }	// Red
	else if (.@col$ == "P") { .@sec = 30; .@bar$ = "0xCC33FF"; }	// Purple

	// Pet bonuses only apply at the same Loyal intimacy used by pet_db.yml.
	if (getpetinfo(PETINFO_INTIMATE) >= PET_INTIMATE_LOYAL) {
		.@pet_class = getpetinfo(PETINFO_CLASS);
		if (.@pet_class == 1157) { // Pharaoh
			.@sec = .@sec * 80 / 100;
		}
		else {
			setarray .@fast_open_pet[0], 22397,22402;
			if (inarray(.@fast_open_pet, .@pet_class) >= 0)
				.@sec = .@sec * 40 / 100;
		}
	}

	.@sec = vip_rank_box_time(.@sec, 1);

	progressbar .@bar$, .@sec;	// moving -> script ENDs here, chest is kept

	// --- reached only on a completed open ---
	if (getd("$@mtbo_"+.@id)) end;	// someone else finished first
	setd "$@mtbo_"+.@id, 1;	// claim
	stopnpctimer;

	// Grant one weighted reward from item_group_db; quantity 0 uses its configured Amount.
	.@grp = IG_MVP_TREASURE_BLUE;
	if (.@col$ == "R")      .@grp = IG_MVP_TREASURE_RED;
	else if (.@col$ == "P") .@grp = IG_MVP_TREASURE_PURPLE;
	.@sub = .@tier;
	getrandgroupitem .@grp,0,.@sub;

	disablenpc strnpcinfo(3);
	end;

OnArm:
	initnpctimer;
	end;

OnTimer120000:
	stopnpctimer;
	disablenpc strnpcinfo(3);
	end;
}
