function	script	F_Cabinet_Roll	{
	.@type$ = getarg(0);
	.@count = getd("$Cabinet" + .@type$ + "Count");
	.@total = 0;
	for (.@i = 0; .@i < .@count; .@i++)
		.@total += getd("$Cabinet" + .@type$ + "Rate[" + .@i + "]");

	.@roll = rand(.@total) + 1;
	.@rate = 0;
	for (.@i = 0; .@i < .@count; .@i++) {
		.@rate += getd("$Cabinet" + .@type$ + "Rate[" + .@i + "]");
		if (.@roll <= .@rate)
			return .@i;
	}
	return .@count - 1;
}

function	script	F_Cabinet_ShowItems	{
	.@type$ = getarg(0);
	if (.@type$ == "Zeny") {
		.@shop$ = "CabinetZeny_Show";
	} else {
		.@shop$ = "CabinetCash_Show";
	}

	close2;
	npcshopattach .@shop$;
	callshop .@shop$, 1;
	end;
}

function	script	F_Cabinet_ShowTop	{
	.@type$ = getarg(0);
	if (.@type$ == "Zeny") {
		.@table$ = "cabinet_zeny_ranking";
		.@title$ = "ตู้ Zeny";
	} else {
		.@table$ = "cabinet_cash_ranking";
		.@title$ = "ตู้ Cash";
	}

	query_sql("SELECT `player_name`,`spins` FROM `" + .@table$ + "` ORDER BY `spins` DESC, `player_name` ASC LIMIT 10", .@name$, .@spins);
	mes "[" + .@title$ + " อันดับ 1-10]";
	if (!getarraysize(.@name$)) {
		mes "ยังไม่มีสถิติผู้กดตู้";
		return;
	}
	for (.@i = 0; .@i < getarraysize(.@name$); .@i++)
		mes (.@i + 1) + ". " + .@name$[.@i] + " - " + .@spins[.@i] + " ครั้ง";
	return;
}

function	script	F_Cabinet_RecordSpin	{
	.@type$ = getarg(0);
	if (.@type$ == "Zeny")
		.@table$ = "cabinet_zeny_ranking";
	else
		.@table$ = "cabinet_cash_ranking";

	query_sql("INSERT INTO `" + .@table$ + "` (`char_id`,`player_name`,`spins`) VALUES (" + getcharid(0) + ",'" + escape_sql(strcharinfo(0)) + "',1) ON DUPLICATE KEY UPDATE `player_name`=VALUES(`player_name`), `spins`=`spins`+1");
	return;
}

function	script	F_Cabinet_ResetRank	{
	.@type$ = getarg(0);
	if (.@type$ == "Zeny")
		query_sql("TRUNCATE TABLE `cabinet_zeny_ranking`");
	else if (.@type$ == "Cash")
		query_sql("TRUNCATE TABLE `cabinet_cash_ranking`");
	else {
		query_sql("TRUNCATE TABLE `cabinet_zeny_ranking`");
		query_sql("TRUNCATE TABLE `cabinet_cash_ranking`");
	}
	return;
}
