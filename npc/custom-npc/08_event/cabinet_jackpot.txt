function	script	F_Cabinet_Roll	{
	.@type$ = getarg(0);
	.@count = getd("$CabinetHW" + .@type$ + "Count");
	.@total = 0;
	for (.@i = 0; .@i < .@count; .@i++)
		.@total += getd("$CabinetHW" + .@type$ + "Rate[" + .@i + "]");

	.@roll = rand(.@total) + 1;
	.@rate = 0;
	for (.@i = 0; .@i < .@count; .@i++) {
		.@rate += getd("$CabinetHW" + .@type$ + "Rate[" + .@i + "]");
		if (.@roll <= .@rate)
			return .@i;
	}
	return .@count - 1;
}

function	script	F_Cabinet_ShowItems	{
	.@type$ = getarg(0);
	if (.@type$ == "Zeny")
		.@shop$ = "CabinetZeny_Show";
	else
		.@shop$ = "CabinetCash_Show";

	close2;
	npcshopattach .@shop$;
	callshop .@shop$, 1;
	end;
}

	.@choice = select("กดตู้:ดูตัวอย่างไอเทม:ยกเลิก");
	if (.@choice == 2) {
		callfunc "F_Cabinet_ShowItems", "Zeny";
		end;
	}
	if (.@choice != 1)
		close;

	.@index = callfunc("F_Cabinet_Roll", "Zeny");
	.@item = $CabinetHWZenyItem[.@index];
	.@amount = $CabinetHWZenyAmount[.@index];
	.@rate = $CabinetHWZenyRate[.@index];
	.@next_spins = CabinetHWZenySpins071069 + 1;
	.@guarantee = (.@next_spins % 120 == 0);
	if (Zeny < 500000) {
		mes "[ตู้ Zeny]";
		mes "ต้องมี Zeny 500,000";
		close;
	}
	deletearray .@check_item;
	deletearray .@check_amount;
	.@check_item[0] = .@item;
	.@check_amount[0] = .@amount;
	if (.@guarantee) {
		.@check_item[1] = 1030066;
		.@check_amount[1] = 1;
	}
	if (!checkweight2(.@check_item, .@check_amount)) {
		mes "[ตู้ Zeny]";
		mes "กรุณาลดน้ำหนักหรือเคลียร์ช่องเก็บของก่อน";
		close;
	}

	Zeny -= 500000;
	$CabinetHWZenyJackpot071069 += 30000;
	CabinetHWZenySpins071069 = .@next_spins;
	specialeffect2 EF_COIN;
	getitem .@item, .@amount;
	if (.@rate <= 300)
		announce "[ตู้ Zeny] " + strcharinfo(0) + " ได้รับ " + getitemname(.@item) + " x" + .@amount + ".", bc_all, 0xFFFF00;
	if (.@guarantee) {
		getitem 1030066, 1;
		announce "[ตู้ Zeny] " + strcharinfo(0) + " ได้รับ " + getitemname(1030066) + " x1 จากการันตีครบ 120 ครั้ง.", bc_all, 0xFFFF00;
	}
	.@won_jackpot = 0;
	if (rand(1000) == 0) {
		.@jackpot = $CabinetHWZenyJackpot071069;
		$CabinetHWZenyJackpot071069 = 0;
		Zeny += .@jackpot;
		.@won_jackpot = .@jackpot;
		announce "[ตู้ Zeny] " + strcharinfo(0) + " ได้รับ Jackpot " + callfunc("F_InsertComma", .@jackpot) + " Zeny.", bc_all, 0xFFFF00;
	}
	donpcevent "ตู้ Zeny#cabinet::OnUpdateSign";
	mes "[ตู้ Zeny]";
	mes "คุณได้รับ " + getitemname(.@item) + " x" + .@amount + ".";
	if (.@guarantee)
		mes "ได้รับ " + getitemname(1030066) + " x1 จากการันตี";
	if (.@won_jackpot > 0)
		mes "ได้รับ Jackpot " + callfunc("F_InsertComma", .@won_jackpot) + " Zeny";
	next;
	goto L_Main;

OnBuyItem:
	deletearray @bought_nameid;
	deletearray @bought_quantity;
	message strcharinfo(0), "[ตู้ Zeny] รายการนี้ใช้แสดงตัวอย่างเท่านั้น ไม่สามารถซื้อได้";
	end;

OnUpdateSign:
	delwaitingroom;
	waitingroom "ตู้Zeny (Zenyสะสม " + callfunc("F_InsertComma", $CabinetHWZenyJackpot071069) + ")", 0;
	end;
}
