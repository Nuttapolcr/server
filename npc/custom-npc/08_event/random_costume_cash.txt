//!rathena
// Random costume cash gachas.
// Default cost can be changed in OnInit: $@RandomCostumeCash_Cost.

function	script	F_RandomCostumeCash_AddRange	{
	.@type$ = getarg(0);
	.@start = getarg(1);
	.@end = getarg(2);
	.@location = getarg(3);
	.@size = getd("$@RandomCostumeCash_" + .@type$ + "_Size");

	for (.@id = .@start; .@id <= .@end; .@id++) {
		if (getitemname(.@id) == "null")
			continue;

		if (getiteminfo(.@id, ITEMINFO_LOCATIONS) & .@location) {
			setd "$@RandomCostumeCash_" + .@type$ + "[" + .@size + "]", .@id;
			.@size++;
		}
	}

	setd "$@RandomCostumeCash_" + .@type$ + "_Size", .@size;
	return;
}

function	script	F_RandomCostumeCash_ShowItems	{
	.@type$ = getarg(0);
	.@title$ = getarg(1);
	.@size = getd("$@RandomCostumeCash_" + .@type$ + "_Size");
	.@per_page = 12;
	.@page = 0;

	if (.@size <= 0) {
		mes "[" + .@title$ + "]";
		mes "ไม่มีรายการไอเทม";
		close;
	}

	while (1) {
		clear;
		mes "[" + .@title$ + "]";
		mes "^777777==============================^000000";
		mes "รายการทั้งหมด: ^0000FF" + callfunc("F_InsertComma", .@size) + "^000000 ชิ้น";
		mes "หน้า: ^0000FF" + (.@page + 1) + " / " + ((.@size + .@per_page - 1) / .@per_page) + "^000000";
		mes "^777777==============================^000000";
		next;

		.@start = .@page * .@per_page;
		.@end = .@start + .@per_page;
		if (.@end > .@size)
			.@end = .@size;

		mes "^FF9900รายการของรางวัลในหน้านี้^000000";
		mes "^777777------------------------------^000000";
		for (.@i = .@start; .@i < .@end; .@i++) {
			.@item_id = getd("$@RandomCostumeCash_" + .@type$ + "[" + .@i + "]");
			mes "^FF9900" + (.@i + 1) + ".^000000 ^i[" + .@item_id + "] ^0000FF" + getitemname(.@item_id) + "^000000";
		}
		mes "^777777------------------------------^000000";
		next;

		if (.@page > 0 && .@end < .@size) {
			.@select = select("ก่อนหน้า:ถัดไป:กลับ");
			if (.@select == 1)
				.@page--;
			else if (.@select == 2)
				.@page++;
			else
				return;
		} else if (.@page > 0) {
			.@select = select("ก่อนหน้า:กลับ");
			if (.@select == 1)
				.@page--;
			else
				return;
		} else if (.@end < .@size) {
			.@select = select("ถัดไป:กลับ");
			if (.@select == 1)
				.@page++;
			else
				return;
		} else {
			select("กลับ");
			return;
		}
	}
}

function	script	F_RandomCostumeCash	{
	.@type$ = getarg(0);
	.@title$ = getarg(1);
	.@cost = getarg(2);
	.@size = getd("$@RandomCostumeCash_" + .@type$ + "_Size");

	while (1) {
		mes "[" + .@title$ + "]";
		mes "ราคา: ^FF9900" + callfunc("F_InsertComma", .@cost) + " Cash^000000 / ครั้ง";
		mes "Cash ของคุณ: ^009900" + callfunc("F_InsertComma", #CASHPOINTS) + "^000000 pt.";
		mes "จำนวนไอเทม: ^0000FF" + callfunc("F_InsertComma", .@size) + "^000000 ชิ้น";

		next;
		.@menu = select("สุ่ม 1 ครั้ง:ดูของทั้งหมด:ยกเลิก");
		if (.@menu == 2) {
			callfunc "F_RandomCostumeCash_ShowItems", .@type$, .@title$;
			clear;
			continue;
		}
		if (.@menu == 3)
			close;

		if (.@size <= 0) {
			mes "[" + .@title$ + "]";
			mes "ไม่มีรายการไอเทม";
			close;
		}

		if (#CASHPOINTS < .@cost) {
			mes "[" + .@title$ + "]";
			mes "Cash ของคุณไม่เพียงพอ";
			mes "ขาดอีก: ^FF0000" + callfunc("F_InsertComma", .@cost - #CASHPOINTS) + "^000000 pt.";
			close;
		}

		if (select("ยืนยันสุ่ม:ยกเลิก") == 2)
			close;

		.@item = getd("$@RandomCostumeCash_" + .@type$ + "[" + rand(.@size) + "]");
		if (!checkweight(.@item, 1)) {
			mes "[" + .@title$ + "]";
			mes "กรุณาลดน้ำหนักหรือช่องเก็บของก่อน";
			close;
		}

		#CASHPOINTS -= .@cost;
		getitem .@item, 1;
		announce "[" + .@title$ + "] " + strcharinfo(0) + " ได้รับ " + getitemname(.@item) + " x1.", bc_all, 0xFFFF00;
		mes "[" + .@title$ + "]";
		mes "คุณได้รับ:";
		mes "^i[" + .@item + "] ^0000FF" + getitemname(.@item) + "^000000";
		mes "Cash คงเหลือ: ^009900" + callfunc("F_InsertComma", #CASHPOINTS) + "^000000 pt.";
		close;
	}
}
