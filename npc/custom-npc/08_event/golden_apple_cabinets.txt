// Golden Apple cabinets from spreadsheet "ตู้5/8/69".

function	script	F_GoldenAppleCabinet_Roll	{
	.@type$ = getarg(0);
	.@count = getd("$GoldenApple" + .@type$ + "CabinetCount");
	.@total = 0;
	for (.@i = 0; .@i < .@count; .@i++)
		.@total += getd("$GoldenApple" + .@type$ + "CabinetRate[" + .@i + "]");

	.@roll = rand(.@total) + 1;
	.@rate = 0;
	for (.@i = 0; .@i < .@count; .@i++) {
		.@rate += getd("$GoldenApple" + .@type$ + "CabinetRate[" + .@i + "]");
		if (.@roll <= .@rate)
			return .@i;
	}
	return .@count - 1;
}

function	script	F_GoldenAppleCabinet_ShowItems	{
	.@type$ = getarg(0);
	.@shop$ = "GoldenApple" + .@type$ + "Cabinet_Show";
	close2;
	npcshopattach .@shop$;
	callshop .@shop$, 1;
	end;
}

function	script	F_GoldenAppleCabinet_ShowTop	{
	.@type$ = getarg(0);
	if (.@type$ == "Slice") {
		.@table$ = "golden_apple_slice_cabinet_ranking";
		.@title$ = "ตู้ Golden Apple Slice";
	} else {
		.@table$ = "golden_apple_cabinet_ranking";
		.@title$ = "ตู้ Golden Apple";
	}

	query_sql("SELECT `player_name`,`spins` FROM `" + .@table$ + "` ORDER BY `spins` DESC, `player_name` ASC LIMIT 10", .@name$, .@spins);
	mes "[" + .@title$ + " อันดับ 1-10]";
	if (!getarraysize(.@name$)) {
		mes "ยังไม่มีสถิติผู้หมุนตู้";
		return;
	}
	for (.@i = 0; .@i < getarraysize(.@name$); .@i++)
		mes (.@i + 1) + ". " + .@name$[.@i] + " - " + .@spins[.@i] + " ครั้ง";
	return;
}

function	script	F_GoldenAppleCabinet_RecordSpin	{
	.@type$ = getarg(0);
	if (.@type$ == "Slice")
		.@table$ = "golden_apple_slice_cabinet_ranking";
	else
		.@table$ = "golden_apple_cabinet_ranking";
	query_sql("INSERT INTO `" + .@table$ + "` (`char_id`,`player_name`,`spins`) VALUES (" + getcharid(0) + ",'" + escape_sql(strcharinfo(0)) + "',1) ON DUPLICATE KEY UPDATE `player_name`=VALUES(`player_name`), `spins`=`spins`+1");
	return;
}

	.@amount = $GoldenAppleSliceCabinetAmount[.@index];

		mes "กรุณาลดน้ำหนักหรือเคลียร์ช่องเก็บของก่อน";
		close;

	specialeffect2 EF_COIN;
	getitem .@item, .@amount;
	if (.@rate <= 600)
		announce "[ตู้ Golden Apple Slice] " + strcharinfo(0) + " ได้รับ " + getitemname(.@item) + " x" + .@amount + ".", bc_all, 0xFFFF00;

	deletearray @bought_nameid;
	deletearray @bought_quantity;
	message strcharinfo(0), "[ตู้ Golden Apple Slice] รายการนี้ใช้แสดงตัวอย่างเท่านั้น";
	end;
}
