//===== Valor MVP Battlefield ===============================================
//= Three public PK MVP maps cloned from gef_dun03.
//= Entry is free for every difficulty.
//===========================================================================

function	script	F_ValorMvpRestrictedGear	{
	.@difficulty = getarg(0);
	@valor_blocked_item = 0;
	if (.@difficulty < 1 || .@difficulty > 2)
		return 0;

	setarray .@slots[0],
		EQI_HEAD_TOP,EQI_HEAD_MID,EQI_HEAD_LOW,EQI_ARMOR,EQI_HAND_R,EQI_HAND_L,
		EQI_GARMENT,EQI_SHOES,EQI_ACC_L,EQI_ACC_R,EQI_COSTUME_HEAD_TOP,
		EQI_COSTUME_HEAD_MID,EQI_COSTUME_HEAD_LOW,EQI_COSTUME_GARMENT,
		EQI_SHADOW_ARMOR,EQI_SHADOW_WEAPON,EQI_SHADOW_SHIELD,EQI_SHADOW_SHOES,
		EQI_SHADOW_ACC_R,EQI_SHADOW_ACC_L;
	setarray .@epic[0],
		600017,630012,640020,540020,520008,500026,550024,560018,610021,510032,
		700032,570018,580017,540023,550026,650009,800003,500025,1030018,1030019,
		530013,700031,1030051,1030052,452029,452030,452031,452032,480425,480426,
		480427,480428,470282,470283,470284,470285,490503,490504,490505,490506,
		490491,490492,490493,490494,1030010;
	setarray .@legend[0],
		21016,1450,1600,28605,28107,13455,16041,1800,28008,28706,
		18121,1900,1996,1029019,2026,13328,13128,13468,1030022,1030023,
		1400,1030024,1030053,1030054,452033,452034,452035,452036,480429,480430,
		480431,480432,470286,470287,470288,470289,490507,490508,490509,490510,
		490495,490496,490497,490498,1030011;

	for (.@i = 0; .@i < getarraysize(.@slots); ++.@i) {
		if (!getequipisequiped(.@slots[.@i]))
			continue;
		.@item_id = getequipid(.@slots[.@i]);
		if (inarray(.@legend,.@item_id) >= 0 ||
			(.@difficulty == 1 && inarray(.@epic,.@item_id) >= 0)) {
			@valor_blocked_item = .@item_id;
			return 1;
		}
	}
	return 0;
}

morocc,159,105,3	script	Valor MVP Battlefield	752,{
	mes "[ สนามรบ MVP Valor ]";
	mes "สนามรบ MVP แบบ PK เปิดตลอดเวลา";
	mes "เข้าฟรีทุกระดับ ไม่ใช้ตั๋ว";

	switch(select("เข้าดัน Easy:เข้าดัน Normal:เข้าดัน Hard:ยกเลิก")) {
	case 1:
		.@difficulty = 1;
		.@map$ = "valor_easy";
		break;
	case 2:
		.@difficulty = 2;
		.@map$ = "valor_norm";
		break;
	case 3:
		.@difficulty = 3;
		.@map$ = "valor_hard";
		break;
	default:
		close;
	}

	if (callfunc("F_ValorMvpRestrictedGear",.@difficulty)) {
		mes "ไม่สามารถเข้าดันระดับนี้ขณะสวมใส่ไอเทมต่อไปนี้:";
		mes "^FF0000" + getitemname(@valor_blocked_item) + "^000000";
		close;
	}
	warp .@map$,0,0;
	end;
}

-	script	ValorMvpBattlefield	FAKE_NPC,{
	end;

OnPCLoginEvent:
OnPCLoadMapEvent:
	deltimer strnpcinfo(0)+"::OnGearCheck";
	.@difficulty = callsub(S_GetDifficulty);
	if (!.@difficulty)
		end;
	if (!callsub(S_ValidatePlayer,.@difficulty))
		end;
	callsub S_ScheduleChecks;
	end;

OnGearCheck:
	.@difficulty = callsub(S_GetDifficulty);
	if (!.@difficulty)
		end;
	if (!callsub(S_ValidatePlayer,.@difficulty))
		end;
	addtimer 1000,strnpcinfo(0)+"::OnGearCheck";
	end;

S_GetDifficulty:
	.@map$ = strcharinfo(3);
	if (.@map$ == "valor_easy") return 1;
	if (.@map$ == "valor_norm") return 2;
	if (.@map$ == "valor_hard") return 3;
	return 0;

S_ValidatePlayer:
	.@difficulty = getarg(0);
	if (callfunc("F_ValorMvpRestrictedGear",.@difficulty)) {
		dispbottom "[สนามรบ MVP Valor] ตรวจพบไอเทมต้องห้าม: " + getitemname(@valor_blocked_item);
		warp "morocc",170,84;
		return 0;
	}
	return 1;

S_ScheduleChecks:
	addtimer 1000,strnpcinfo(0)+"::OnGearCheck";
	return;

// Kills on these maps do not reach global OnNPCKillEvent handlers.
OnValorMobKill:
	.@map$ = strcharinfo(3);
	if (.@map$ == "valor_easy") {
		.@rare_rate = 300;
		.@stone_rate = 500;
	} else if (.@map$ == "valor_norm") {
		.@rare_rate = 500;
		.@stone_rate = 1000;
	} else if (.@map$ == "valor_hard") {
		.@rare_rate = 1000;
		.@stone_rate = 2000;
	} else {
		end;
	}
	getmapxy .@map$,.@x,.@y,BL_PC;
	.@boost = 100 + getstatus(SC_ITEMBOOST,1);
	.@rare_rate = min(10000,.@rare_rate * .@boost / 100);
	.@stone_rate = min(10000,.@stone_rate * .@boost / 100);
	if (rand(10000) < .@rare_rate) makeitem 1000445,1,.@map$,.@x,.@y;
	if (rand(10000) < .@rare_rate) makeitem 1000401,1,.@map$,.@x,.@y;
	if (rand(10000) < .@rare_rate) makeitem 1001140,1,.@map$,.@x,.@y;
	if (rand(10000) < .@stone_rate) makeitem 1000373,1,.@map$,.@x,.@y;
	end;
}
