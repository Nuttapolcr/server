//===== Valor MVP Battlefield ===============================================
//= Public PK MVP map cloned from gef_dun03 (client alias: valor_pk).
//= One ticket grants the account three hours of re-entry, including after death.
//===========================================================================

morocc,170,84,3	script	Valor MVP Battlefield	752,{
	.@now = gettimetick(2);
	mes "[ สนามรบ MVP Valor ]";
	mes "สนามรบ MVP แบบ PK เปิดตลอดเวลา";
	mes "ตั๋วใช้งานได้นาน 3 ชั่วโมง และกลับเข้าใหม่ได้แม้เสียชีวิต";
	if (#valor_mvp_until > .@now) {
		.@remain = #valor_mvp_until - .@now;
		mes "สิทธิ์เข้าสนามรบคงเหลือ " + (.@remain / 3600) + " ชั่วโมง " + ((.@remain % 3600) / 60) + " นาที";
	}
	switch(select("เข้าสนามรบ:ซื้อตั๋ว - 10,000,000 Zeny:ซื้อตั๋ว - CashCoin 30 เหรียญ + 1,000,000 Zeny:ยกเลิก")) {
	case 1:
		if (#valor_mvp_until <= .@now) {
			if (countitem(1030071) < 1) {
				mes "คุณต้องมี Gold Ticket MVP Battlefield จำนวน 1 ใบ";
				close;
			}
			delitem 1030071,1;
			#valor_mvp_until = .@now + 10800;
		}
		callsub S_ScheduleExpiry;
		warp "valor_pk",0,0;
		end;
	case 2:
		if (Zeny < 10000000) {
			mes "คุณมี Zeny ไม่เพียงพอ ต้องใช้ 10,000,000 Zeny";
			close;
		}
		Zeny -= 10000000;
		getitem 1030071,1;
		mes "ได้รับ Gold Ticket MVP Battlefield จำนวน 1 ใบแล้ว";
		close;
	case 3:
		if (Zeny < 1000000 || countitem(30001) < 30) {
			mes "คุณต้องมี CashCoin 30 เหรียญ และ Zeny 1,000,000";
			close;
		}
		delitem 30001,30;
		Zeny -= 1000000;
		getitem 1030071,1;
		mes "ได้รับ Gold Ticket MVP Battlefield จำนวน 1 ใบแล้ว";
		close;
	}
	close;

OnInit:
	.map$ = "valor_pk";
	.return_map$ = "morocc";
	.return_x = 170;
	.return_y = 84;
	end;

OnPCLoginEvent:
OnPCLoadMapEvent:
	if (strcharinfo(3) != .map$) end;
	if (#valor_mvp_until <= gettimetick(2)) {
		dispbottom "[สนามรบ MVP Valor] สิทธิ์เข้าสนามรบของคุณหมดเวลาแล้ว";
		warp .return_map$,.return_x,.return_y;
		end;
	}
	callsub S_ScheduleExpiry;
	end;

OnExpire:
	if (strcharinfo(3) == .map$ && #valor_mvp_until <= gettimetick(2)) {
		dispbottom "[สนามรบ MVP Valor] สิทธิ์เข้าสนามรบของคุณหมดเวลาแล้ว";
		warp .return_map$,.return_x,.return_y;
	}
	end;

S_ScheduleExpiry:
	deltimer strnpcinfo(0)+"::OnExpire";
	.@seconds = #valor_mvp_until - gettimetick(2);
	if (.@seconds > 0)
		addtimer .@seconds * 1000, strnpcinfo(0)+"::OnExpire";
	return;
}

// The special Thanatos keeps its normal drops/card and creates Titan-style chests.
-	script	ValorMvpBattlefield	FAKE_NPC,{
	end;

OnInit:
	.map$ = "valor_pk";
	.chest_radius = 4;
	.chest_sprite = 1324;
	setarray .regular_mvp[0],
		1087,1147,1190,1115,1086,1038,1159,1389,1046,
		1059,1150,1039,1157,1112,1251,2068,1373,1272,
		1885,1630,1252,1623,1492,1583,1312,1785,1734,
		1685,1719,1511,1418,1768,1779,1688,1871;
	end;

// Valor spawns use this dedicated event so their kills never reach global
// OnNPCKillEvent handlers (quests and the MVP ranking).
OnValorMobKill:
	if (strcharinfo(3) != .map$) end;
	getmapxy .@map$,.@x,.@y,BL_PC;
	if (.@map$ != .map$) end;
	if (killedrid != 1708) {
		if (inarray(.regular_mvp,killedrid) < 0) end;
		callsub S_DropRegularRewards,.@map$,.@x,.@y;
		end;
	}

	.@purple = (rand(100) < 1) ? rand(1,3) : 0;
	.@red = (rand(100) < 10) ? rand(1,4) : 0;
	.@blue = rand(2,6);
	.@blue_cap = 10 - .@purple - .@red;
	if (.@blue > .@blue_cap) .@blue = .@blue_cap;
	callsub S_SpawnChests,"P",.@purple,.@x,.@y;
	callsub S_SpawnChests,"R",.@red,.@x,.@y;
	callsub S_SpawnChests,"B",.@blue,.@x,.@y;
	announce "[ สนามรบ MVP Valor ] กำจัด Memory of Thanatos สำเร็จ! หีบสมบัติ Titan ปรากฏแล้ว!",bc_map|bc_blue;
	end;

// Regular Valor MVPs only drop the six sheet-defined reward rolls. Rates use
// SC_ITEMBOOST's val1 so Bubble Gum modifies the drops just like normal loot.
S_DropRegularRewards:
	.@map$ = getarg(0);
	.@x = getarg(1);
	.@y = getarg(2);
	.@boost = 100 + getstatus(SC_ITEMBOOST,1);
	.@slice_rate = min(10000,2000 * .@boost / 100);
	.@apple_rate = min(10000,1000 * .@boost / 100);
	.@fragment_rate = min(10000,300 * .@boost / 100);
	if (rand(10000) < .@slice_rate) makeitem 1500181,1,.@map$,.@x,.@y;
	if (rand(10000) < .@slice_rate) makeitem 1500181,1,.@map$,.@x,.@y;
	if (rand(10000) < .@slice_rate) makeitem 1500181,1,.@map$,.@x,.@y;
	if (rand(10000) < .@apple_rate) makeitem 1500182,1,.@map$,.@x,.@y;
	if (rand(10000) < .@apple_rate) makeitem 1500182,1,.@map$,.@x,.@y;
	if (rand(10000) < .@fragment_rate) makeitem 1000373,1,.@map$,.@x,.@y;
	return;

S_SpawnChests:
	.@color$ = getarg(0);
	.@count = getarg(1);
	.@base_x = getarg(2);
	.@base_y = getarg(3);
	if (.@count <= 0) return;
	for (.@i = 0; .@i < .@count; .@i++) {
		$VMB_CHEST_UID++;
		if ($VMB_CHEST_UID > 100000 || $VMB_CHEST_UID < 1) $VMB_CHEST_UID = 1;
		.@id = $VMB_CHEST_UID;
		setd "$@vmbchest_"+.@id,0;
		.@x = .@base_x + rand(-.chest_radius,.chest_radius);
		.@y = .@base_y + rand(-.chest_radius,.chest_radius);
		.@name$ = "หีบสมบัติทานาทอส#"+.@color$+"_"+.@id;
		.@npc$ = duplicate("ValorMvpThanatosChest",.map$,.@x,.@y,.@name$,.chest_sprite);
		if (.@npc$ != "") donpcevent .@npc$+"::OnArm";
	}
	return;
}

-	script	ValorMvpThanatosChest	FAKE_NPC,{
	explode(.@data$,strnpcinfo(2),"_");
	.@color$ = .@data$[0];
	.@id = atoi(.@data$[1]);
	if (getd("$@vmbchest_"+.@id) || !playerattached()) end;

	.@seconds = 5;
	.@bar$ = "0x33B5FF";
	.@group = IG_CITY_TITAN_BLUE;
	if (.@color$ == "R") {
		.@seconds = 10;
		.@bar$ = "0xFF3333";
		.@group = IG_CITY_TITAN_RED;
	} else if (.@color$ == "P") {
		.@seconds = 15;
		.@bar$ = "0xCC33FF";
		.@group = IG_CITY_TITAN_PURPLE;
	}
	progressbar .@bar$,.@seconds;
	if (getd("$@vmbchest_"+.@id)) end;
	setd "$@vmbchest_"+.@id,1;
	stopnpctimer;
	getrandgroupitem .@group,0,1;
	disablenpc strnpcinfo(3);
	end;

OnArm:
	initnpctimer;
	end;

OnTimer1800000:
	stopnpctimer;
	disablenpc strnpcinfo(3);
	end;
}
