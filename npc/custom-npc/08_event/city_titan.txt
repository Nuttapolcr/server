//===== Custom NPC: City Titan Event ========================================
//= Source: Google Sheet "Titan" (gid=234041347)
//= Spawns in Death Match 2 (guild_vs3 49,49) at 08:00 / 12:00 / 16:00 / 20:00.
//= At 75%, 50%, and 25% HP, standing players within 25 cells die.
//===========================================================================

-	script	CityTitan	FAKE_NPC,{
	end;

OnInit:
	.map$ = "guild_vs3";
	.spawn_x = 49;
	.spawn_y = 49;
	.boss_id = 31100;
	.cast_radius = 25;
	.chest_radius = 4;
	.chest_sprite = 1324;
	.active = 0;
	.ending = 0;
	.boss_gid = 0;
	.boss_x = .spawn_x;
	.boss_y = .spawn_y;
	end;

OnClock0800:
OnClock1200:
OnClock1600:
OnClock2000:
	if (.active) end;
	.active = 1;
	.ending = 0;
	announce "[ City Titan ] Titan กำลังมุ่งหน้าสู่แมพ Death Match2 อีก 30 วินาที!",bc_all|bc_blue;
	sleep 10000;
	if (!.active) end;
	announce "[ City Titan ] Titan จะปรากฏในอีก 20 วินาที!",bc_all;
	sleep 10000;
	if (!.active) end;
	announce "[ City Titan ] Titan จะปรากฏในอีก 10 วินาที! เตรียมตัวให้พร้อม!",bc_all;
	sleep 10000;
	if (!.active) end;
	callsub S_SpawnBoss;
	end;

// GM test: @doevent CityTitan::OnGMStart
OnGMStart:
	if (.active) {
		announce "[ City Titan ] Event กำลังทำงานอยู่",bc_self;
		end;
	}
	.active = 1;
	.ending = 0;
	callsub S_SpawnBoss;
	end;

// GM test: @doevent CityTitan::OnGMStop
OnGMStop:
	.ending = 1;
	.active = 0;
	stopnpctimer;
	killmonster .map$,strnpcinfo(0)+"::OnTitanKilled";
	.boss_gid = 0;
	callsub S_ClearChests;
	.ending = 0;
	announce "[ City Titan ] GM ยุติ Event และล้างวัตถุทั้งหมดแล้ว",bc_all;
	end;

S_SpawnBoss:
	callsub S_ClearChests;
	.ending = 0;
	.phase75 = 0;
	.phase50 = 0;
	.phase25 = 0;
	.boss_x = .spawn_x;
	.boss_y = .spawn_y;
	.spawn_time = gettimetick(2);

	announce "[ City Titan ] CITY TITAN ปรากฏตัวแล้ว ณ แมพ Death Match2!",bc_all|bc_blue;
	announce "[ City Titan ] เมื่อ Titan ร่ายสกิล ทุกคนในบริเวณต้องนั่ง มิฉะนั้นจะเสียชีวิต!",bc_all;
	monster .map$,.spawn_x,.spawn_y,"[City Titan]",.boss_id,1,strnpcinfo(0)+"::OnTitanKilled";
	.boss_gid = $@mobid[0];
	if (!.boss_gid) {
		.active = 0;
		announce "[ City Titan ] ไม่สามารถสร้างบอสได้ กรุณาแจ้งทีมงาน",bc_all;
		return;
	}
	initnpctimer;
	return;

OnTimer1000:
	if (!.active) {
		stopnpctimer;
		end;
	}
	if (!.boss_gid || !unitexists(.boss_gid)) {
		.active = 0;
		.boss_gid = 0;
		stopnpctimer;
		end;
	}
	if (gettimetick(2)-.spawn_time >= 1800) {
		.ending = 1;
		.active = 0;
		stopnpctimer;
		killmonster .map$,strnpcinfo(0)+"::OnTitanKilled";
		.boss_gid = 0;
		.ending = 0;
		announce "[ City Titan ] Titan หลบหนีออกจากแมพ Death Match2 ไปแล้ว!",bc_all;
		end;
	}

	getunitdata .boss_gid,.@boss;
	.boss_x = .@boss[UMOB_X];
	.boss_y = .@boss[UMOB_Y];
	.@hp_rate = (.@boss[UMOB_HP]*100)/.@boss[UMOB_MAXHP];

	if (!.phase75 && .@hp_rate <= 75) {
		.phase75 = 1;
		callsub S_SitCast,75;
	}
	else if (!.phase50 && .@hp_rate <= 50) {
		.phase50 = 1;
		callsub S_SitCast,50;
	}
	else if (!.phase25 && .@hp_rate <= 25) {
		.phase25 = 1;
		callsub S_SitCast,25;
	}

	if (.active && .boss_gid && unitexists(.boss_gid))
		setnpctimer 0;
	else
		stopnpctimer;
	end;

// NV_FIRSTAID is harmless and supplies the visible five-second cast bar.
S_SitCast:
	.@phase = getarg(0);
	if (!.active || !unitexists(.boss_gid)) return;

	getunitdata .boss_gid,.@boss;
	.boss_x = .@boss[UMOB_X];
	.boss_y = .@boss[UMOB_Y];
	announce "[ City Titan ] คำเตือน! Titan ร่ายสกิลที่ HP "+.@phase+"% — นั่งภายใน 5 วินาที!",bc_all|bc_blue;
	unitskilluseid .boss_gid,"NV_FIRSTAID",1,.boss_gid,5,0;
	sleep 5000;

	if (!.active || !unitexists(.boss_gid)) return;
	getunitdata .boss_gid,.@boss;
	.boss_x = .@boss[UMOB_X];
	.boss_y = .@boss[UMOB_Y];
	.@count = getareaunits(BL_PC,.map$,
		.boss_x-.cast_radius,.boss_y-.cast_radius,
		.boss_x+.cast_radius,.boss_y+.cast_radius,.@players[0]);

	for (.@i = 0; .@i < .@count; .@i++) {
		if (!attachrid(.@players[.@i])) continue;
		if (getmapxy(.@map$,.@player_x,.@player_y,BL_PC) != 0) {
			detachrid;
			continue;
		}
		if (distance(.boss_x,.boss_y,.@player_x,.@player_y) <= .cast_radius && !readparam(Sitting))
			unitkill .@players[.@i];
		detachrid;
	}
	return;

OnTitanKilled:
	stopnpctimer;
	.@was_ending = .ending;
	.active = 0;
	.boss_gid = 0;
	if (.@was_ending || !playerattached()) end;

	.@mvp$ = strcharinfo(0);
	getitem 6635,1;
	getitem 1001012,1;
	if (rand(100) < 10) getitem 6635,10;
	if (rand(100) < 10) getitem 1001012,10;
	announce "[ City Titan ] "+.@mvp$+" ได้รับรางวัล MVP! กล่องสมบัติปรากฏขึ้นแล้ว!",bc_all|bc_blue;

	// Purple and Red roll first; Blue is reduced last to enforce the cap.
	.@purple = (rand(100) < 1) ? rand(1,3) : 0;
	.@red = (rand(100) < 10) ? rand(1,4) : 0;
	.@blue = rand(2,6);
	.@blue_cap = 10-.@purple-.@red;
	if (.@blue > .@blue_cap) .@blue = .@blue_cap;

	callsub S_SpawnChests,"P",.@purple;
	callsub S_SpawnChests,"R",.@red;
	callsub S_SpawnChests,"B",.@blue;
	end;

S_SpawnChests:
	.@color$ = getarg(0);
	.@count = getarg(1);
	if (.@count <= 0) return;

	if (.@color$ == "P")
		.@display$ = "กล่องสมบัติสีม่วง";
	else if (.@color$ == "R")
		.@display$ = "กล่องสมบัติสีแดง";
	else
		.@display$ = "กล่องสมบัติสีฟ้า";

	for (.@i = 0; .@i < .@count; .@i++) {
		$CTB_UID++;
		if ($CTB_UID > 100000 || $CTB_UID < 1) $CTB_UID = 1;
		.@id = $CTB_UID;
		setd "$@ctbo_"+.@id,0;

		.@x = .boss_x+rand(-.chest_radius,.chest_radius);
		.@y = .boss_y+rand(-.chest_radius,.chest_radius);
		.@name$ = .@display$+"#"+.@color$+"_"+.@id;
		.@npc$ = duplicate("CityTitanChest",.map$,.@x,.@y,.@name$,.chest_sprite);
		if (.@npc$ != "") {
			$@CTB_NPC$[getarraysize($@CTB_NPC$)] = .@npc$;
			donpcevent .@npc$+"::OnArm";
		}
	}
	return;

S_ClearChests:
	for (.@i = 0; .@i < getarraysize($@CTB_NPC$); .@i++) {
		if ($@CTB_NPC$[.@i] != "")
			disablenpc $@CTB_NPC$[.@i];
	}
	deletearray $@CTB_NPC$[0],getarraysize($@CTB_NPC$);
	return;
}

//=== Treasure chest template ==============================================
-	script	CityTitanChest	FAKE_NPC,{
	explode(.@data$,strnpcinfo(2),"_");
	.@color$ = .@data$[0];
	.@id = atoi(.@data$[1]);

	if (getd("$@ctbo_"+.@id)) {
		disablenpc strnpcinfo(3);
		end;
	}
	if (!playerattached()) end;

	.@seconds = 10;
	.@bar$ = "0x33B5FF";
	.@group = IG_CITY_TITAN_BLUE;
	if (.@color$ == "R") {
		.@seconds = 20;
		.@bar$ = "0xFF3333";
		.@group = IG_CITY_TITAN_RED;
	}
	else if (.@color$ == "P") {
		.@seconds = 30;
		.@bar$ = "0xCC33FF";
		.@group = IG_CITY_TITAN_PURPLE;
	}

	progressbar .@bar$,.@seconds;
	if (getd("$@ctbo_"+.@id)) end;
	setd "$@ctbo_"+.@id,1;
	stopnpctimer;

	.@reward = groupranditem(.@group,1);
	getitem .@reward,1;
	setarray .@rare[0],1001686,1001689,1001692;
	if (inarray(.@rare,.@reward) >= 0)
		announce "[ City Titan ] "+strcharinfo(0)+" ได้รับ "+getitemname(.@reward)+" จากกล่องสมบัติ!",bc_all;

	disablenpc strnpcinfo(3);
	end;

OnArm:
	initnpctimer;
	end;

OnTimer1800000:
	stopnpctimer;
	disablenpc strnpcinfo(3);
	end;
}
