//===== Custom NPC: City Thanatos Event =====================================
//= Source: Google Sheet MA item 22 (gid=1015278093)
//= Spawns in Death Match 2 (guild_vs3 49,49) at 18:00 / 20:00 / 22:00 / 00:00.
//= Storm Gust every 10s, Meteor every 12s, Coma at 50/25/5% HP,
//= and Earthquake at 35/10% HP. Treasure chest behavior is unchanged.
//===========================================================================

-	script	CityTitan	FAKE_NPC,{
	end;

OnInit:
	.map$ = "guild_vs3";
	.spawn_x = 49;
	.spawn_y = 49;
	.boss_id = 31100;
	.cast_radius = 25;
	.chest_radius = 4;
	.chest_sprite = 1324;
	.active = 0;
	.ending = 0;
	.boss_gid = 0;
	.boss_x = .spawn_x;
	.boss_y = .spawn_y;
	end;

OnClock1800:
OnClock2000:
OnClock2200:
OnClock0000:
	if (.active) end;
	.active = 1;
	.ending = 0;
	announce "[ City Thanatos ] Thanatos กำลังมุ่งหน้าสู่แมพ Death Match2 อีก 30 วินาที!",bc_all|bc_blue;
	sleep 10000;
	if (!.active) end;
	announce "[ City Thanatos ] Thanatos จะปรากฏในอีก 20 วินาที!",bc_all;
	sleep 10000;
	if (!.active) end;
	announce "[ City Thanatos ] Thanatos จะปรากฏในอีก 10 วินาที! เตรียมตัวให้พร้อม!",bc_all;
	sleep 10000;
	if (!.active) end;
	callsub S_SpawnBoss;
	end;

// GM test: @doevent CityTitan::OnGMStart
OnGMStart:
	if (.active) {
		announce "[ City Thanatos ] Event กำลังทำงานอยู่",bc_self;
		end;
	}
	.active = 1;
	.ending = 0;
	callsub S_SpawnBoss;
	end;

// GM test: @doevent CityTitan::OnGMStop
OnGMStop:
	.ending = 1;
	.active = 0;
	stopnpctimer;
	killmonster .map$,strnpcinfo(0)+"::OnThanatosKilled";
	.boss_gid = 0;
	callsub S_ClearChests;
	.ending = 0;
	announce "[ City Thanatos ] GM ยุติ Event และล้างวัตถุทั้งหมดแล้ว",bc_all;
	end;

S_SpawnBoss:
	callsub S_ClearChests;
	.ending = 0;
	.coma50 = 0;
	.coma25 = 0;
	.coma5 = 0;
	.quake35 = 0;
	.quake10 = 0;
	.boss_x = .spawn_x;
	.boss_y = .spawn_y;
	.spawn_time = gettimetick(2);
	.next_storm = .spawn_time + 10;
	.next_meteor = .spawn_time + 12;

	announce "[ City Thanatos ] MEMORY OF THANATOS ปรากฏตัวแล้ว ณ แมพ Death Match2!",bc_all|bc_blue;
	announce "[ City Thanatos ] ระวัง Storm Gust, Meteor, Coma และ Earthquake!",bc_all;
	monster .map$,.spawn_x,.spawn_y,"[City Thanatos]",.boss_id,1,strnpcinfo(0)+"::OnThanatosKilled";
	.boss_gid = $@mobid[0];
	if (!.boss_gid) {
		.active = 0;
		announce "[ City Thanatos ] ไม่สามารถสร้างบอสได้ กรุณาแจ้งทีมงาน",bc_all;
		return;
	}
	initnpctimer;
	return;

OnTimer1000:
	if (!.active) {
		stopnpctimer;
		end;
	}
	if (!.boss_gid || !unitexists(.boss_gid)) {
		.active = 0;
		.boss_gid = 0;
		stopnpctimer;
		end;
	}
	if (gettimetick(2)-.spawn_time >= 1800) {
		.ending = 1;
		.active = 0;
		stopnpctimer;
		killmonster .map$,strnpcinfo(0)+"::OnThanatosKilled";
		.boss_gid = 0;
		.ending = 0;
		announce "[ City Thanatos ] Thanatos หลบหนีออกจากแมพ Death Match2 ไปแล้ว!",bc_all;
		end;
	}

	getunitdata .boss_gid,.@boss;
	.boss_x = .@boss[UMOB_X];
	.boss_y = .@boss[UMOB_Y];
	.@hp_rate = .@boss[UMOB_HP] / max(1,.@boss[UMOB_MAXHP]/100);
	.@now = gettimetick(2);

	if (.@now >= .next_storm) {
		.next_storm = .@now + 10;
		callsub S_CastGroundSkill,"WZ_STORMGUST",10;
	}
	if (.@now >= .next_meteor) {
		.next_meteor = .@now + 12;
		callsub S_CastGroundSkill,"WZ_METEOR",10;
	}

	if (!.coma50 && .@hp_rate <= 50) {
		.coma50 = 1;
		callsub S_Coma,50;
	}
	if (!.quake35 && .@hp_rate <= 35) {
		.quake35 = 1;
		callsub S_Earthquake,35;
	}
	if (!.coma25 && .@hp_rate <= 25) {
		.coma25 = 1;
		callsub S_Coma,25;
	}
	if (!.quake10 && .@hp_rate <= 10) {
		.quake10 = 1;
		callsub S_Earthquake,10;
	}
	if (!.coma5 && .@hp_rate <= 5) {
		.coma5 = 1;
		callsub S_Coma,5;
	}

	if (.active && .boss_gid && unitexists(.boss_gid))
		setnpctimer 0;
	else
		stopnpctimer;
	end;

S_CastGroundSkill:
	.@skill$ = getarg(0);
	.@level = getarg(1);
	if (!.active || !unitexists(.boss_gid)) return;

	.@x = .boss_x;
	.@y = .boss_y;
	.@count = getmapunits(BL_PC,.map$,.@players[0]);
	if (.@count > 0) {
		.@target_gid = .@players[rand(.@count)];
		if (unitexists(.@target_gid)) {
			getunitdata .@target_gid,.@target;
			.@x = .@target[UMOB_X];
			.@y = .@target[UMOB_Y];
		}
	}
	unitskillusepos .boss_gid,.@skill$,.@level,.@x,.@y,0,0;
	return;

S_Coma:
	.@phase = getarg(0);
	if (!.active || !unitexists(.boss_gid)) return;

	getunitdata .boss_gid,.@boss;
	.boss_x = .@boss[UMOB_X];
	.boss_y = .@boss[UMOB_Y];
	announce "[ City Thanatos ] Thanatos ใช้ Coma ที่ HP "+.@phase+"%!",bc_all|bc_blue;
	.@count = getareaunits(BL_PC,.map$,
		.boss_x-.cast_radius,.boss_y-.cast_radius,
		.boss_x+.cast_radius,.boss_y+.cast_radius,.@players[0]);

	for (.@i = 0; .@i < .@count; .@i++) {
		if (!attachrid(.@players[.@i])) continue;
		if (readparam(Hp) > 0)
			sc_start SC_COMA,1000,1;
		detachrid;
	}
	return;

S_Earthquake:
	.@phase = getarg(0);
	if (!.active || !unitexists(.boss_gid)) return;
	announce "[ City Thanatos ] Thanatos ใช้ Earthquake ที่ HP "+.@phase+"%!",bc_all|bc_blue;
	unitskilluseid .boss_gid,"NPC_EARTHQUAKE",10,.boss_gid,2000,0;
	return;

OnThanatosKilled:
	stopnpctimer;
	.@was_ending = .ending;
	.active = 0;
	.boss_gid = 0;
	if (.@was_ending || !playerattached()) end;

	.@mvp$ = strcharinfo(0);
	callfunc "F_DWQ_RecordTitanOrThanatosKill";
	getitem 6635,1;
	getitem 1001012,1;
	if (rand(100) < 10) getitem 6635,10;
	if (rand(100) < 10) getitem 1001012,10;
	announce "[ City Thanatos ] "+.@mvp$+" ได้รับรางวัล MVP! กล่องสมบัติปรากฏขึ้นแล้ว!",bc_all|bc_blue;

	// Purple and Red roll first; Blue is reduced last to enforce the cap.
	.@purple = (rand(100) < 1) ? rand(1,3) : 0;
	.@red = (rand(100) < 10) ? rand(1,4) : 0;
	.@blue = rand(2,6);
	.@blue_cap = 10-.@purple-.@red;
	if (.@blue > .@blue_cap) .@blue = .@blue_cap;

	callsub S_SpawnChests,"P",.@purple;
	callsub S_SpawnChests,"R",.@red;
	callsub S_SpawnChests,"B",.@blue;
	end;

S_SpawnChests:
	.@color$ = getarg(0);
	.@count = getarg(1);
	if (.@count <= 0) return;

	if (.@color$ == "P")
		.@display$ = "กล่องสมบัติสีม่วง";
	else if (.@color$ == "R")
		.@display$ = "กล่องสมบัติสีแดง";
	else
		.@display$ = "กล่องสมบัติสีฟ้า";

	for (.@i = 0; .@i < .@count; .@i++) {
		$CTB_UID++;
		if ($CTB_UID > 100000 || $CTB_UID < 1) $CTB_UID = 1;
		.@id = $CTB_UID;
		setd "$@ctbo_"+.@id,0;

		.@x = .boss_x+rand(-.chest_radius,.chest_radius);
		.@y = .boss_y+rand(-.chest_radius,.chest_radius);
		.@name$ = .@display$+"#"+.@color$+"_"+.@id;
		.@npc$ = duplicate("CityTitanChest",.map$,.@x,.@y,.@name$,.chest_sprite);
		if (.@npc$ != "") {
			$@CTB_NPC$[getarraysize($@CTB_NPC$)] = .@npc$;
			donpcevent .@npc$+"::OnArm";
		}
	}
	return;

S_ClearChests:
	for (.@i = 0; .@i < getarraysize($@CTB_NPC$); .@i++) {
		if ($@CTB_NPC$[.@i] != "")
			disablenpc $@CTB_NPC$[.@i];
	}
	deletearray $@CTB_NPC$[0],getarraysize($@CTB_NPC$);
	return;
}

//=== Treasure chest template ==============================================
-	script	CityTitanChest	FAKE_NPC,{
	explode(.@data$,strnpcinfo(2),"_");
	.@color$ = .@data$[0];
	.@id = atoi(.@data$[1]);

	if (getd("$@ctbo_"+.@id)) {
		disablenpc strnpcinfo(3);
		end;
	}
	if (!playerattached()) end;
	if (#pk_protect_until > gettimetick(2)) {
		dispbottom "[City Thanatos] ผู้เล่นที่ใช้ PK Protection ไม่สามารถเปิดกล่องสมบัติ Thanatos ได้";
		end;
	}

	.@seconds = 5;
	.@bar$ = "0x33B5FF";
	.@group = IG_CITY_TITAN_BLUE;
	if (.@color$ == "R") {
		.@seconds = 10;
		.@bar$ = "0xFF3333";
		.@group = IG_CITY_TITAN_RED;
	}
	else if (.@color$ == "P") {
		.@seconds = 15;
		.@bar$ = "0xCC33FF";
		.@group = IG_CITY_TITAN_PURPLE;
	}

	progressbar .@bar$,.@seconds;
	if (getd("$@ctbo_"+.@id)) end;
	setd "$@ctbo_"+.@id,1;
	stopnpctimer;

	// Quantity 0 uses the Amount configured for the selected item group entry.
	getrandgroupitem .@group,0,1;

	disablenpc strnpcinfo(3);
	end;

OnArm:
	initnpctimer;
	end;

OnTimer1800000:
	stopnpctimer;
	disablenpc strnpcinfo(3);
	end;
}
