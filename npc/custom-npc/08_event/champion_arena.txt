//============================================================
//= GM-controlled Champion Arena (BO1)
//= Uses an instance clone of the Geffen Tournament stage.
//============================================================

-	script	ChampionArenaControl	-1,{

OnInit:
	bindatcmd "champcreate", strnpcinfo(3)+"::OnCreate", 99, 99;
	bindatcmd "champpull", strnpcinfo(3)+"::OnPull", 99, 99;
	bindatcmd "champstart", strnpcinfo(3)+"::OnStart", 99, 99;
	bindatcmd "champreset", strnpcinfo(3)+"::OnReset", 99, 99;
	bindatcmd "champclose", strnpcinfo(3)+"::OnClose", 99, 99;
	end;

OnCreate:
	if ($@ChampArena_Instance > 0) {
		dispbottom "[สนามประลองแชมเปียน] สนามประลองเปิดอยู่แล้ว กรุณาใช้ @champclose ก่อน";
		end;
	}
	.@instance_id = instance_create("Champion GM Arena", IM_CHAR, getcharid(0));
	if (.@instance_id < 0) {
		dispbottom "[สนามประลองแชมเปียน] ไม่สามารถสร้างอินสแตนซ์ได้ (ข้อผิดพลาด "+.@instance_id+")";
		end;
	}
	$@ChampArena_Instance = .@instance_id;
	$@ChampArena_Map$ = instance_mapname("champvs", .@instance_id);
	$@ChampArena_State = 1;
	$@ChampArena_Fighter[0] = 0;
	$@ChampArena_Fighter[1] = 0;
	$@ChampArena_Name$[0] = "";
	$@ChampArena_Name$[1] = "";
	setmapflag $@ChampArena_Map$, MF_PVP_NOPARTY;
	setmapflag $@ChampArena_Map$, MF_PVP_NOGUILD;
	setmapflag $@ChampArena_Map$, MF_PVP_NOCALCRANK;
	setmapflag $@ChampArena_Map$, MF_NOPENALTY;
	enablenpc instance_npcname("ร้านค้าสนามประลอง#champ", .@instance_id);
	warp $@ChampArena_Map$,104,62;
	setpcblock PCBLOCK_MOVE|PCBLOCK_ATTACK|PCBLOCK_SKILL|PCBLOCK_USEITEM, true;
	dispbottom "[สนามประลองแชมเปียน] เปิดสนามแล้ว ใช้ @champpull <Account ID> สองครั้ง แล้วใช้ @champstart";
	end;

OnPull:
	if ($@ChampArena_Instance <= 0 || $@ChampArena_State != 1) {
		dispbottom "[สนามประลองแชมเปียน] สนามประลองยังไม่เปิดรับผู้เข้าแข่งขัน";
		end;
	}
	if (.@atcmd_numparameters < 1) {
		dispbottom "วิธีใช้: @champpull <Account ID>";
		end;
	}
	.@target_aid = atoi(.@atcmd_parameters$[0]);
	if (.@target_aid <= 0 || !isloggedin(.@target_aid)) {
		dispbottom "[สนามประลองแชมเปียน] Account ID นี้ไม่ได้ออนไลน์อยู่: "+.@atcmd_parameters$[0];
		end;
	}
	if (.@target_aid == $@ChampArena_Fighter[0] || .@target_aid == $@ChampArena_Fighter[1]) {
		dispbottom "[สนามประลองแชมเปียน] ผู้เล่นคนนี้เป็นผู้เข้าแข่งขันอยู่แล้ว";
		end;
	}
	if ($@ChampArena_Fighter[0] && $@ChampArena_Fighter[1]) {
		dispbottom "[สนามประลองแชมเปียน] ช่องผู้เข้าแข่งขันเต็มทั้งสองช่องแล้ว";
		end;
	}
	.@gm_aid = getcharid(3);
	if (!attachrid(.@target_aid)) {
		attachrid(.@gm_aid);
		dispbottom "[สนามประลองแชมเปียน] ผู้เล่นออฟไลน์ไปแล้ว";
		end;
	}
	if (Class != Job_Champion) {
		attachrid(.@gm_aid);
		dispbottom "[สนามประลองแชมเปียน] อนุญาตเฉพาะอาชีพ Champion เท่านั้น";
		end;
	}
	getinventorylist;
	if (@inventorylist_count > 0) {
		atcommand "@storeall";
		getinventorylist;
		if (@inventorylist_count > 0) {
			dispbottom "[สนามประลองแชมเปียน] ไม่สามารถเข้าได้ กรุณาย้ายไอเทมที่เหลือทั้งหมดเข้าคลังก่อน";
			attachrid(.@gm_aid);
			dispbottom "[สนามประลองแชมเปียน] ยกเลิกการดึงตัว: ช่องเก็บของของผู้เล่นยังไม่ว่าง (คลังอาจเต็ม)";
			end;
		}
	}
	.@slot = ($@ChampArena_Fighter[0] == 0) ? 0 : 1;
	$@ChampArena_Fighter[.@slot] = getcharid(3);
	$@ChampArena_Name$[.@slot] = strcharinfo(0);
	percentheal 100,100;
	setpcblock PCBLOCK_MOVE|PCBLOCK_ATTACK|PCBLOCK_SKILL, true;
	if (.@slot == 0)
		warp $@ChampArena_Map$,99,50;
	else
		warp $@ChampArena_Map$,109,50;
	announce "[ สนามประลองแชมเปียน ] "+$@ChampArena_Name$[.@slot]+" เข้าประจำฝั่งที่ "+(.@slot+1)+"", bc_all|bc_blue;
	attachrid(.@gm_aid);
	dispbottom "[สนามประลองแชมเปียน] ดึงตัว "+$@ChampArena_Name$[.@slot]+" เข้าสู่สนามแล้ว";
	end;

OnStart:
	if ($@ChampArena_Instance <= 0 || $@ChampArena_State != 1) {
		dispbottom "[สนามประลองแชมเปียน] สนามประลองยังไม่พร้อมเริ่มการแข่งขัน";
		end;
	}
	if (!$@ChampArena_Fighter[0] || !$@ChampArena_Fighter[1]) {
		dispbottom "[สนามประลองแชมเปียน] ต้องมีผู้เข้าแข่งขันอาชีพ Champion สองคน";
		end;
	}
	if (!isloggedin($@ChampArena_Fighter[0]) || !isloggedin($@ChampArena_Fighter[1])) {
		dispbottom "[สนามประลองแชมเปียน] มีผู้เข้าแข่งขันออฟไลน์";
		end;
	}
	for (.@i = 0; .@i < 2; ++.@i) {
		if (getmapxy(.@fighter_map$,.@fighter_x,.@fighter_y,BL_PC,$@ChampArena_Name$[.@i]) != 0 || .@fighter_map$ != $@ChampArena_Map$) {
			dispbottom "[สนามประลองแชมเปียน] "+$@ChampArena_Name$[.@i]+" ไม่ได้อยู่ภายในสนาม";
			end;
		}
	}
	disablenpc instance_npcname("ร้านค้าสนามประลอง#champ", $@ChampArena_Instance);
	$@ChampArena_State = 2;
	mapannounce $@ChampArena_Map$, "[ สนามประลองแชมเปียน ] "+$@ChampArena_Name$[0]+" พบกับ "+$@ChampArena_Name$[1], bc_map|bc_blue;
	mapannounce $@ChampArena_Map$, "การแข่งขันจะเริ่มในอีก 5 วินาที", bc_map|bc_yellow;
	sleep 2000;
	mapannounce $@ChampArena_Map$, "3", bc_map|bc_yellow;
	sleep 1000;
	mapannounce $@ChampArena_Map$, "2", bc_map|bc_yellow;
	sleep 1000;
	mapannounce $@ChampArena_Map$, "1", bc_map|bc_yellow;
	sleep 1000;
	if ($@ChampArena_State != 2) end;
	setmapflag $@ChampArena_Map$, MF_PVP;
	for (.@i = 0; .@i < 2; ++.@i) {
		if (attachrid($@ChampArena_Fighter[.@i])) {
			percentheal 100,100;
			setpcblock PCBLOCK_MOVE|PCBLOCK_ATTACK|PCBLOCK_SKILL, false;
		}
	}
	$@ChampArena_State = 3;
	mapannounce $@ChampArena_Map$, "เริ่มต่อสู้!", bc_map|bc_red;
	detachrid;
	end;

OnPCDieEvent:
	if ($@ChampArena_State != 3 || strcharinfo(3) != $@ChampArena_Map$) end;
	.@loser = getcharid(3);
	if (.@loser != $@ChampArena_Fighter[0] && .@loser != $@ChampArena_Fighter[1]) end;
	.@winner_slot = (.@loser == $@ChampArena_Fighter[0]) ? 1 : 0;
	$@ChampArena_State = 4;
	removemapflag $@ChampArena_Map$, MF_PVP;
	announce "[ สนามประลองแชมเปียน ] ผู้ชนะคือ "+$@ChampArena_Name$[.@winner_slot]+"!", bc_all|bc_blue;
	mapannounce $@ChampArena_Map$, "ใช้ @champreset เพื่อเตรียมการแข่งขันรอบถัดไป", bc_map|bc_yellow;
	if (attachrid($@ChampArena_Fighter[.@winner_slot]))
		setpcblock PCBLOCK_MOVE|PCBLOCK_ATTACK|PCBLOCK_SKILL|PCBLOCK_USEITEM, true;
	end;

OnPCLogoutEvent:
	if ($@ChampArena_State != 3 || strcharinfo(3) != $@ChampArena_Map$) end;
	.@loser = getcharid(3);
	if (.@loser != $@ChampArena_Fighter[0] && .@loser != $@ChampArena_Fighter[1]) end;
	.@winner_slot = (.@loser == $@ChampArena_Fighter[0]) ? 1 : 0;
	$@ChampArena_State = 4;
	removemapflag $@ChampArena_Map$, MF_PVP;
	announce "[ สนามประลองแชมเปียน ] "+$@ChampArena_Name$[.@winner_slot]+" ชนะ เนื่องจากคู่ต่อสู้หลุดจากเกม!", bc_all|bc_blue;
	if (attachrid($@ChampArena_Fighter[.@winner_slot]))
		setpcblock PCBLOCK_MOVE|PCBLOCK_ATTACK|PCBLOCK_SKILL|PCBLOCK_USEITEM, true;
	end;

OnReset:
	if ($@ChampArena_Instance <= 0) {
		dispbottom "[สนามประลองแชมเปียน] สนามประลองยังไม่ได้เปิด";
		end;
	}
	.@gm_aid = getcharid(3);
	callsub S_ClearArena;
	$@ChampArena_State = 1;
	enablenpc instance_npcname("ร้านค้าสนามประลอง#champ", $@ChampArena_Instance);
	$@ChampArena_Fighter[0] = 0;
	$@ChampArena_Fighter[1] = 0;
	$@ChampArena_Name$[0] = "";
	$@ChampArena_Name$[1] = "";
	attachrid(.@gm_aid);
	dispbottom "[สนามประลองแชมเปียน] รีเซ็ตสนามแล้ว พร้อมสำหรับการแข่งขันรอบถัดไป";
	end;

OnClose:
	if ($@ChampArena_Instance <= 0) {
		dispbottom "[สนามประลองแชมเปียน] สนามประลองยังไม่ได้เปิด";
		end;
	}
	.@gm_aid = getcharid(3);
	.@instance_id = $@ChampArena_Instance;
	callsub S_ClearArena;
	$@ChampArena_Instance = 0;
	$@ChampArena_State = 0;
	$@ChampArena_Map$ = "";
	$@ChampArena_Fighter[0] = 0;
	$@ChampArena_Fighter[1] = 0;
	$@ChampArena_Name$[0] = "";
	$@ChampArena_Name$[1] = "";
	instance_destroy .@instance_id;
	attachrid(.@gm_aid);
	dispbottom "[สนามประลองแชมเปียน] ปิดสนามประลองแล้ว";
	end;

S_ClearArena:
	removemapflag $@ChampArena_Map$, MF_PVP;
	.@count = getmapunits(BL_PC, $@ChampArena_Map$, .@players[0]);
	for (.@i = 0; .@i < .@count; ++.@i) {
		if (!attachrid(.@players[.@i])) continue;
		setpcblock PCBLOCK_MOVE|PCBLOCK_ATTACK|PCBLOCK_SKILL|PCBLOCK_USEITEM, false;
		if (Hp <= 0) atcommand "@alive";
		warp "SavePoint",0,0;
	}
	detachrid;
	return;
}

// Spectators enter the active arena themselves instead of using @champspec.
morocc,149,104,4	script	ผู้ชมสนามประลองแชมเปียน	4_M_MANAGER,{
	mes "[ สนามประลองแชมเปียน ]";
	if ($@ChampArena_Instance <= 0 || $@ChampArena_Map$ == "") {
		mes "ขณะนี้สนามประลองยังไม่ได้เปิด";
		close;
	}
	if (getcharid(3) == $@ChampArena_Fighter[0] || getcharid(3) == $@ChampArena_Fighter[1]) {
		mes "ผู้เข้าแข่งขันไม่สามารถเข้าทางประตูสำหรับผู้ชมได้";
		close;
	}
	mes "ต้องการเข้าสู่พื้นที่สำหรับผู้ชมหรือไม่?";
	if (select("เข้าสู่สนามประลอง:ยกเลิก") == 2)
		close;
	close2;
	setpcblock PCBLOCK_MOVE|PCBLOCK_ATTACK|PCBLOCK_SKILL|PCBLOCK_USEITEM, true;
	warp $@ChampArena_Map$,104,62;
	dispbottom "[สนามประลองแชมเปียน] คุณเข้าสู่พื้นที่สำหรับผู้ชมแล้ว";
	end;

OnInit:
	waitingroom "เข้าชมการแข่งขัน Champ", 0;
	end;
}
