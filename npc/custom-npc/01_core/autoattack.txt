autobuy_menu:
	clear;
	.@menu$ = "[ซื้อไอเทมอัตโนมัติ]";
	mes .@menu$;
	mes "--- การตั้งค่าปัจจุบัน ---";
	mes "เลือกไอเทมที่ต้องการซื้ออัตโนมัติ";
	mes aa_getautoattackstring(15,0);
	.@autobuy_menu = select("เพิ่ม/แก้ไขไอเทม","ลบไอเทม","ล้างรายการ","[ยกเลิก]");
	switch(.@autobuy_menu){
		case 1:
			clear;
			mes .@menu$;
			mes "--- รายการไอเทมที่สามารถซื้อได้ ---";
			mes aa_getautoattackstring(15,-1);
			
			if(getarraysize(.@buy_add_menu_ids) == 0){
				clear;
				mes .@menu$;
				mes "ขออภัย ไม่มีไอเทมที่สามารถซื้อได้";
				next;
				goto(autobuy_menu);
			}
			
			.@buy_add_menu_list$ = .@buy_add_menu_list$ + ":[ยกเลิก]";
			.@i = select(.@buy_add_menu_list$);
			
			if(.@i > getarraysize(.@buy_add_menu_ids))
				goto(autobuy_menu);
			
			.@i--;
			.@selected_item = .@buy_add_menu_ids[.@i];
			
			do {
				clear;
				mes .@menu$;
				mes "ระบุจำนวนที่ต้องการซื้อ (1-1000)";
				input .@buy_quantity;
			} while(.@buy_quantity < 1 || .@buy_quantity > 1000);
			
			do {
				clear;
				mes .@menu$;
				mes "ซื้อเมื่อไอเทมเหลือน้อยกว่า (0-999)";
				input .@buy_trigger;
			} while(.@buy_trigger < 0 || .@buy_trigger > 999);
			
			aa_setautoattack("25;1;" + .@selected_item + ";" + .@buy_quantity + ";" + .@buy_trigger);
			break;
			
		case 2:
			clear;
			mes .@menu$;
			mes "--- เลือกไอเทมที่ต้องการลบ ---";
			mes aa_getautoattackstring(15,0);
			
			if (getarraysize(.@buy_menu_ids) == 0) {
				clear;
				mes .@menu$;
				mes "ไม่มีไอเทมในรายการซื้ออัตโนมัติ";
				next;
				goto(autobuy_menu);
			}
			
			// แสดง menu
			.@buy_menu_list$ = .@buy_menu_list$ + ":[ยกเลิก]";
			.@i = select(.@buy_menu_list$);
			
			if(.@i > getarraysize(.@buy_menu_ids))
				goto(autobuy_menu);
			
			.@i--;
			aa_setautoattack("25;0;" + .@buy_menu_ids[.@i] + ";0;0");
			break;
			
		case 3:
			aa_setautoattack("25;-1;0;0;0");
			break;
			
		case 4:
			goto(config_menu);
			break;
	}
	goto(autobuy_menu);
	end;

pickupitems_menu:
	clear;
	.@menu$ = "[เลือกไอเทมเก็บ]";
	mes .@menu$;
	mes "เลือกไอเทมที่ต้องการเก็บ";
	mes aa_getautoattackstring(13,0);
	.@pickupitems_menu = select("ตั้งค่าการเก็บไอเทม","เพิ่มไอเทม","ลบไอเทม","ล้างรายการ","[ยกเลิก]");
	switch(.@pickupitems_menu){
		case 1:
			.@pickup_config_choice = 0;
			// every item = 0
			// on the list = 1
			// no item = 2
			switch(aa_getautoattackint(9,0)){
				case 0:
					.@pickupitems_config_menu  = select("เก็บเฉพาะไอเทมในรายการ","ไม่เก็บไอเทมเลย","[ยกเลิก]");
					switch(.@pickupitems_config_menu){
						case 1:
							.@pickup_config_choice = 1;
							break;
						case 2 :
							.@pickup_config_choice = 2;
							break;
						case 3 :
							goto(pickupitems_menu);
							break;
					}
						
					break;
				case 1:
					.@pickupitems_config_menu  = select("เก็บทุกไอเทม","ไม่เก็บไอเทมเลย","[ยกเลิก]");
					switch(.@pickupitems_config_menu){
						case 1:
							.@pickup_config_choice = 0;
							break;
						case 2 :
							.@pickup_config_choice = 2;
							break;
						case 3 :
							goto(pickupitems_menu);
							break;
					}
					break;
				case 2:
					.@pickupitems_config_menu  = select("เก็บทุกไอเทม","เก็บเฉพาะไอเทมในรายการ","[ยกเลิก]");
					switch(.@pickupitems_config_menu){
						case 1:
							.@pickup_config_choice = 0;
							break;
						case 2 :
							.@pickup_config_choice = 1;
							break;
						case 3 :
							goto(pickupitems_menu);
							break;
					}
					break;				
			}
			aa_setautoattack("20;" + .@pickup_config_choice);
			break;
		case 2:
			clear;
			mes .@menu$;
			mes "ใส่ ID ของไอเทมที่ต้องการเพิ่มในรายการ (สูงสุด 20)";
			do {
				input .@itempickup_id;
			} while(.@itempickup_id < 1);
			aa_setautoattack("21;1;" + .@itempickup_id);
			break;
		case 3:
			if(aa_getautoattackint(17,0) > 0){
				clear;
				mes .@menu$;
				mes "เลือกไอเทมที่ต้องการลบจากรายการ";
				
				// ดึงรายการ item IDs และ names
				.@count = aa_getpickupitemlist();
				
				if(.@count > 0){
					// สร้างเมนูแบบสั้น (แค่ชื่อไอเทม)
					.@delete_menu$ = "";
					for(.@i = 0; .@i < .@count; .@i++){
						.@delete_menu$ += @pickup_item_names$[.@i];
						if(.@i < .@count - 1)
							.@delete_menu$ += ":";
					}
					.@delete_menu$ += ":[ยกเลิก]";
					
					// แสดง ID ก่อนให้เลือก
					clear;
					mes .@menu$;
					mes "รายการไอเทมที่สามารถลบได้:";
					for(.@i = 0; .@i < .@count; .@i++){
						mes "^FF0000" + (.@i + 1) + ".^000000 " + @pickup_item_names$[.@i] + " ^0000FF(ID: " + @pickup_item_ids[.@i] + ")^000000";
					}
					mes " ";
					
					.@selected = select(.@delete_menu$);
					
					// ถ้าเลือกไอเทม (ไม่ใช่ยกเลิก)
					if(.@selected <= .@count){
						.@itempickup_id = @pickup_item_ids[.@selected - 1];
						aa_setautoattack("21;0;" + .@itempickup_id);
					}
				}
			} else {
				clear;
				mes .@menu$;
				mes "ขออภัย คุณไม่มีไอเทมในรายการนี้";
				next;
			}
			break;
		case 4:
			aa_setautoattack("21;-1;0");
			break;
		case 5:
			goto(config_menu);
			break;
	}
	goto(pickupitems_menu);
	end;

	switch(.@teleport_menu){

			break;

			else

				clear;

			aa_setautoattack("17;"+.@teleport_delayvalue);
			break;
		case 7:
			do {
				clear;
				mes .@menu$;

			aa_setautoattack("23;"+.@monster_surround);
			break;
		case 8:
			goto(config_menu);
			break;

	end;

heal_menu:

	setarray .@lheal_skill[0],28,2051,2043;
	.@healskill_menu1$ = "";
	.@j = 0;
	
	// ตรวจสอบว่ามีสกิล
	for (.@i = 0; .@i < getarraysize(.@lheal_skill); .@i++){
		if(getskilllv(.@lheal_skill[.@i]))
			.@lheal_skill2[.@j++] = .@lheal_skill[.@i];
	}
	if(getarraysize(.@lheal_skill2) == 0){
		clear;
		mes .@menu$;
		mes "ขออภัย คุณไม่มีสกิลรักษาที่ได้รับอนุญาต";
		next;
		goto(regeneration_menu);
	}
	for (.@i = 0; .@i < getarraysize(.@lheal_skill2); .@i++)
		.@healskill_menu1$ = .@healskill_menu1$ + (.@i > 0 ? ":" : "") + aa_getautoattackstring(0,.@lheal_skill2[.@i],1);
	.@i = select(.@healskill_menu1$) -1;
	if(.@skill_menu_state[.@lheal_skill2[.@i]]){
		.@heal_menu = select("แก้ไข","ปิด","[ยกเลิก]");

			case 1 :
				clear;
				mes .@menu$;

				mes "";
				mes "การตั้งค่าปัจจุบัน:";

				for(.@k = 1; .@k <= .@max_lv; .@k++){

					if(.@k < .@max_lv)
						.@skill_lv_menu$ += ":";

				clear;
				mes .@menu$;
				mes "เลือกค่า HP % ที่จะใช้สกิล";
				.@hp_select = select("30%:50%:70%:90%");
				switch(.@hp_select){
					case 1: .@heal_minhpvalue = 30; break;
					case 2: .@heal_minhpvalue = 50; break;
					case 3: .@heal_minhpvalue = 70; break;
					case 4: .@heal_minhpvalue = 90; break;
				}
				
				aa_setautoattack("0;1;"+ .@lheal_skill2[.@i] + ";" + .@heal_skilllv + ";" + .@heal_minhpvalue);
				goto(heal_menu);
				break;
			case 2:
				aa_setautoattack("0;0;"+ .@lheal_skill[.@i] + ";0;0");
				goto(heal_menu);
				break;
			case 3:
				goto(heal_menu);
				break;
		}
	} else {
		.@heal_menu = select("เปิด","[ยกเลิก]");
		switch(.@heal_menu){
			case 1 :
				clear;
				mes .@menu$;
				mes "เลือกเลเวลสกิลที่จะใช้";

				break;
		}
	}
	else{
		.@skill_menu2 = select("เพิ่ม","[ยกเลิก]");
		switch(.@skill_menu2){
			case 1:
				clear;
				mes .@menu$;
				mes "เลือกเลเวลสกิลที่จะใช้";
				mes "เลเวลสูงสุด: " + getskilllv(.@skill_menu_ids[.@i]);
				mes "";
				mes "การตั้งค่าปัจจุบัน:";
				mes aa_getautoattackstring(3,.@skill_menu_ids[.@i],1);
				
				// สร้างเมนู Level ตามที่ผู้เล่นมี
				.@max_lv = getskilllv(.@skill_menu_ids[.@i]);
				.@skill_lv_menu$ = "";
				for(.@k = 1; .@k <= .@max_lv; .@k++){
					.@skill_lv_menu$ += "Level " + .@k;
					if(.@k < .@max_lv)
						.@skill_lv_menu$ += ":";
				}
				.@buff_skilllv = select(.@skill_lv_menu$);
				
				clear;
				mes .@menu$;
				mes "เลือกการตั้งค่าบัฟ";
				
				aa_setautoattack("3;1;"+ .@skill_menu_ids[.@i] + ";" + .@buff_skilllv);
				
				goto(buffskills_menu);
				break;
			case 2:
				goto(config_menu);
				break;
		}
	}
	goto(config_menu);
	end;

buffitems_menu:
	clear;
	.@menu$ = "[ไอเทมบัฟ]";

	mes .@menu$;
	mes "--- การตั้งค่าปัจจุบัน ---";
	setarray .@lbuff_item[0],645,656,657,12208,12210;
	setarray .@lbuff_sc[0],SC_ASPDPOTION0,SC_ASPDPOTION1,SC_ASPDPOTION2,SC_EXPBOOST,SC_ITEMBOOST;
	.@buffitems_menu1$ = "";
	
	for (.@i = 0; .@i < getarraysize(.@lbuff_item); .@i++)
		.@buffitems_menu1$ = .@buffitems_menu1$ + (.@i > 0 ? ":" : "") + aa_getautoattackstring(5,.@lbuff_item[.@i]);


	.@buffitems_menu1$ = "[ยกเลิก]:" + .@buffitems_menu1$;
	.@i = select(.@buffitems_menu1$) -1;

	if(.@i == 0)
		goto(config_menu);
	else
		.@i--;
	
	.@buffitems_menu = select(.@buffitem_menu_state[.@lbuff_item[.@i]] ? "ปิด" : "เปิด","[ยกเลิก]");
	if( .@buffitems_menu == 1 ){
		if(.@buffitem_menu_state[.@lbuff_item[.@i]])
			aa_setautoattack("5;0;"+ .@lbuff_item[.@i] + ";" + .@lbuff_sc[.@i]);
		else
			aa_setautoattack("5;1;"+ .@lbuff_item[.@i] + ";" + .@lbuff_sc[.@i]);
		goto(buffitems_menu);
	}
	else if( .@buffitems_menu == 2 ){
		aa_setautoattack("5;0;"+ .@lbuff_item[.@i] + ";0");
		goto(buffitems_menu);
	}
	
	goto(config_menu);
	end;

sit_regen_menu:
	clear;
	.@menu$ = "[นั่งพัก]";

	mes .@menu$;
	mes "--- การตั้งค่าปัจจุบัน ---";
	mes aa_getautoattackstring(4,0);

	.@sit_regen_menu = select(
	aa_getautoattackint(14,0) == 1?"^008000(เปิด)^000000 นั่งถ้า HP < 75%":"^FF0000(ปิด)^000000 นั่งถ้า HP < 75%"
	,aa_getautoattackint(14,0) == 2?"^008000(เปิด)^000000 นั่งถ้า HP < 50%":"^FF0000(ปิด)^000000 นั่งถ้า HP < 50%"
	,aa_getautoattackint(14,0) == 3?"^008000(เปิด)^000000 นั่งถ้า HP < 25%":"^FF0000(ปิด)^000000 นั่งถ้า HP < 25%"
	,aa_getautoattackint(15,0) == 1?"^008000(เปิด)^000000 นั่งถ้า SP < 75%":"^FF0000(ปิด)^000000 นั่งถ้า SP < 75%"
	,aa_getautoattackint(15,0) == 2?"^008000(เปิด)^000000 นั่งถ้า SP < 50%":"^FF0000(ปิด)^000000 นั่งถ้า SP < 50%"
	,aa_getautoattackint(15,0) == 3?"^008000(เปิด)^000000 นั่งถ้า SP < 25%":"^FF0000(ปิด)^000000 นั่งถ้า SP < 25%"
	,"[ยกเลิก]"
	);

	switch(.@sit_regen_menu){
		case 1:
			if(aa_getautoattackint(14,0) == 1)
				aa_setautoattack("14;0;0");
			else
				aa_setautoattack("14;75;100");
			break;
		case 2:
			if(aa_getautoattackint(14,0) == 2)
				aa_setautoattack("14;0;0");
			else
				aa_setautoattack("14;50;100");
			break;
		case 3:
			if(aa_getautoattackint(14,0) == 3)
				aa_setautoattack("14;0;0");
			else
				aa_setautoattack("14;25;100");
			break;
		case 4:
			if(aa_getautoattackint(15,0) == 1)
				aa_setautoattack("15;0;0");
			else
				aa_setautoattack("15;75;100");
			break;
		case 5:
			if(aa_getautoattackint(15,0) == 2)
				aa_setautoattack("15;0;0");
			else
				aa_setautoattack("15;50;100");
			break;
		case 6:
			if(aa_getautoattackint(15,0) == 3)
				aa_setautoattack("15;0;0");
			else
				aa_setautoattack("15;25;100");
			break;
		case 7:
			clear;
			goto(regeneration_menu);
			break;
	}

	clear;
	goto(sit_regen_menu);
	end;
