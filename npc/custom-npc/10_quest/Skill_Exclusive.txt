function	script	F_UB_GiveIfUnlocked	{
	.@var$ = getarg(0);
	.@skill$ = getarg(1);

	.@lv = getd(.@var$);
	if (.@lv <= 0) end;

	if (.@lv > 3) .@lv = 3;
	skill .@skill$, .@lv, SKILL_PERM;
	end;
}

S_UpgradeOne:

	.@skill$ = getarg(1);

	.@cur = getd(.@var$);

	if (.@cur <= 0) { mes .mes$; mes "คุณยังไม่ได้ปลดล็อกสกิลนี้"; return; }
	if (.@cur >= 3) { mes .mes$; mes "สกิลนี้เต็มเลเวลแล้ว"; return; }

	.@need = .ScrollNeed1;
	if (.@cur == 2) .@need = .ScrollNeed2;

	mes .mes$;
	mes "ต้องใช้:";
	mes "^777777- ^i["+ .ScrollID +"] "+getitemname(.ScrollID)+" x"+.@need+"^000000";
	mes "^777777- ^i["+ .SoulID +"] "+getitemname(.SoulID)+" x"+.SoulNeed+"^000000";
	mes "คุณมี Scroll: ^FF9900"+countitem(.ScrollID)+"^000000";
	mes "คุณมี Soul: ^FF9900"+countitem(.SoulID)+"^000000";
	next;

	if (countitem(.ScrollID) < .@need) { mes .mes$; mes "คุณมี Unlimit Break Skill Scroll ไม่พอ"; return; }
	if (countitem(.SoulID) < .SoulNeed) { mes .mes$; mes "คุณมี Soul of Emperium ไม่พอ"; return; }

	if (select("ยืนยันอัปเลเวล","ยกเลิก") == 2) return;

	delitem .ScrollID, .@need;
	delitem .SoulID, .SoulNeed;

	.@new = .@cur + 1;
	if (.@new > 3) .@new = 3;

	mes "Lv."+.@cur+" -> Lv."+.@new;
	mes "^777777ระบบจะคืนสกิลให้ทุกครั้งที่ Login^000000";
