function	script	F_DWQ_RecordCityTitanKill	{
	.@dow = gettime(DT_DAYOFWEEK);
	.@offset = (.@dow == SUNDAY) ? 6 : .@dow - MONDAY;
	if (.@dow == MONDAY && gettime(DT_HOUR) == 0 && gettime(DT_MINUTE) == 0)
		.@offset = 7;
	.@week = gettime(DT_YEAR) * 1000 + gettime(DT_DAYOFYEAR) - .@offset;

	if (!getd("#dwq_71031_active")) return;
	if (getd("#dwq_71031_key") != .@week) return;
	if (getd("#dwq_71031_k2") >= 1) return;
	setd "#dwq_71031_k2", 1;
	dispbottom "[Daily-Weekly Quest] เควสต์รายสัปดาห์ Hardcore : 1 / 1";
	return;
}

OnNPCKillEvent:
	.@today = gettime(DT_YYYYMMDD);
	.@dow = gettime(DT_DAYOFWEEK);
	.@offset = (.@dow == SUNDAY) ? 6 : .@dow - MONDAY;
	if (.@dow == MONDAY && gettime(DT_HOUR) == 0 && gettime(DT_MINUTE) == 0)
		.@offset = 7;
	.@week = gettime(DT_YEAR) * 1000 + gettime(DT_DAYOFYEAR) - .@offset;

	.@mvp = getmonsterinfo(killedrid, MOB_MVPEXP) > 0;
	.@lv = getmonsterinfo(killedrid, MOB_LV);

	if (.@mvp) {
		callsub S_AddKill, 71020, 1, 1, .@today;
		callsub S_AddKill, 71021, 2, 1, .@today;
		callsub S_AddKill, 71022, 2, 3, .@today;
		callsub S_AddKill, 71023, 2, 3, .@today;
		callsub S_AddKill, 71024, 2, 5, .@today;
		callsub S_AddKill, 71025, 1, 20, .@today;
		callsub S_AddKill, 71030, 1, 100, .@week;
		callsub S_AddKill, 71031, 1, 200, .@week;
	}

	if (!.@mvp && .@lv > 0 && .@lv < 50) {
		callsub S_AddKill, 71021, 1, 1000, .@today;
		callsub S_AddKill, 71022, 1, 3000, .@today;
	}

	if (!.@mvp && .@lv > 70) {
		callsub S_AddKill, 71023, 1, 1000, .@today;
		callsub S_AddKill, 71024, 1, 3000, .@today;
	}

	if (killedrid == 1833) {
		callsub S_AddKill, 71026, 1, 1000, .@week;
		callsub S_AddKill, 71027, 1, 2000, .@week;
	}
	if (killedrid == 1831) {
		callsub S_AddKill, 71026, 2, 1000, .@week;
		callsub S_AddKill, 71027, 2, 2000, .@week;
	}
	if (killedrid == 1638) { callsub S_AddKill, 71028, 1, 500, .@week; callsub S_AddKill, 71029, 1, 1000, .@week; }
	if (killedrid == 1635) { callsub S_AddKill, 71028, 2, 500, .@week; callsub S_AddKill, 71029, 2, 1000, .@week; }
	if (killedrid == 1637) { callsub S_AddKill, 71028, 3, 500, .@week; callsub S_AddKill, 71029, 3, 1000, .@week; }
	if (killedrid == 1639) { callsub S_AddKill, 71028, 4, 500, .@week; callsub S_AddKill, 71029, 4, 1000, .@week; }
	if (killedrid == 1634) { callsub S_AddKill, 71028, 5, 500, .@week; callsub S_AddKill, 71029, 5, 1000, .@week; }
	if (killedrid == 1636) { callsub S_AddKill, 71028, 6, 500, .@week; callsub S_AddKill, 71029, 6, 1000, .@week; }
	if (killedrid == 31100) callsub S_AddKill, 71031, 2, 1, .@week;
	end;

S_AddKill:
	.@qid = getarg(0);
	.@slot = getarg(1);
	.@target = getarg(2);
	.@key = getarg(3);
	if (!getd("#dwq_" + .@qid + "_active")) return;
	if (getd("#dwq_" + .@qid + "_key") != .@key) return;
	.@var$ = "#dwq_" + .@qid + "_k" + .@slot;
	.@cur = getd(.@var$);
	if (.@cur >= .@target) return;
	.@cur++;
	setd .@var$, .@cur;
	.@title$ = getd("$@dwq_title_" + .@qid + "$");
	if (.@title$ == "") .@title$ = "Quest " + .@qid;
	dispbottom "[Daily-Weekly Quest] " + .@title$ + " : " + .@cur + " / " + .@target;
	return;
}
