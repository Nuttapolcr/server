-	script	IG_DailyQuest_WoECard	-1,{

	bindatcmd "resetdailyhunt", strnpcinfo(3) + "::OnResetDailyHunt", 99, 99;

	#dh_daily_date = 0;

	if (checkquest(99058) >= 0) erasequest 99058;
	if (checkquest(99059) >= 0) erasequest 99059;
	if (checkquest(99060) >= 0) erasequest 99060;
	if (checkquest(99061) >= 0) erasequest 99061;
	if (checkquest(99062) >= 0) erasequest 99062;
	if (checkquest(99063) >= 0) erasequest 99063;
	if (checkquest(99064) >= 0) erasequest 99064;
	if (checkquest(99055) >= 0) erasequest 99055;
	if (checkquest(99056) >= 0) erasequest 99056;
	dispbottom "[Daily Hunt] รีเซ็ตข้อมูล/เควสเรียบร้อยแล้ว";
	end;
}
