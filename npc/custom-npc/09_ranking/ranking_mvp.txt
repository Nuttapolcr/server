morocc,196,91,4	script	MvP Ranking	11137,{

	mes .Name$;
	mes "^339933รายการอันดับของผู้เล่น ทั้ง Server^000000";
	mes "^FF0000แข่งจัดอันดับ 1-10 ทุกสัปดาห์^000000";
	mes "รับ ^FF66FFGift voucher^000000 จาก GM";
	mes "สามารถนำแต้มมาแลกของรางวัลได้";
	mes "^0000FFหมายเหตุ^000000 : แต้มจะไม่ลดลงจากอันดับ";
	mes "จะลดลงเพียงแต้มแลกของเท่านั้น";
	next;
	switch(select("~ ^FF0000จัดอันดับนักล่า MvP^000000:~ ^0000FFMvP Shop^000000:~ ยกเลิก"))
	{
		case 1:
			mes .Name$;
			query_sql( "SELECT `name`, `mvprank` FROM `ranking` ORDER BY `mvprank` DESC LIMIT "+ .query_limit +"", .@mvpname$, .@mvppoint);
			for ( .@i = 0; .@i < getarraysize(.@mvpname$); .@i++ )
				mes ""+ (.@i+1) +"." + ((.@mvpname$[.@i] == "") ? "ไม่มีการบันทึกอันดับล่าสุด" : "^ff0000"+ .@mvpname$[.@i]+"^000000 : ^ff0000" + .@mvppoint[.@i] + "^000000 แต้ม") + "";
			close;
			
		case 2:
			query_sql( "SELECT `mvprank_t` FROM `ranking` WHERE `char_id` = '"+ getcharid(0) +"'", .@mvppointshop);
			mes .Name$;
			mes "คุณมี MvP Kill ^0055FF"+ .@mvppointshop +"^000000 Point";
			callshop "mvp_shop",1;
			npcshopattach "mvp_shop";
			end;
			
		default: close;
	}
	end;

OnBuyItem:
	@MvPCost = 0;
	for (.@i = 0; .@i < getarraysize(@bought_nameid); .@i++)
		for (.@j = 0; .@j < getarraysize(.MvPShop); .@j += 2)
			if (@bought_nameid[.@i] == .MvPShop[.@j]) {
				set @MvPCost, @MvPCost + (.MvPShop[.@j+1] * @bought_quantity[.@i]);
				break;
			}
	mes .Name$;
	query_sql( "SELECT `mvprank_t` FROM `ranking` WHERE `char_id` = '"+ getcharid(0) +"'", .@mvppointshop);
	if (@MvPCost > .@mvppointshop)
		mes "คุณมีจำนวน MvP Point ไม่เพียงพอ.";
	else {
		for (.@i = 0; .@i < getarraysize(@bought_nameid); .@i++) {
			getitem @bought_nameid[.@i], @bought_quantity[.@i];
			dispbottom "แลกเปลี่ยน " + @bought_quantity[.@i] + " " + getitemname(@bought_nameid[.@i]) + ".";
		}
		query_sql( "UPDATE `ranking` SET `mvprank_t` = `mvprank_t`-'"+@MvPCost+"' WHERE `char_id` = '"+ getcharid(0) +"';" );
		.@mvppointshop -= @MvPCost;
		mes "การแลกเปลี่ยนเสร็จสมบูรณ์.";
		emotion ET_MONEY;
	}
	@MvPCost = 0;
	deletearray @bought_nameid[0], getarraysize(@bought_nameid);
	deletearray @bought_quantity[0], getarraysize(@bought_quantity);
	close;
	
OnInit:
	set .Name$,"[ อันดับนักล่า MvP ]";
	// ตั้งค่าแสดงอันดับผู้เล่นสูงสุด 10รายการ
	set .query_limit,10;
	// ตั้งค่าไอเท็ม เช่น รหัสไอเท็ม,จำนวนไอเท็ม,รหัสไอเท็ม,จำนวนไอเท็ม;
	setarray .MvPShop[0],30001,10,1000322,1;
					 
	query_sql "CREATE TABLE IF NOT EXISTS `ranking` (  `char_id` int(11) NOT NULL default '0',  `name` varchar(24) NOT NULL DEFAULT 'NULL',  `pvprank` mediumint(6) NOT NULL default '0', `pvprank_t` mediumint(6) NOT NULL default '0',	`mvprank` mediumint(6) NOT NULL default '0', `mvprank_t` mediumint(6) NOT NULL default '0',	PRIMARY KEY  (`char_id`)) ENGINE=MyISAM";
	
	npcshopdelitem "mvp_shop",512;
	for (.@i = 0; .@i < getarraysize(.MvPShop); .@i += 2)
		npcshopadditem "mvp_shop", .MvPShop[.@i], .MvPShop[.@i+1];
	end;	
}
-	shop	mvp_shop	-1,512:-1

-	script	mvp_count	FAKE_NPC,{

OnNPCKillEvent:
	if ( getmonsterinfo(killedrid,22) )
	{
		if ( query_sql( "SELECT `mvprank`,`mvprank_t` FROM `ranking` WHERE `char_id` = '"+ getcharid(0) +"';", @mvpcount , @mvpcount_t ) )
			query_sql( "UPDATE `ranking` SET `mvprank` = `mvprank`+'1',`mvprank_t` = `mvprank_t`+'1' WHERE `char_id` = '"+ getcharid(0) +"';" );
		else
			query_sql( "INSERT INTO `ranking` ( `char_id`, `name`, `mvprank` , `mvprank_t` ) VALUES ( "+ getcharid(0) +", '"+ strcharinfo(0) +"', '1' , '1' );" );
			dispbottom "คุณได้รับ MvP Kill จำนวน : [ "+(@mvpcount+1)+" ] Point";
			end;
	}
	end;
}
