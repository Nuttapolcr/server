morocc,173,100,3	script	PvP Ranking	847,{

	mes .Name$;
	mes "^339933รายการอันดับของผู้เล่น ทั้ง Server^000000";
	mes "^FF0000แข่งจัดอันดับ 1-10 ทุกสัปดาห์^000000";
	mes "รับ ^FF66FFGift voucher^000000 จาก GM";
	mes "สามารถนำแต้มมาแลกของรางวัลได้";
	mes "^0000FFหมายเหตุ^000000 : แต้มจะไม่ลดลงจากอันดับ";
	mes "จะลดลงเพียงแต้มแลกของเท่านั้น";
	next;
	switch(select("~ ^FF0000จัดอันดับนักฆ่า PvP^000000:~ ^0000FFPvP Shop^000000:~ ยกเลิก"))
	{
		case 1:
			mes .Name$;
			query_sql( "SELECT `name`, `pvprank` FROM `ranking` ORDER BY `pvprank` DESC LIMIT "+ .query_limit +"", .@pvpname$, .@pvppoint);
			for ( .@j = 0; .@j < getarraysize(.@pvpname$); .@j++ )
				mes ""+ (.@j+1) +"." + ((.@pvpname$[.@i] == "") ? "ไม่มีการบันทึกอันดับล่าสุด" : "^ff0000"+ .@pvpname$[.@j]+"^000000 : ^ff0000" + .@pvppoint[.@j] + "^000000 แต้ม") + "";
			close;
			
		case 2:
			query_sql( "SELECT `pvprank_t` FROM `ranking` WHERE `char_id` = '"+ getcharid(0) +"'", .@pvppointshop);
			mes .Name$;
			mes "คุณมี PvP Kill ^0055FF"+ .@pvppointshop +"^000000 Point";
			callshop "pvp_shop",1;
			npcshopattach "pvp_shop";
			end;
			
		default: close;
	}
	end;

OnBuyItem:
	@PvPCost = 0;
	for (.@i = 0; .@i < getarraysize(@bought_nameid); .@i++)
		for (.@j = 0; .@j < getarraysize(.PvPShop); .@j += 2)
			if (@bought_nameid[.@i] == .PvPShop[.@j]) {
				set @PvPCost, @PvPCost + (.PvPShop[.@j+1] * @bought_quantity[.@i]);
				break;
			}
	mes .Name$;
	query_sql( "SELECT `pvprank_t` FROM `ranking` WHERE `char_id` = '"+ getcharid(0) +"'", .@pvppointshop);
	if (@PvPCost > .@pvppointshop)
		mes "คุณมีจำนวน PvP Point ไม่เพียงพอ.";
	else {
		for (.@i = 0; .@i < getarraysize(@bought_nameid); .@i++) {
			getitem @bought_nameid[.@i], @bought_quantity[.@i];
			dispbottom "แลกเปลี่ยน " + @bought_quantity[.@i] + " " + getitemname(@bought_nameid[.@i]) + ".";
		}
		query_sql( "UPDATE `ranking` SET `pvprank_t` = `pvprank_t`-'"+@PvPCost+"' WHERE `char_id` = '"+ getcharid(0) +"';" );
		.@pvppointshop -= @PvPCost;
		mes "การแลกเปลี่ยนเสร็จสมบูรณ์.";
		emotion ET_MONEY;
	}
	@PvPCost = 0;
	deletearray @bought_nameid[0], getarraysize(@bought_nameid);
	deletearray @bought_quantity[0], getarraysize(@bought_quantity);
	close;
	
OnInit:
	set .Name$,"[ อันดับนักฆ่า PvP ]";
	// ตั้งค่าแสดงอันดับผู้เล่นสูงสุด 10รายการ
	set .query_limit,10;
	// ตั้งค่าไอเท็ม เช่น รหัสไอเท็ม,จำนวนไอเท็ม,รหัสไอเท็ม,จำนวนไอเท็ม;
	setarray .PvPShop[0],31459,700;
					 
	query_sql "CREATE TABLE IF NOT EXISTS `ranking` (  `char_id` int(11) NOT NULL default '0',  `name` varchar(24) NOT NULL DEFAULT 'NULL',  `pvprank` mediumint(6) NOT NULL default '0', `pvprank_t` mediumint(6) NOT NULL default '0',	`mvprank` mediumint(6) NOT NULL default '0', `mvprank_t` mediumint(6) NOT NULL default '0',	PRIMARY KEY  (`char_id`)) ENGINE=MyISAM";
	
	npcshopdelitem "pvp_shop",512;
	for (.@i = 0; .@i < getarraysize(.PvPShop); .@i += 2)
		npcshopadditem "pvp_shop", .PvPShop[.@i], .PvPShop[.@i+1];
	end;
}
-	shop	pvp_shop	-1,512:-1

-	script	pvp_count	FAKE_NPC,{

OnPCKillEvent:
	if( strcharinfo(3) != "pvp_y_7-2" && strcharinfo(3) != "pvp_y_8-2" && strcharinfo(3) != "pvp_y_8-1" && strcharinfo(3) != "pvp_n_1-3" && strcharinfo(3) != "pvp_n_1-5" && strcharinfo(3) != "map" || killedrid == killchar )
	killchar = killedrid;
		if ( query_sql( "SELECT `pvprank`,`pvprank_t` FROM `ranking` WHERE `char_id` = '"+ getcharid(0) +"';", @pvpcount ) )
			query_sql( "UPDATE `ranking` SET `pvprank` = `pvprank`+'1',`pvprank_t` = `pvprank_t`+'1' WHERE `char_id` = '"+ getcharid(0) +"';" );
		else
			query_sql( "INSERT INTO `ranking` ( `char_id`, `name`, `pvprank`, `pvprank_t` ) VALUES ( "+ getcharid(0) +", '"+ strcharinfo(0) +"', '1' , '1' );" );
			dispbottom "PvP Kill จำนวน : [ "+(@pvpcount+1)+" ] Point";
			end;
}
