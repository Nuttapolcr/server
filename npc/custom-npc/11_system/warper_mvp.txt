// Boss warp NPCs generated from the HICLASS boss-warp sheet.
// Destination coordinates use 0,0 so rAthena picks a random walkable cell.
// Each NPC passes its mob ID so F_BossWarp can list the boss's drop rates.

function	script	F_BossWarp	{
	.@mob_id = getarg(0);
	.@boss$ = getarg(1);
	.@map_count = getargcount() - 2;

	mes "[Boss Warp]";
	mes "Boss: ^FF0000" + .@boss$ + "^000000";

	// List the boss's normal drops with rates (1 = 0.01%).
	if (getmobdrops(.@mob_id) && $@MobDrop_count > 0) {
		.@count = $@MobDrop_count;
		copyarray .@d_item[0], $@MobDrop_item[0], .@count;
		copyarray .@d_rate[0], $@MobDrop_rate[0], .@count;
		mes " ";
		mes "^0000FFMvpDrops:^000000";
		for (.@i = 0; .@i < .@count; ++.@i) {
			.@whole = .@d_rate[.@i] / 100;
			.@frac  = .@d_rate[.@i] % 100;
			.@rate$ = "" + .@whole;
			if (.@frac > 0) {
				.@rate$ += "." + (.@frac / 10);
				if (.@frac % 10 > 0)
					.@rate$ += (.@frac % 10);
			}
			mes "^i[" + .@d_item[.@i] + "] ^0055FF" + getitemname(.@d_item[.@i]) + "^000000 = ^00CC00" + .@rate$ + "%^000000";
		}
	}
	next;

	.@menu$ = "";
	for (.@i = 1; .@i <= .@map_count; ++.@i)
		.@menu$ += getarg(.@i + 1) + ":";
	.@menu$ += "Cancel";

	.@choice = select(.@menu$);
	if (.@choice > .@map_count)
		close;

	.@ticket = 1000602;
	.@ticket_nt = 1000603;
	if (countitem(.@ticket_nt) > 0)
		delitem .@ticket_nt, 1;
	else if (countitem(.@ticket) > 0)
		delitem .@ticket, 1;
	else {
		mes "[Boss Warp]";
		mes "Need ^FF0000" + getitemname(.@ticket_nt) + "^000000 or ^FF0000" + getitemname(.@ticket) + "^000000.";
		close;
	}

	warp getarg(.@choice + 1), 0, 0;
	end;
}
