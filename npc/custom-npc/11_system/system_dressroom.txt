function	script	F_HiclassStylistPreview	{
	.@look_type = getarg(0);
	.@min_value = getarg(1);
	.@max_value = getarg(2);
	.@type_name$ = getarg(3);
	.@coupon_id = 1030061;
	.@old_value = getlook(.@look_type);
	.@preview_value = .@old_value;

	if (.@preview_value < .@min_value || .@preview_value > .@max_value)
		.@preview_value = .@min_value;

	while (1) {
		clear;
		changelook .@look_type,.@preview_value;
		mes "^FF0000[ Stylist ]^000000";
		mes "กำลังแสดง"+.@type_name$+"หมายเลข ^0055FF"+.@preview_value+"^000000";
		mes "ช่วงหมายเลขที่เลือกได้: "+.@min_value+" ถึง "+.@max_value;
		switch (select(
			"~ สีถัดไป",
			"~ สีก่อนหน้า",
			"~ ระบุเลขที่ต้องการ",
			"~ ยืนยันสีนี้",
			"~ ยกเลิก"
		)) {
			case 1:
				.@preview_value = (.@preview_value < .@max_value) ? .@preview_value + 1 : .@min_value;
				break;
			case 2:
				.@preview_value = (.@preview_value > .@min_value) ? .@preview_value - 1 : .@max_value;
				break;
			case 3:
				next;
				mes "^FF0000[ Stylist ]^000000";
				mes "กรุณากรอกหมายเลข"+.@type_name$;
				mes "หมายเลข: "+.@min_value+" ถึง "+.@max_value;
				input .@preview_value,.@min_value,.@max_value;
				break;
			case 4:
				if (countitem(.@coupon_id) < 1) {
					changelook .@look_type,.@old_value;
					next;
					mes "^FF0000[ Stylist ]^000000";
					mes "ไม่พบ ^FF0000"+getitemname(.@coupon_id)+" 1ea^000000";
					mes "ระบบคืน"+.@type_name$+"เดิมให้คุณแล้ว";
					next;
					return 0;
				}
				delitem .@coupon_id,1;
				setlook .@look_type,.@preview_value;
				next;
				mes "^FF0000[ Stylist ]^000000";
				mes "ยืนยัน"+.@type_name$+"ใหม่เรียบร้อยแล้ว";
				next;
				return 1;
			case 5:
				changelook .@look_type,.@old_value;
				next;
				mes "^FF0000[ Stylist ]^000000";
				mes "ยกเลิกแล้ว ระบบคืน"+.@type_name$+"เดิมให้คุณ";
				next;
				return 0;
		}
	}
}
