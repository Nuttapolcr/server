
// PK PASS Function

-	script	PKPASS	-1,{


OnInit:
set $pkpass_hateffect_attack, 242; //เลขhatteffect สำหรับ attacker pass
set $pkpass_hateffect_defend, 92; //เลขhatteffect สำหรับ protect,defend pass
end;



OnPCDieEvent:
OnPCLoginEvent:
	sc_end SC_PK_ATTACK;
	sc_end SC_PK_DEFEND;
	hateffect $pkpass_hateffect_defend, false;
    hateffect $pkpass_hateffect_attack, false;
	end;

OnPCLoadMapEvent:
	// เข้าเมือง (town) หรือเข้าแมพที่ไม่อนุญาต pkpass ให้ล้างสถานะทันที
	if (getmapflag(strcharinfo(3), MF_TOWN) || !getmapflag(strcharinfo(3), MF_PKPASS)) {
		deltimer "PKPASS::OnPKPASSEND";
		sc_end SC_PK_ATTACK;
		sc_end SC_PK_DEFEND;
		hateffect $pkpass_hateffect_defend, false;
		hateffect $pkpass_hateffect_attack, false;
		dispbottom "[PK PASS] ออกจากพื้นที่ PK PASS แล้ว ระบบปิดใช้งานและล้างสถานะเรียบร้อย";
	}
	end;


OnPKPASSEND:
	sc_end SC_PK_ATTACK;
	sc_end SC_PK_DEFEND;
	hateffect $pkpass_hateffect_defend, false;
    hateffect $pkpass_hateffect_attack, false;
    end;
}

function	script	PK_Defend_Pass	{
    // ตรวจสอบว่าแมพอนุญาตให้ใช้หรือไม่
	if (!getmapflag(strcharinfo(3), MF_PKPASS)) {
        dispbottom "[PK PASS] ไม่สามารถใช้ได้กับแผนที่นี้!";
        getitem 40001,1;
        end;
    }
    
    // ตรวจสอบ cooldown
    .@cooldown_time = gettimetick(2) - @pk_defend_last_use;
    if (.@cooldown_time < 1800) { // 600 วินาที = 10 นาที
        .@remaining = 1800 - .@cooldown_time;
        .@minutes = .@remaining / 60;
        .@seconds = .@remaining % 60;
        
        if (.@minutes > 0) {
            dispbottom "[PK PASS] โปรดรออีก " + .@minutes + " นาที " + .@seconds + " วินาที ถึงจะใช้งานได้อีกครั้ง";
        } else {
            dispbottom "[PK PASS] โปรดรออีก " + .@seconds + " วินาที ถึงจะใช้งานได้อีกครั้ง";
        }
        getitem 40001,1;
        end;
    }
    
    // บันทึกเวลาที่ใช้งาน
    @pk_defend_last_use = gettimetick(2);
    
	deltimer "PKPASS::OnPKPASSEND";
	dispbottom " "+$pkpass_hateffect_defend+" ";
	sc_end SC_PK_ATTACK;
	hateffect $pkpass_hateffect_attack, false;
    hateffect $pkpass_hateffect_defend, true;
	dispbottom "[PK PASS] เปิดใช้งาน PK Defender ได้รับการป้องกัน PK";
	showscript "ใช้งาน PK Defender";
	sc_start SC_PK_DEFEND,1800000,1;
    addtimer 1800000, "PKPASS::OnPKPASSEND";
    end;
}

function	script	PK_Attack_Pass	{

        dispbottom "[PK PASS] ไม่สามารถใช้ได้กับแผนที่นี้!";

    }
    
    // ตรวจสอบ cooldown

        .@remaining = 900 - .@cooldown_time;
        .@minutes = .@remaining / 60;
        .@seconds = .@remaining % 60;
        
        if (.@minutes > 0) {
            dispbottom "[PK PASS] โปรดรออีก " + .@minutes + " นาที " + .@seconds + " วินาที ถึงจะใช้งานได้อีกครั้ง";

        }
        getitem 40000,1;
        end;
    }
    
    // บันทึกเวลาที่ใช้งาน
    @pk_attack_last_use = gettimetick(2);
    
	deltimer "PKPASS::OnPKPASSEND";
	sc_end SC_PK_DEFEND;
	hateffect $pkpass_hateffect_defend, false;
    hateffect $pkpass_hateffect_attack, true;
	dispbottom "[PK PASS] เปิดใช้งาน PK Attacker สามารถโจมตีผู้เล่นได้";
	showscript "ใช้งาน PK Attacker";
	sc_start SC_PK_ATTACK,900000,1;
    addtimer 900000, "PKPASS::OnPKPASSEND";
    end;
}
