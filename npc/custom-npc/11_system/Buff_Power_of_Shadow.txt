function	script	F_PowerShadow_RemainTimePart$	{
	.@part = getarg(0);
	.@now = gettimetick(2);
	.@end = 0;

	if (.@part == 1) .@end = #POWER_SHADOW_END_ARMOR;
	else if (.@part == 2) .@end = #POWER_SHADOW_END_SHIELD;
	else if (.@part == 3) .@end = #POWER_SHADOW_END_SHOES;
	else if (.@part == 4) .@end = #POWER_SHADOW_END_WEAPON;

	.@sec = .@end - .@now;
	if (.@sec <= 0) return "0 นาที";

	.@hour = .@sec / 3600;
	.@min  = (.@sec % 3600) / 60;

	if (.@hour > 0 && .@min > 0) return .@hour + " ชม " + .@min + " นาที";
	if (.@hour > 0) return .@hour + " ชม";
	return .@min + " นาที";
}

function	script	F_PowerShadow_RemainTime$	{
	.@now = gettimetick(2);
	.@end = 0;

	if (#POWER_SHADOW_END_ARMOR  > .@end) .@end = #POWER_SHADOW_END_ARMOR;
	if (#POWER_SHADOW_END_SHIELD > .@end) .@end = #POWER_SHADOW_END_SHIELD;
	if (#POWER_SHADOW_END_SHOES  > .@end) .@end = #POWER_SHADOW_END_SHOES;
	if (#POWER_SHADOW_END_WEAPON > .@end) .@end = #POWER_SHADOW_END_WEAPON;

	.@sec = .@end - .@now;
	if (.@sec <= 0) return "0 นาที";

	.@hour = .@sec / 3600;
	.@min  = (.@sec % 3600) / 60;

	if (.@hour > 0 && .@min > 0) return .@hour + " ชม " + .@min + " นาที";
	if (.@hour > 0) return .@hour + " ชม";
	return .@min + " นาที";
}

function	script	F_PowerShadow_ActiveList	{
	.@now = gettimetick(2);
	.@cnt = 0;

	if (#POWER_SHADOW_END_ARMOR > .@now && #POWER_SHADOW_LAST_ARMOR$ != "") {
		.@line$[.@cnt++] = "Armor : " + #POWER_SHADOW_LAST_ARMOR$ + " (เหลือ " + callfunc("F_PowerShadow_RemainTimePart$",1) + ")";
	}
	if (#POWER_SHADOW_END_SHIELD > .@now && #POWER_SHADOW_LAST_SHIELD$ != "") {
		.@line$[.@cnt++] = "Shield : " + #POWER_SHADOW_LAST_SHIELD$ + " (เหลือ " + callfunc("F_PowerShadow_RemainTimePart$",2) + ")";
	}
	if (#POWER_SHADOW_END_SHOES > .@now && #POWER_SHADOW_LAST_SHOES$ != "") {
		.@line$[.@cnt++] = "Shoes : " + #POWER_SHADOW_LAST_SHOES$ + " (เหลือ " + callfunc("F_PowerShadow_RemainTimePart$",3) + ")";
	}
	if (#POWER_SHADOW_END_WEAPON > .@now && #POWER_SHADOW_LAST_WEAPON$ != "") {
		.@line$[.@cnt++] = "Weapon : " + #POWER_SHADOW_LAST_WEAPON$ + " (เหลือ " + callfunc("F_PowerShadow_RemainTimePart$",4) + ")";
	}

	if (.@cnt == 0) return 0;

	mes "บัฟที่กำลังมีอยู่:";
	for (.@i = 0; .@i < .@cnt; .@i++) mes "- " + .@line$[.@i];
	next;
	return 1;
}

function	script	F_ClearPowerShadowSC_Part	{
	.@part = getarg(0); // 1=Armor 2=Shield 3=Shoes 4=Weapon

	if (.@part == 1) callfunc "F_ClearPowerShadowSC_Armor";
	else if (.@part == 2) callfunc "F_ClearPowerShadowSC_Shield";
	else if (.@part == 3) callfunc "F_ClearPowerShadowSC_Shoes";
	else if (.@part == 4) callfunc "F_ClearPowerShadowSC_Weapon";

	return;
}
