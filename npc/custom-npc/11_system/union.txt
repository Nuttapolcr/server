//! rAthena

-	script	UnionSystem::UnionSystem	-1,{

OnPCLoginEvent:
	callsub S_ReapplyNow;
	end;

OnPCBaseLvUpEvent:
	callsub S_ReapplyNow;
	end;

OnPCJobLvUpEvent:
	callsub S_ReapplyNow;
	end;

S_ReapplyNow:
	callsub S_ClearPerClass;

	callsub S_ApplyClass, @UNION_KNIGHT    , SC_UNION_KNIGHT_90    , SC_UNION_KNIGHT_95    , SC_UNION_KNIGHT_99;
	callsub S_ApplyClass, @UNION_PRIEST    , SC_UNION_PRIEST_90    , SC_UNION_PRIEST_95    , SC_UNION_PRIEST_99;
	callsub S_ApplyClass, @UNION_WIZARD    , SC_UNION_WIZARD_90    , SC_UNION_WIZARD_95    , SC_UNION_WIZARD_99;
	callsub S_ApplyClass, @UNION_HUNTER    , SC_UNION_HUNTER_90    , SC_UNION_HUNTER_95    , SC_UNION_HUNTER_99;
	callsub S_ApplyClass, @UNION_ASSASSIN  , SC_UNION_ASSASSIN_90  , SC_UNION_ASSASSIN_95  , SC_UNION_ASSASSIN_99;
	callsub S_ApplyClass, @UNION_BLACKSMITH, SC_UNION_BLACKSMITH_90, SC_UNION_BLACKSMITH_95, SC_UNION_BLACKSMITH_99;

	sc_end SC_UNION_LEVEL_1;
	sc_end SC_UNION_LEVEL_2;
	sc_end SC_UNION_LEVEL_3;
	set .@tier, callsub(S_CalculateTotalTier, @UNION_TOTAL_LEVEL);
	if (.@tier == 1)      sc_start SC_UNION_LEVEL_1, INFINITE_TICK, 1;
	else if (.@tier == 2) sc_start SC_UNION_LEVEL_2, INFINITE_TICK, 1;
	else if (.@tier == 3) sc_start SC_UNION_LEVEL_3, INFINITE_TICK, 1;
	return;

S_ApplyClass:
	.@lv  = getarg(0);
	.@sc90 = getarg(1); // 90..94
	.@sc95 = getarg(2); // 95..98
	.@sc99 = getarg(3); // 99

	if (.@lv == 99)
		sc_start .@sc99, INFINITE_TICK, 1;
	else if (.@lv >= 95 && .@lv <= 98)
		sc_start .@sc95, INFINITE_TICK, 1;
	else if (.@lv >= 90 && .@lv <= 94)
		sc_start .@sc90, INFINITE_TICK, 1;
	return;

S_ClearPerClass:
	sc_end SC_UNION_KNIGHT_90;     sc_end SC_UNION_KNIGHT_95;     sc_end SC_UNION_KNIGHT_99;
	sc_end SC_UNION_KNIGHT;
	sc_end SC_UNION_PRIEST_90;     sc_end SC_UNION_PRIEST_95;     sc_end SC_UNION_PRIEST_99;
	sc_end SC_UNION_PRIEST;
	sc_end SC_UNION_WIZARD_90;     sc_end SC_UNION_WIZARD_95;     sc_end SC_UNION_WIZARD_99;
	sc_end SC_UNION_WIZARD;
	sc_end SC_UNION_HUNTER_90;     sc_end SC_UNION_HUNTER_95;     sc_end SC_UNION_HUNTER_99;
	sc_end SC_UNION_HUNTER;
	sc_end SC_UNION_ASSASSIN_90;   sc_end SC_UNION_ASSASSIN_95;   sc_end SC_UNION_ASSASSIN_99;
	sc_end SC_UNION_ASSASSIN;
	sc_end SC_UNION_BLACKSMITH_90; sc_end SC_UNION_BLACKSMITH_95; sc_end SC_UNION_BLACKSMITH_99;
	sc_end SC_UNION_BLACKSMITH;
	return;

S_CalculateTotalTier:
	.@lv = getarg(0);
	if (.@lv >= 594)      return 3;
	else if (.@lv >= 570) return 2;
	else if (.@lv >= 540) return 1;
	return 0;
}
