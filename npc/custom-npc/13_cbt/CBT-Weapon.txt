OnBuyItem:
	sleep2 100;
	mes "[CBT Weapon]";
	mes "ท่านต้องการกี่ชิ้นต่อชนิด?";
	input .@amount;
	if (.@amount < 1) set .@amount,1;
	if (.@amount > 5) set .@amount,5;

	.@refine = 0;

	for (.@i = 0; .@i < getarraysize(@bought_nameid); .@i++) {
		if (@bought_nameid[.@i] <= 0) continue;
		if (!checkweight(@bought_nameid[.@i], .@amount)) {
			mes "^FF0000น้ำหนักเกิน กรุณาลดจำนวนหรือเคลียร์ช่องกระเป๋าก่อน^000000";
			close;
		}
	}

	for (.@i = 0; .@i < getarraysize(@bought_nameid); .@i++) {
		if (@bought_nameid[.@i] <= 0) continue;
		getitem2 @bought_nameid[.@i], .@amount, 1, .@refine, 0,0,0,0,0;
	}

	close;

OnInit:
	waitingroom "     [Weapon]",0;
	end;
