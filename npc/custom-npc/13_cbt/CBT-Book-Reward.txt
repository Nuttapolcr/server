morocc,171,87,4	script	NPC รับรางวัล CBT Book	617,{
	if (getmapflag(strcharinfo(3),MF_PVP) || getmapflag(strcharinfo(3),MF_GVG) || getmapflag(strcharinfo(3),MF_GVG_CASTLE))
		end;

	mes .name$;
	.@rows = query_sql("SELECT `book_point`, `reward_tier`, `claimed` FROM `cbt_book_rewards` WHERE `account_id` = "+getcharid(3)+" LIMIT 1", .@book_point, .@reward_tier, .@claimed);
	if (!.@rows) {
		mes "ไม่พบสิทธิ์รับรางวัล CBT Book ของบัญชีนี้";
		close;
	}
	mes "ยอด CBT Book สะสมในบัญชี: ^00AA00"+.@book_point+"^000000 เล่ม";
	mes "รับรางวัลสูงสุดตามยอดสะสมได้เพียงครั้งเดียว";
	next;

	if (.@claimed) {
		mes .name$;
		mes "บัญชีนี้รับรางวัล CBT Book ไปแล้ว";
		close;
	}
	callsub S_ClaimReward, .@reward_tier, .@book_point;
	close;

S_ClaimReward:
	.@index = getarg(0);
	.@book_point = getarg(1);
	.@rows = query_sql("SELECT `claimed` FROM `cbt_book_rewards` WHERE `account_id` = "+getcharid(3)+" LIMIT 1", .@claimed);
	if (!.@rows || .@claimed) {
		mes .name$;
		mes "บัญชีนี้รับรางวัล CBT Book ไปแล้ว";
		return;
	}

	switch (.@index) {
	case 0:
		getitem 1001720,1;
		getitem 1000603,10;
		break;
	case 1:
		getitem 1001720,1;
		getitem 1000603,15;
		break;
	case 2:
		getitem 1001720,1;
		getitem 1000603,10;
		getitem 22571,1;
		break;
	case 3:
		getitem 1001721,1;
		getitem 1000603,10;
		getitem 22571,1;
		getitem 101194,1;
		getitem 12804,1;
		getitem 14455,1;
		break;
	case 4:
		getitem 1001721,2;
		getitem 1000603,30;
		getitem 22571,2;
		getitem 101194,2;
		getitem 12804,2;
		getitem 14455,2;
		getitem 1030007,1;
		break;
	}
	query_sql("UPDATE `cbt_book_rewards` SET `claimed` = 1, `claimed_char_id` = "+getcharid(0)+", `claimed_at` = NOW() WHERE `account_id` = "+getcharid(3)+" AND `claimed` = 0 LIMIT 1");
	mes .name$;
	mes "รับรางวัลจากยอด ^00AA00"+.@book_point+"^000000 เล่มเรียบร้อยแล้ว";
	return;

OnInit:
	.name$ = "[^0000FFNPC รับรางวัล CBT Book^000000]";
	query_sql("CREATE TABLE IF NOT EXISTS `cbt_book_rewards` (`account_id` int(11) unsigned NOT NULL, `book_point` bigint(11) unsigned NOT NULL, `reward_tier` tinyint(1) unsigned NOT NULL, `claimed` tinyint(1) unsigned NOT NULL DEFAULT 0, `claimed_char_id` int(11) unsigned DEFAULT NULL, `snapshot_at` datetime NOT NULL, `claimed_at` datetime DEFAULT NULL, PRIMARY KEY (`account_id`)) ENGINE=MyISAM DEFAULT CHARSET=utf8;");
	query_sql("CREATE TABLE IF NOT EXISTS `cbt_book_reward_meta` (`key_name` varchar(32) NOT NULL, `value` varchar(32) NOT NULL, PRIMARY KEY (`key_name`)) ENGINE=MyISAM DEFAULT CHARSET=utf8;");
	query_sql("SELECT `value` FROM `cbt_book_reward_meta` WHERE `key_name` = 'book_snapshot_done' LIMIT 1", .@snapshot_done$);
	if (.@snapshot_done$ != "4") {
		query_sql("DELETE FROM `cbt_book_rewards`");
		query_sql("INSERT IGNORE INTO `cbt_book_rewards` (`account_id`, `book_point`, `reward_tier`, `snapshot_at`) SELECT `account_id`, `value`, CASE WHEN `value` >= 5000 THEN 4 WHEN `value` >= 3000 THEN 3 WHEN `value` >= 1500 THEN 2 WHEN `value` >= 1000 THEN 1 ELSE 0 END, NOW() FROM `acc_reg_num` WHERE `key` = '#hiclass_cbt_book_point' AND `index` = 0 AND `value` >= 500");
		query_sql("REPLACE INTO `cbt_book_reward_meta` (`key_name`, `value`) VALUES ('book_snapshot_done', '4')");
	}
	query_sql("INSERT INTO `cbt_book_rewards` (`account_id`, `book_point`, `reward_tier`, `snapshot_at`) SELECT `account_id`, 5000, 4, NOW() FROM `login` WHERE `userid` IN ('gmtest01','gmtest02','gmtest03') ON DUPLICATE KEY UPDATE `book_point` = 5000, `reward_tier` = 4");
	waitingroom "NPC รับรางวัล CBT Book",0;
	end;
