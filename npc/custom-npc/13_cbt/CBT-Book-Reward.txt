// ระบบส่ง HICLASS CBT Book เพื่อสะสมแต้ม
// ตัวแปรบัญชี: #hiclass_cbt_book_point

morocc,171,87,4	script	NPC แลกแต้ม CBT Book	617,{
	if (getmapflag(strcharinfo(3),MF_PVP) || getmapflag(strcharinfo(3),MF_GVG) || getmapflag(strcharinfo(3),MF_GVG_CASTLE))
		end;

	.@book_id = 905027; // CBT Book
	.@book_count = countitem(.@book_id);

	mes .name$;
	mes "ส่ง CBT Book เพื่อสะสมแต้มในบัญชี";
	mes "CBT Book ที่มีตอนนี้: ^00AA00"+.@book_count+"^000000";
	mes "แต้มบัญชีของคุณ: ^00AA00"+#hiclass_cbt_book_point+"^000000";
	next;

	switch(select(
		"ส่ง CBT Book ทั้งหมด",
		"ส่งตามจำนวนที่กำหนด",
		"ดูแต้มของฉัน",
		"ยกเลิก"
	)) {
	case 1:
		if (.@book_count <= 0) {
			callsub S_NotEnough, .@book_id;
			close;
		}
		callsub S_DepositBook, .@book_id, .@book_count;
		close;
	case 2:
		if (.@book_count <= 0) {
			callsub S_NotEnough, .@book_id;
			close;
		}
		mes .name$;
		mes "กรุณาใส่จำนวนที่ต้องการส่ง";
		mes "จำนวนที่มี: ^00AA00"+.@book_count+"^000000";
		next;
		input .@amount, 1, .@book_count;
		callsub S_DepositBook, .@book_id, .@amount;
		close;
	case 3:
		callsub S_ShowPoints;
		close;
	default:
		close;
	}

S_DepositBook:
	.@item_id = getarg(0);
	.@amount = getarg(1);
	if (.@amount <= 0)
		return;
	if (countitem(.@item_id) < .@amount) {
		callsub S_NotEnough, .@item_id;
		return;
	}
	delitem .@item_id, .@amount;
	#hiclass_cbt_book_point += .@amount;
	mes .name$;
	mes "ส่ง CBT Book แล้ว: ^00AA00"+.@amount+"^000000 เล่ม";
	mes "แต้มบัญชีปัจจุบัน: ^00AA00"+#hiclass_cbt_book_point+"^000000";
	return;

S_NotEnough:
	mes .name$;
	mes "คุณไม่มี "+getitemname(getarg(0))+".";
	return;

S_ShowPoints:
	mes .name$;
	mes "แต้มบัญชีของคุณ: ^00AA00"+#hiclass_cbt_book_point+"^000000";
	return;

OnInit:
	.name$ = "[^0000FFNPC แลกแต้ม CBT Book^000000]";
	waitingroom "NPC แลกแต้ม CBT Book",0;
	end;
}
