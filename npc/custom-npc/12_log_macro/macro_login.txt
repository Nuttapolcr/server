-	script	Macro_Login	-1,{
end;
	

OnPCLoginEvent:
	// เรียก NPC ตรวจคนเข้าเมือง
	checker:
		mes "[ เจ้าหน้าที่ตรวจคนเข้าเมือง ]";
		mes "ตามคำสั่งของกษัตริย์ ผู้เดินทางต้องยืนยันตัวตนก่อนเข้าเมือง";
		mes "กรุณากรอกหมายเลขที่แสดงด้านล่าง เพื่อยืนยันว่าท่านไม่ใช่ผู้บุกรุก";
		set .@Code, rand(100,999); // สุ่มรหัส 3 หลัก
		mes "^FF0000" + .@Code + "^000000";
		input .@Input;

		if (.@Input != .@Code) {
			clear;
			mes "[เจ้าหน้าที่ตรวจคนเข้าเมือง]";
			mes "ขออภัย... รหัสไม่ถูกต้อง";
			mes "โปรดลองอีกครั้ง";
			mes "^FF0000" + .@Code + "^000000";
			input .@Input2;

			if (.@Input2 != .@Code) {
			clear;
				//mes "[เจ้าหน้าที่ตรวจคนเข้าเมือง]";
				//mes "ท่านไม่สามารถยืนยันตัวตนได้";
				//mes "กรุณาออกจากเขตแดนของเรา และกลับมาใหม่เมื่อพร้อม";
				//atcommand "@kick " + strcharinfo(0);
				//end;
				goto checker;
			}
		}
		clear;
		mes "[เจ้าหน้าที่ตรวจคนเข้าเมือง]";
		mes "การยืนยันตัวตนสำเร็จแล้ว";
		mes "ขอให้การเดินทางของท่านเป็นไปอย่างราบรื่น";
		close;

}
