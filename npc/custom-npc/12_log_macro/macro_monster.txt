-	script	Macro_monster	FAKE_NPC,{
OnNPCKillEvent:
	set .@gained,1;
	set count, count + .@gained;	
	if ( count >= 1000 ) // ตรวจเมื่อจบม่อนทุก 1000 ตัว
		doevent .name$+"::OnBlock";
	end;

OnBlock:

	sleep2 300;
	if (hide == 0 ) { 
	atcommand "@hide";
	set hide, 1;
	}
	sc_start SC_FREEZE, 1000000000, 1;
	sc_start SC_BERSERK, 1000000000, 1;
	sleep2 100;


	mes "^FF0000 คุณมีระยะเวลา 60 วินาที ในการซื้อของให้ถูกต้อง ห้ามวิง ห้ามออกเกมส์ โดยเด็ดขาด มิฉะนั้นจะถูกส่งเข้าคุกในทันที เมื่อครบระยะเวลา ^000000";
	deltimer .name$+"::OnJail";
	addtimer (.timer*1000),.name$+"::OnJail";
	@rnd_item = .itemid[rand(.item)];
	@rnd_amt = rand(1,100);
	mes "' กรุณาซื้อ "+ getitemname(@rnd_item) +" จำนวน "+ @rnd_amt +" '";
	npcshopattach "anti_bot",1;
	callshop "anti_bot",1;
	end;
	
OnBuyItem:
	if ( @bought_nameid[0] != @rnd_item || @bought_quantity[0] != @rnd_amt )
	{
		mes "คุณได้ทำการเลือกซื้อผิด โปรดลองใหม่อีกครั้ง"; close2; doevent .name$+"::OnBlock"; end;
	}
	if (hide == 1) { 
	atcommand "@hide";
	set hide, 0;
	}
	sc_end SC_FREEZE; 
	sc_end SC_BERSERK;	
	percentheal 0,100;
	set count,0;
	deltimer .name$+"::OnJail";
	end;
	
OnJail:
	if (hide == 1) { 
	sc_end SC_FREEZE; 
	sc_end SC_BERSERK;	
	atcommand "@hide";
	set hide, 0;
	}
	atcommand "@jail "+ strcharinfo(0);
	end;
	
OnPCLoginEvent:
	if (hide == 1) { 
	sc_end SC_FREEZE; 
	sc_end SC_BERSERK;	
	atcommand "@hide";
	set hide, 0;
	sleep 1000;
	atcommand "@jail "+ strcharinfo(0);
	}
	end;


OnInit:
	.monster_min = 1;
	.monster_max = 10;
	.timer = 60;
	.name$ = strnpcinfo(3);
	setarray .itemid,501,502,503,504,505,506,507,508,509,510;
	.item = getarraysize(.itemid);
	npcshopdelitem "anti_bot",501;
	for ( .@i = 0; .@i < .item; .@i++ )
		npcshopadditem "anti_bot", .itemid[.@i], 0; // Price set to 0
	end;
}

-	shop	anti_bot	FAKE_NPC,501:-1
